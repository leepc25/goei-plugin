        const after = cur.slice(i + 1).find((t) => t && find(t) >= 0);
        if (before) swap(find(before), before, `${before}

${v}`);
        else if (after) swap(find(after), after, `${v}

${after}`);
        else return;
      }
      cur[i] = v;
      changed++;
    });
