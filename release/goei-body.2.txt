  function wrapText(g2, text4, maxWidth, maxLines) {
    const src = text4.replace(/\s+/g, " ").trim();
    if (!src) return [];
    const lines3 = [];
    let cur = "";
    for (const ch of src) {
      const next = cur + ch;
      if (g2.measureText(next).width <= maxWidth || !cur) cur = next;
      else {
        lines3.push(cur);
        cur = ch;
        if (lines3.length === maxLines) break;
      }
    }
    if (lines3.length < maxLines) {
      if (cur) lines3.push(cur);
      return lines3;
    }
    let last3 = lines3[maxLines - 1];
    while (last3.length && g2.measureText(`${last3}…`).width > maxWidth) last3 = last3.slice(0, -1);
    lines3[maxLines - 1] = `${last3}…`;
    return lines3.slice(0, maxLines);
  }
  function roundRect(g2, x, y, w, h, r) {
    const rr = Math.min(r, w / 2, h / 2);
    g2.beginPath();
    g2.moveTo(x + rr, y);
    g2.lineTo(x + w - rr, y);
    g2.quadraticCurveTo(x + w, y, x + w, y + rr);

  }
