    const v = value ?? "";

      try {

      return true;
    }
    switch (action) {

        go("home", "", "");
        wiki.history = [];
        wiki.listOpen = false;
        break;
      case "wiki-back": {
        const prev = wiki.history.pop();
        if (prev && prev.view !== "home") Object.assign(wiki, { view: prev.view, title: prev.title }, { confirmDelete: null, notice: "" });
