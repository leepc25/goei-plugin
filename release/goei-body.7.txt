  async function handlePhone(action, value, ctx2, env3) {
    if (!action.startsWith("ph-")) return false;
    liveEnv = env3;
    const v = value ?? "";
    if (!["ph-room-del", "ph-view-del", "ph-contact-del", "ph-notice-del"].includes(action)) phone.confirm = null;
    if (action.startsWith("ph-def-")) {
      try {
        const msg = await defaultsAction("ph", action, v, ctx2, env3.refresh);
        if (msg) phone.notice = msg;
      } catch (error) {
        phone.notice = `기본값을 처리하지 못했어요: ${error instanceof Error ? error.message : String(error)}`;
      }
      env3.render();
      return true;
    }
    switch (action) {
      case "ph-sub":
        Object.assign(phone, { sub: v, query: "", ...phone.view === "room" ? {} : { view: "list", invite: null, picked: [] } });
        break;
      case "ph-settings":
        Object.assign(phone, { view: phone.view === "settings" ? "list" : "settings", invite: null, picked: [] });
        break;
      case "ph-att-menu":
        phone.attachMenu = !phone.attachMenu;
        break;
      case "ph-ctab":
        phone.tab = v;
        phone.query = "";
        break;
      case "ph-chat-filter":
        phone.chatFilter = v === "unread" ? "unread" : "all";
        break;
      case "ph-notice-filter":
        phone.noticeFilter = v === "unread" ? "unread" : "all";
        phone.noticePage = 1;
        break;
      case "ph-notice-faction":
        phone.noticeFaction = phone.noticeFaction === v ? "" : v;
        phone.noticePage = 1;
        break;
      case "ph-pick-tab":
        phone.pickTab = v;
        phone.pickQuery = "";
        break;
      case "ph-dm":
        if (phone.view === "room" && phone.room === `dm_${v}` && !isPhone()) return handlePhone("ph-back", void 0, ctx2, env3);
        await openDm(ctx2, env3, v);
        return true;
      case "ph-room":
        if (phone.view === "room" && phone.room === v && !isPhone()) return handlePhone("ph-back", void 0, ctx2, env3);
        await openRoom(ctx2, env3, v);
        return true;
      case "ph-back":
        if (phone.view === "group" && phone.invite) {
          Object.assign(phone, { view: "room", room: phone.invite, invite: null, picked: [] });
          break;
        }
        if (phone.view === "room" && phone.room) {
          const rid = phone.room;
          Object.assign(phone, { view: "list", room: null, draft: "", attach: [] });
          await mutate(ctx2, env3, (st) => {
            const r = st.rooms[rid];
            if (!r || r.messages.length || r.pending) return false;
            delete st.rooms[rid];
            st.order = st.order.filter((x) => x !== rid);
          });
          return true;
        }
        Object.assign(phone, { view: "list", invite: null, picked: [] });
        break;
      case "ph-more":
        phone.page += 1;
        break;
      case "ph-att-add":
      case "ph-att-del": {
        const r = readPhone(ctx2.chat).rooms[phone.room ?? ""];
        if (r) attachAction(phone.attach, action === "ph-att-add" ? "add" : "del", v, phKinds(r));
        break;
      }
      case "ph-send": {
        if (!phone.draft.trim() && !phone.attach.length) {
          phone.notice = "메시지를 입력해 주세요.";
          break;
        }
        if (emptyAttachment(phone.attach)) {
          phone.notice = "첨부 내용을 입력하거나 ✕로 빼 주세요.";
          break;
        }
        void send(ctx2, env3, v, composeBody(phone.attach, phone.draft));
        return true;
      }
      case "ph-retry": {
        const r = readPhone(ctx2.chat).rooms[v];
        if (r?.pending) void generate2(ctx2, env3, v, `${roomLabel(r)}의 답장을 다시 기다리는 중이에요.`);
        return true;
      }
      case "ph-skip":
        if (phoneBusy(ctx2)) break;
        await mutate(ctx2, env3, (st) => {
          const r = st.rooms[v];
          if (!r?.pending) return false;
          r.pending = void 0;
        });
        return true;
      case "ph-room-refresh":
        void roomRefresh(ctx2, env3, v);
        return true;
      case "ph-refresh":
        void refreshInbox(ctx2, env3);
        return true;
      case "ph-view-new":
        void makeView(ctx2, env3);
        return true;
      case "ph-view-open": {
        const list4 = readViews(ctx2.chat.scriptstate);
        const r = list4.find((x) => x.id === v);
        if (!r) break;
        if (r.unread) {
          r.unread = false;
          await setVars(ctx2, env3, writeViews(list4));
        }
        Object.assign(phone, { sub: "chats", view: "view", viewId: v, confirm: null });
        break;
      }
      case "ph-view-del": {
        if (phone.confirm !== `view:${v}`) {
          phone.confirm = `view:${v}`;
          break;
        }
        phone.confirm = null;
        await setVars(ctx2, env3, writeViews(readViews(ctx2.chat.scriptstate).filter((x) => x.id !== v)));
        if (phone.viewId === v) Object.assign(phone, { view: phone.view === "view" ? "list" : phone.view, viewId: null });
        return true;
      }
      case "ph-group":
        Object.assign(phone, { view: "group", invite: null, picked: [], topic: "", gname: "", pickQuery: "", pickTab: "" });
        break;
      case "ph-invite":
        Object.assign(phone, { view: "group", invite: v, room: v, picked: [], pickQuery: "", pickTab: "" });
        break;
      case "ph-pick": {
        const set2 = new Set(phone.picked);
        if (set2.has(v)) set2.delete(v);
        else {
          const existing = phone.invite ? readPhone(ctx2.chat).rooms[phone.invite]?.members.length ?? 0 : 0;
          if (set2.size + existing >= LIMITS2.members) {
            phone.notice = `상대는 최대 ${LIMITS2.members}명까지예요.`;
            break;
          }
          set2.add(v);
        }
        phone.picked = [...set2];
        break;
      }
      case "ph-group-create":
        await createGroup(ctx2, env3);
        return true;
      case "ph-room-del":
        if (phone.confirm !== `room:${v}`) {
          phone.confirm = `room:${v}`;
          break;
        }
        phone.confirm = null;
        await mutate(ctx2, env3, (st) => {
          const r = st.rooms[v];
          if (!r || r.pending) return false;
          delete st.rooms[v];
          st.order = st.order.filter((x) => x !== v);
        });
        if (phone.room === v) Object.assign(phone, { view: phone.view === "room" ? "list" : phone.view, room: null });
        return true;
      case "ph-notice": {
        const id2 = Number(v);
        if (isPhone()) Object.assign(phone, { openNotice: phone.openNotice === id2 ? null : id2, view: phone.view === "notice" ? "list" : phone.view });
        else if (phone.view === "notice" && phone.openNotice === id2) Object.assign(phone, { openNotice: null, view: "list" });
        else Object.assign(phone, { openNotice: id2, view: "notice" });
        {

          const row11 = b.items.find((x) => x.id === id2);
          if (row11 && row11.read === false) {

          }
        }
        break;
      }
      case "ph-notice-page":
        phone.noticePage = Math.max(1, Number(v) || 1);
        phone.openNotice = null;
        break;
      case "ph-notice-all": {
        const b = readBulletins(ctx2.chat);
        if (!b.items.some((x) => x.read === false)) break;
        for (const row11 of b.items) row11.read = true;
        await setVars(ctx2, env3, writeBulletins(b, ctx2.chat.scriptstate));
        return true;
      }
      case "ph-notice-del": {
        if (phone.confirm !== `notice:${v}`) {
          phone.confirm = `notice:${v}`;
          break;
        }
        phone.confirm = null;
        const b = readBulletins(ctx2.chat);
        b.items = b.items.filter((x) => x.id !== Number(v));
        if (phone.openNotice === Number(v)) Object.assign(phone, { openNotice: null, view: phone.view === "notice" ? "list" : phone.view });
        await setVars(ctx2, env3, writeBulletins(b, ctx2.chat.scriptstate));
        return true;
      }
      case "ph-call":
        Object.assign(phone, { view: "call", call: v, callText: "" });
        break;
      case "ph-call-place":
        await placeCall(ctx2, env3, phone.call, phone.callText);
        return true;
      case "ph-add":
        Object.assign(phone, { view: "add", addName: "", addNote: "" });
        break;
      case "ph-add-save":
        await addContact(ctx2, env3);
        return true;
      case "ph-contact-del":
        if (phone.confirm !== `contact:${v}`) {
          phone.confirm = `contact:${v}`;
          break;
        }
        phone.confirm = null;
        await deleteContact(ctx2, env3, v);
        return true;
      case "ph-cfg": {
        const [k, val] = v.split("=");
        if (k === "joint") {
          await setVars(ctx2, env3, { [JOINT_VAR]: val === "0" ? "0" : "1" });
          return true;
        }
        const vars = { [PH(k)]: val };
        if (k === "all_contacts" || k === "anomaly_contacts") phone.tab = "";
        if (k === "announcement_page_size") phone.noticePage = 1;
        await setVars(ctx2, env3, vars);
        return true;
      }
      case "ph-text": {
        const [k, ...rest] = v.split("=");
        if (k === "view_pov") await setVars(ctx2, env3, { [PH(k)]: rest.join("=").trim().slice(0, 40) });
        return true;
      }
      case "ph-num": {

        const range = {
          incoming_chance: [0, 100],
          group_invite_chance: [0, 100],
          banter_chance: [0, 100],
          banter_min: BANTER_ROWS,
          banter_max: BANTER_ROWS,

  async function addContact(ctx2, env3) {
    const name = phone.addName.trim();
    if (!validCallTarget(name)) {
      phone.notice = "이름은 60자 이내로, 줄바꿈·괄호 없이 적어 주세요.";
      return env3.render();
    }
    if (tokens(phone.addNote) > LIMITS2.memo_tokens) {

      return env3.render();
    }
    const d = dir(ctx2);
    const key = name.toLowerCase();
    const found = [...new Set(d.map.values())].filter((c) => c.name === name || (c.english ?? "").toLowerCase() === key || c.terms.some((t) => t.toLowerCase() === key));

      return env3.render();

      const data3 = readCustom(chat);
      if (match) data3.items.push({ id: match.id, name: match.name, agentId: match.agentId });

        data3.seq += 1;
        data3.items.push({ id: contactId(`custom:${data3.seq}`), name, note: phone.addNote.trim() });
      }
