  async function handlePhone(action, value, ctx2, env3) {
    if (!action.startsWith("ph-")) return false;
    liveEnv = env3;
    const v = value ?? "";
    if (!["ph-room-del", "ph-view-del", "ph-contact-del", "ph-notice-del"].includes(action)) phone.confirm = null;
    if (action.startsWith("ph-def-")) {
      try {
        const msg = await defaultsAction("ph", action, v, ctx2, env3.refresh);
        if (msg) phone.notice = msg;
      } catch (error) {
        phone.notice = `기본값을 처리하지 못했어요: ${error instanceof Error ? error.message : String(error)}`;
      }
      env3.render();
      return true;
    }
    switch (action) {
      case "ph-sub":
        Object.assign(phone, { sub: v, query: "", ...phone.view === "room" ? {} : { view: "list", invite: null, picked: [] } });
        break;
      case "ph-settings":
        Object.assign(phone, { view: phone.view === "settings" ? "list" : "settings", invite: null, picked: [] });
        break;
      case "ph-att-menu":
        phone.attachMenu = !phone.attachMenu;
        break;
      case "ph-ctab":
        phone.tab = v;
        phone.query = "";
        break;
      case "ph-chat-filter":
        phone.chatFilter = v === "unread" ? "unread" : "all";
        break;
      case "ph-notice-filter":
        phone.noticeFilter = v === "unread" ? "unread" : "all";
        phone.noticePage = 1;
        break;
      case "ph-notice-faction":
        phone.noticeFaction = phone.noticeFaction === v ? "" : v;
        phone.noticePage = 1;
        break;
      case "ph-pick-tab":
        phone.pickTab = v;
        phone.pickQuery = "";
        break;
      case "ph-dm":
        if (phone.view === "room" && phone.room === `dm_${v}` && !isPhone()) return handlePhone("ph-back", void 0, ctx2, env3);
        await openDm(ctx2, env3, v);
        return true;
      case "ph-room":
        if (phone.view === "room" && phone.room === v && !isPhone()) return handlePhone("ph-back", void 0, ctx2, env3);
        await openRoom(ctx2, env3, v);

      case "ph-back":

          break;

          await mutate(ctx2, env3, (st) => {
            const r = st.rooms[rid];

            delete st.rooms[rid];
            st.order = st.order.filter((x) => x !== rid);
          });
          return true;
        }
        Object.assign(phone, { view: "list", invite: null, picked: [] });
        break;
      case "ph-more":
        phone.page += 1;
        break;
      case "ph-att-add":
      case "ph-att-del": {
        const r = readPhone(ctx2.chat).rooms[phone.room ?? ""];
        if (r) attachAction(phone.attach, action === "ph-att-add" ? "add" : "del", v, phKinds(r));
        break;

      case "ph-send": {

  async function addContact(ctx2, env3) {

    const found = [...new Set(d.map.values())].filter((c) => c.name === name || (c.english ?? "").toLowerCase() === key || c.terms.some((t) => t.toLowerCase() === key));

    if (phoneBusy(ctx2)) return;
    await updateChat(ctx2.charIndex, ctx2.chatIndex, (chat) => {

    });

    await env3.refresh();
  }
