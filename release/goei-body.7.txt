  async function handleImgPrompt(action, value, env3) {
    const v = value ?? "";
    imgUi.notice = "";
    switch (action) {
      case "ip-template-save":
        await savePrefs((p) => ({ ...p, imgPrompt: v.trim() && v.trim() !== IMG_PROMPT_DEFAULT ? v.slice(0, IMG_TEMPLATE_MAX) : void 0 }));
        imgUi.template = null;
        imgUi.info = "지침 틀을 저장했어요.";
        env3.render();
        return true;
      case "ip-template-reset":
        await savePrefs((p) => ({ ...p, imgPrompt: void 0 }));
        imgUi.template = null;
        imgUi.info = "기본 틀로 되돌렸어요.";
        env3.render();
        return true;
      case "ip-preview":
        imgUi.preview = !imgUi.preview;
        env3.render();
        return true;
      case "ip-toggle": {
        const [name] = v.split("=");
        const key2 = name.trim();
        if (!key2) return true;
        await savePrefs((p) => {
          const off = new Set((p.imgNamesOff ?? []).map((n2) => n2.toLowerCase()));
          const list4 = off.has(key2.toLowerCase()) ? (p.imgNamesOff ?? []).filter((n2) => n2.toLowerCase() !== key2.toLowerCase()) : [...p.imgNamesOff ?? [], key2];
          return { ...p, imgNamesOff: list4.length ? list4 : void 0 };
        });
        env3.render();
        return true;
      }
      case "ip-extra-add": {
        const name = v.replace(/[<>"{}_\n]/g, "").trim().slice(0, 40);
        if (!name) {
          imgUi.notice = "영문명을 적어 주세요.";
          env3.render();
          return true;
        }
        await savePrefs((p) => ({ ...p, imgNamesExtra: [.../* @__PURE__ */ new Set([...p.imgNamesExtra ?? [], name])] }));
        imgUi.extra = "";
        imgUi.info = `「${name}」을 더했어요.`;
        env3.render();
        return true;
      }
      case "ip-extra-del":
        await savePrefs((p) => {
          const list4 = (p.imgNamesExtra ?? []).filter((n2) => n2 !== v);
          return { ...p, imgNamesExtra: list4.length ? list4 : void 0 };
        });
        env3.render();
        return true;
      case "ip-keywords-save": {
        const list4 = v.split(/[,\n]/).map((x) => x.trim().toLowerCase()).filter(Boolean);
        const same3 = list4.join(",") === IMG_KEYWORDS_DEFAULT.join(",");
        await savePrefs((p) => ({ ...p, imgKeywords: list4.length && !same3 ? list4 : void 0 }));
        imgUi.keywords = null;
        imgUi.info = `표정 ${list4.length ? list4.length : IMG_KEYWORDS_DEFAULT.length}종을 저장했어요.`;
        env3.render();
        return true;
      }
      case "ip-keywords-reset":
        await savePrefs((p) => ({ ...p, imgKeywords: void 0 }));
        imgUi.keywords = null;
        imgUi.info = "기본 표정 세트로 되돌렸어요.";
        env3.render();
        return true;
      default:
        return false;
    }
  }

    const x = (side) => `<span class="gx-mclose ${side}" aria-hidden="true"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><path d="M6 6l12 12M18 6L6 18"/></svg><span>닫기</span></span>`;

  var rubber = (dx, w) => Math.sign(dx) * (Math.abs(dx) / (1 + Math.abs(dx) / (4 * w)));
  function bindMbar(root3, onClose) {
    const bar = root3.querySelector(".gx-mbar");
    const track2 = bar?.querySelector(".gx-mtrack");
    if (!bar || !track2) return;
    const closeW = () => track2.querySelector(".gx-mclose")?.offsetWidth || 72;
    let x0 = 0;
    let y0 = 0;
    let id = null;
    let dragging = false;
    let shown2 = 0;
    let ready = false;
    const paint = () => {
      track2.style.transform = shown2 ? `translateX(${shown2}px)` : "";
      const was = ready;
      ready = Math.abs(shown2) >= closeW();
      track2.querySelectorAll(".gx-mclose").forEach((el) => el.classList.toggle("ready", ready));
      if (ready && !was) navigator.vibrate?.(8);
    };
    const end = () => {
      if (id === null) return;
      id = null;
      if (!dragging) return;
      const close3 = ready;
      track2.classList.add("back");
      shown2 = 0;
      paint();
      setTimeout(() => track2.classList.remove("back"), 260);
      if (close3) onClose();
    };
    bar.addEventListener("pointerdown", (e) => {
      if (e.button !== 0) return;
      id = e.pointerId;
      x0 = e.clientX;
      y0 = e.clientY;
      dragging = false;
      track2.classList.remove("back");
    });
    bar.addEventListener("pointermove", (e) => {
      if (e.pointerId !== id) return;
      const dx = e.clientX - x0;
      if (!dragging) {
        if (Math.abs(dx) < 8 || Math.abs(dx) < Math.abs(e.clientY - y0)) return;
        dragging = true;
        bar.setPointerCapture?.(e.pointerId);
      }
      shown2 = rubber(dx, closeW());
      paint();
    });
    bar.addEventListener("pointerup", end);
    bar.addEventListener("pointercancel", end);
    bar.addEventListener("click", (e) => {
      if (!dragging) return;
      dragging = false;
      e.stopPropagation();
      e.preventDefault();
    }, true);
  }

${v}` : v;

        recUi.importResult = null;

        edUi.info = "가져온 글을 칸에 넣었어요. 저장을 눌러야 카드에 남아요.";
        return true;
      }

        recUi.adding = !recUi.adding;
        return true;
      case "ed-add": {
        const name = (value ?? "").trim();
        if (!name) return true;
        if (worldRecord(kind, name, over) || over[kind]?.[name]) {
