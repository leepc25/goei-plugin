  function recordInput(name, key2, value, ctx2) {
    if (name === "ed-import") {
      recUi.importText = value;
      return true;
    }
    if (name === "ed-anote") {
      const over2 = overridesOf(ctx2);
      const next = { ...over2 };
      if (value === ANOMALY_NOTE) delete next.anomalyNote;
      else next.anomalyNote = value;
      setOverrides(next);
      return true;
    }
    if (name !== "ed-field" && name !== "ed-extra" && name !== "ed-extra-name" && name !== "ed-keys") return false;
    const [kind, recName, id] = key2.split("|");
    if (!(kind in FIELDS) || !recName) return false;
    const over = overridesOf(ctx2);
    const rec2 = structuredClone(worldRecord(kind, recName, over) ?? emptyRecord());
    if (name === "ed-field") {
      if (recUi.variant) {
        rec2.variants = { ...rec2.variants ?? {} };
        const v = { ...rec2.variants[recUi.variant] ?? {} };
        if (value.trim()) v[id] = value;
        else delete v[id];
        if (Object.keys(v).length) rec2.variants[recUi.variant] = v;
        else delete rec2.variants[recUi.variant];
      } else rec2.fields[id] = value;
    } else if (name === "ed-keys") {
      if (value.trim()) rec2.keys = value;
      else delete rec2.keys;
    } else if (name === "ed-extra") {
      const i = Number(id);
      if (!rec2.extra[i]) return false;
      rec2.extra[i] = { ...rec2.extra[i], value };
    } else if (name === "ed-extra-name") {
      const i = Number(id);
      if (!rec2.extra[i]) return false;
      const old = rec2.extra[i].name;
      rec2.extra[i] = { ...rec2.extra[i], name: value };
      if (rec2.order) rec2.order = rec2.order.map((o) => o === `x:${old}` ? `x:${value}` : o);
    } else return false;
    setOverrides(putRecord(over, kind, recName, rec2));
    return true;
  }
  async function recordAction(action, value, ctx2, kind) {
    const over = overridesOf(ctx2);
    switch (action) {
      case "ed-sel":
        edUi.sel[kind] = value ?? "";
        recUi.preview = false;
        recUi.importOpen = false;
        recUi.importResult = null;
        recUi.variant = "";
        return true;
      case "ed-roster-toggle": {
        const cur = current2(ctx2, kind);
        if (!cur || kind !== "agent") return false;
        setOverrides(putRecord(cur.over, kind, cur.name, { ...cur.rec, rosterOff: !cur.rec.rosterOff }));
        return true;
      }
      case "ed-rel-member-toggle": {
        const cur = current2(ctx2, kind);
        if (!cur || kind !== "org") return false;
        setOverrides(putRecord(cur.over, kind, cur.name, { ...cur.rec, relOff: !cur.rec.relOff }));
        return true;
      }
      case "ed-anote-toggle":
        recUi.noteOpen = !recUi.noteOpen;
        return false;
      case "ed-anote-reset": {
        const next = { ...overridesOf(ctx2) };
        delete next.anomalyNote;
        setOverrides(next);
        return true;
      }
      case "ed-preview":
        recUi.preview = !recUi.preview;
        return true;
      case "ed-import-toggle":
        recUi.importOpen = !recUi.importOpen;
        return true;
      case "ed-import-parse":
        recUi.importResult = parseImport(kind, recUi.importText);
        return true;
      case "ed-import-apply": {
        const cur = current2(ctx2, kind);
        const r = recUi.importResult;
        if (!cur || !r) return true;
        for (const [id, v] of Object.entries(r.fields)) cur.rec.fields[id] = cur.rec.fields[id] ? `${cur.rec.fields[id]}

${v}` : v;
        cur.rec.extra = [...cur.rec.extra, ...r.extra];
        setOverrides(putRecord(over, kind, cur.name, cur.rec));
        recUi.importOpen = false;
        recUi.importResult = null;
        recUi.importText = "";
        edUi.info = "가져온 글을 칸에 넣었어요. 저장을 눌러야 카드에 남아요.";
        return true;
      }
