        }

      const [key, op] = v.split(":");
      const s8 = journalSettings(ctx2.chat.scriptstate);
      const dir2 = op === "+" ? 1 : -1;

  function pointsColVar(vars, v) {
    const [id2, what, val] = v.split(":");
    if (id2 !== "points") return null;
    const raw = vars?.[colVar("points")];
    const auto = pointsAuto(vars);
    const hidden = auto ? raw === AUTO_HIDE : raw === "hide";

  }
  function nextColMode(cfg, v) {
    const [id2, what, val] = v.split(":");
