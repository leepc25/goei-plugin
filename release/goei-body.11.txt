      fab = null;
      stopWatcher = startWatcher(null);
      await risuai.registerButton({ name: BRAND.name, icon: GHOST_ICON, iconType: "html", location: "chat", id: `${PLUGIN_ID}-chat` }, () => void openApp({ quietIfNotLite: true })).catch((e) => console.log(`[goei] 채팅 메뉴 단추를 두지 못했어요: ${e instanceof Error ? e.message : String(e)}`));
    }
    return {
      beforeRequest: onBeforeRequest,
      sanitize,
      input: onInput,
      output: onOutput,
      display: onDisplay,
      install: () => openInstall(),
      unload: async () => {
        stopWatcher?.();
        await fab?.remove().catch(() => {
        });
      }
    };
  }
