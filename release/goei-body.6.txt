  async function installAction(action, ctx2, env3) {
    if (!action.startsWith("ed-inst-")) return false;
    if (instUi.busy) return true;
    if (action === "ed-inst-name") {
      instUi.setName = !instUi.setName;
      env3.render();
      return true;
    }
    if (action === "ed-inst-recheck") {
      instUi.forChar = -1;
      instUi.state = null;
      instUi.notice = "";
      env3.render();
      return true;
    }
    const step = action.slice(8);
    if (step !== "all" && step !== "icon-back") return true;
    instUi.busy = step;
    instUi.notice = "";
    instUi.info = "";
    instUi.progress = "";
    env3.render();
    const progress = (t) => {
      instUi.progress = t;
      env3.render();
    };
    const done3 = [];
    const failed = [];
    const run5 = async (label2, f) => {
      try {
        const r = await f();
        if (r) done3.push(r);
      } catch (error) {
        failed.push(`${label2}: ${error instanceof Error ? error.message : String(error)}`);
      }
    };
    const s7 = instUi.state;
    if (step === "all") {
      await run5("로어북·인사말", async () => {
        progress("로어북·인사말 심는 중…");
        await installLoreAndGreetings(ctx2.charIndex, instUi.setName ? { name: BRAND.name } : {});
        return instUi.setName ? "로어북·인사말·봇 이름" : "로어북·인사말";
      });
      notifyInstalled();
      await run5("사진", async () => {
        const n2 = await installPictures(ctx2.charIndex, progress);
        return n2 ? `사진 ${n2}장` : null;
      });
      if (!s7?.icon) await run5("아이콘", async () => (await installIcon(ctx2.charIndex, progress), "아이콘"));
      if (!s7?.wikiPack) await run5("위키 기본 문서", async () => (await installWikiPack(progress), "위키 기본 문서"));
      if (!s7?.tokenizer) await run5("토크나이저", async () => (await installTokenizer(progress), "토크나이저"));
    } else if (step === "icon-back") {
      await run5("이전 아이콘", async () => await restoreIcon(ctx2.charIndex) ? "이전 아이콘" : null);
    }
    if (done3.length) instUi.info = `${done3.join(" · ")} — 끝났어요.`;
    else if (!failed.length) instUi.info = "바꿀 것이 없었어요.";
    if (failed.length) instUi.notice = `안 된 것 — ${failed.join(" / ")}. 다시 누르면 안 된 것만 이어서 해요.`;
    instUi.busy = "";
    instUi.progress = "";
    await env3.refresh();
    await refreshInstall(ctx2);
    env3.render();
    return true;
  }

  function bindMbar(root3, onClose) {
    const bar = root3.querySelector(".gx-mbar");
    const track2 = bar?.querySelector(".gx-mtrack");
    if (!bar || !track2) return;
    const closeW = () => track2.querySelector(".gx-mclose")?.offsetWidth || 72;
    let x0 = 0;
    let y0 = 0;
    let id = null;
    let dragging = false;
    let shown2 = 0;
    let ready = false;
    const paint = () => {
      track2.style.transform = shown2 ? `translateX(${shown2}px)` : "";
      const was = ready;
      ready = Math.abs(shown2) >= closeW();
      track2.querySelectorAll(".gx-mclose").forEach((el) => el.classList.toggle("ready", ready));
      if (ready && !was) navigator.vibrate?.(8);
    };
    const end = () => {
      if (id === null) return;
      id = null;
      if (!dragging) return;
      const close3 = ready;
      track2.classList.add("back");
      shown2 = 0;
      paint();
      setTimeout(() => track2.classList.remove("back"), 260);
      if (close3) onClose();
    };
    bar.addEventListener("pointerdown", (e) => {
      if (e.button !== 0) return;
      id = e.pointerId;
      x0 = e.clientX;
      y0 = e.clientY;
      dragging = false;
      track2.classList.remove("back");
    });
    bar.addEventListener("pointermove", (e) => {
      if (e.pointerId !== id) return;
      const dx = e.clientX - x0;
      if (!dragging) {
        if (Math.abs(dx) < 8 || Math.abs(dx) < Math.abs(e.clientY - y0)) return;
        dragging = true;
        bar.setPointerCapture?.(e.pointerId);
      }
      shown2 = rubber(dx, closeW());
      paint();
    });
    bar.addEventListener("pointerup", end);
    bar.addEventListener("pointercancel", end);
    bar.addEventListener("click", (e) => {
      if (!dragging) return;
      dragging = false;
      e.stopPropagation();
      e.preventDefault();
    }, true);
  }
