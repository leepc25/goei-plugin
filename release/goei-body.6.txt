  async function handleField(action, value, ctx2, env2, opening) {
    const v = value ?? "";
    switch (action) {
      case "map-region":
        field2.region = v === "01" || MAP_CODES.includes(v) ? v : "01";
        field2.place = null;
        field2.confirmEnd = null;
        env2.render();
        return true;
      case "map-jump":
      case "map-place": {
        let v2 = value ?? "";
        if (action === "map-jump") {
          const [code, key2] = v2.split("|");
          if (!MAP_CODES.includes(code) || !key2) return true;
          field2.region = code;
          field2.mapq = "";
          v2 = key2;
        }
        field2.confirmEnd = null;
        if (action === "map-place" && isPC() && field2.place === v2 && sameTarget(field2.target, v2)) {
          field2.place = null;
          field2.target = null;
          env2.render();
          return true;
        }
        field2.place = v2;
        const x = readCtx(ctx2);
        const here = typeof x === "string" || !x.on ? [] : activeEvents(x.world).filter((e) => placeKey(e, x.catalog) === v2);
        if (!here.length) field2.target = placeTarget(v2);
        env2.render();
        return true;
      }
      case "map-unplace":
        field2.place = null;
        field2.target = null;
        field2.confirmEnd = null;
        toStep("goal", env2);
        return true;
      case "map-filter":
        field2.filter = ["all", "an", "in", "ms"].includes(v) ? v : "all";
        env2.render();
        return true;
      case "pick-place":
        if (isPC() && sameTarget(field2.target, v)) {
          field2.target = null;
          env2.render();
          return true;
        }
        field2.target = placeTarget(v);
        toStep("team", env2);
        return true;
      case "field-step":
        toStep(v === "team" ? "team" : "goal", env2);
        return true;
      case "pick-event": {
        const x = readCtx(ctx2);
        if (typeof x === "string") return true;
        const ev = activeEvents(x.world).find((e) => e.key === v);
        if (!ev) return true;
        if (sameTarget(field2.target, ev.key)) {
          field2.target = null;
          field2.confirmEnd = null;
          env2.render();
          return true;
        }
        field2.target = { type: "event", key: ev.key, kind: ev.kind, name: ev.name, loc: ev.loc };
        field2.region = sidoOf(ev.loc);
        field2.confirmEnd = null;
        toStep("team", env2);
        return true;
      }
      case "end-event":
        if (field2.confirmEnd !== v) {
          field2.confirmEnd = v;
          env2.render();
          return true;
        }
        field2.confirmEnd = null;
        if (field2.target?.type === "event" && field2.target.key === v) field2.target = null;
        await saveWorld(ctx2, env2, (w) => endEvent(w, v) ? null : "이미 끝났거나 바뀐 상황이에요.");
        return true;
      case "team-toggle": {
        const tx = readCtx(ctx2);
        if (typeof tx !== "string" && lockedState(tx, v)) return true;
        const i = field2.team.indexOf(v);
        if (i >= 0) field2.team.splice(i, 1);
        else if (field2.team.length + field2.support < TEAM_MAX) field2.team.push(v);
        env2.render();
        return true;
      }
      case "support": {
        const next = field2.support + (Number(v) > 0 ? 1 : -1);
        if (next >= 0 && validTeam(field2.team.length, next)) field2.support = next;
        env2.render();
        return true;
      }
      case "depart":
        await depart(ctx2, env2);
        return true;
      case "mission-kind":
        if (v === "an" || v === "ms" || v === "lm") {
          field2.mkind = v;
          field2.query = "";
        }
        env2.render();

        messages = at < 0 ? [...messages, ...add] : [...messages.slice(0, at), ...add, ...messages.slice(at)];

    let char = await risuai.getCharacterFromIndex(charIndex);

    const cur0 = await risuai.getChatFromIndex(charIndex, chatIndex);
    const mode3 = advValue(cur0?.scriptstate, "gen", "lore_mode");
