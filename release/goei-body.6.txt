      }

        return true;
      case "pick-event": {
        const x = readCtx(ctx2);
        if (typeof x === "string") return true;
        const ev = activeEvents(x.world).find((e) => e.key === v);

        return true;
      }

  function nextWorld(w, v) {
    const [key2, op] = v.split(":");
    const f = WORLD_SPEC.find((x) => x.key === key2);
    if (!f || op === void 0) return null;
    const cur = w[f.key];
    const step = worldStep(f.key);
    const n = op === "+" ? cur + step : op === "-" ? cur - step : Number(op);
    if (!Number.isFinite(n)) return null;
    const clamped = Math.min(f.max ?? 99, Math.max(0, f.fraction ? n : Math.round(n)));
    return clamped === cur ? null : [f.key, clamped];
  }
  async function handleSettings(action, value, ctx2, env2, scope = "chat") {
    const v = value ?? "";
    const lore = ctx2.char.globalLore;
    if (action === "set-bar") {
      settingsUi.confirm = v === "save" || v === "load" ? v : "";
      env2.render();
      return true;
    }
    if (action === "set-bar-no") {
      settingsUi.confirm = "";
      env2.render();
      return true;
    }
    if (action === "set-bar-yes") {
      const what = settingsUi.confirm;
      settingsUi.confirm = "";
      if (what === "save") {
        const cur = current2(ctx2, "chat");
        await saveDefaults(env2, (p) => ({ ...p, basic: { ...cur.basic }, world: { ...cur.world }, theme: cur.theme }));

        env2.render();
      } else if (what === "load") {

        await save(ctx2, env2, (chat) => {
          const vars = { ...basicVars(d.basic), [THEME_KEY]: d.theme, ...loadVarDefaults(["$gen_"], chat.scriptstate) };
          for (const f of WORLD_SPEC) vars[wgKey(f.key)] = String(d.world[f.key]);
          if (worldOn(chat)) {
            const ss = { ...chat.scriptstate ?? {}, ...vars };
            Object.assign(vars, worldVars(readWorld(chat), parseCatalog(lore), worldSettings(lore, ss), String(ss["$st_faction"] ?? "")));
          }
          return vars;
        });
        if (!settingsUi.notice) settingsUi.info = "새 채팅 기본값을 지금 채팅에 불러왔어요.";
        env2.render();
      } else env2.render();
      return true;
    }
    if (action === "set-status" && ["auto", "dark", "light", "paper", "off"].includes(v ?? "")) {
      await saveDefaults(env2, (p) => ({ ...p, status: v === "auto" ? void 0 : v }));
      repaintTheme();
      return true;
    }
    if (action === "set-img-size" || action === "set-img-fold") {
      const size = imgSize(v);
      const fold = imgFold(v);
      if (action === "set-img-size" && size !== null) await saveDefaults(env2, (p) => ({ ...p, imgSize: size === IMG_SIZE.def ? void 0 : size }));
      else if (action === "set-img-fold" && fold !== null) await saveDefaults(env2, (p) => ({ ...p, imgFold: fold === IMG_FOLD.def ? void 0 : fold }));
      repaintTheme();
      return true;
    }
    if (/^set-img-(show|send)(-mode)?$/.test(action)) {
      const key2 = action.includes("show") ? "imgShow" : "imgSend";
      const cur = getPrefs()[key2];
      const n = action.endsWith("-mode") ? v === "all" ? 0 : cur === 0 || cur === void 0 ? IMG_RECENT.def : cur : imgRecent(v);
      if (n !== null) await saveDefaults(env2, (p) => ({ ...p, [key2]: n === IMG_RECENT.def ? void 0 : n }));
      return true;
    }
    if (action === "set-agents") {
      await saveDefaults(env2, (p) => ({ ...p, agents: v === "list" ? "list" : void 0 }));

    }
    if (action === "set-nav" || action === "set-dim") {

      switch (action) {

        case "set-agent": {

        }

          return true;
        }
      }
      return false;
    }
    switch (action) {
      case "set-world":
        await save(ctx2, env2, (chat) => {
          const hit3 = nextWorld(worldSettings(lore, chat.scriptstate), v);

        });
        return true;
      case "set-anomaly":
        if (v === "off") {
          if (!worldOn(ctx2.chat)) return true;
          settingsUi.confirmOff = true;
          env2.render();
          return true;
        }
        if (v === "off-no") {
          settingsUi.confirmOff = false;
          env2.render();
          return true;
        }
        settingsUi.confirmOff = false;
        if (v === "off-yes") {
          await save(ctx2, env2, (chat) => worldOn(chat) ? worldOffVars() : null);
          return true;
        }
        if (v === "on") {
          await save(ctx2, env2, (chat) => {
            if (worldOn(chat)) return null;
            const faction = String(chat.scriptstate?.["$st_faction"] ?? "");
            return startWorldVars(parseCatalog(lore), worldSettings(lore, chat.scriptstate), faction, Math.random, String(chat.scriptstate?.["$st_player"] ?? ""));
          });

          if (worldOn(chat)) {
            const ss = { ...chat.scriptstate ?? {}, ...vars };
            Object.assign(vars, worldVars(readWorld(chat), parseCatalog(lore), worldSettings(lore, ss), String(ss["$st_faction"] ?? "")));
          }
          return vars;
        });
        return true;
      case "set-theme":
        if (v !== "light" && v !== "dark" && v !== "paper") return true;
        await save(ctx2, env2, (chat) => chat.scriptstate?.[THEME_KEY] === v ? null : { [THEME_KEY]: v });
        return true;
      case "set-support":
        await save(ctx2, env2, (chat) => withBasic(ctx2, chat, { supportOn: v === "1" }));
        return true;
      case "set-asset":
        await save(ctx2, env2, (chat) => withBasic(ctx2, chat, { assetOn: v === "1" }));

        return true;

      case "set-asset-limit": {

        await save(ctx2, env2, (chat) => withBasic(ctx2, chat, { assetLimit: n }));
        return true;
      }
    }
    return false;
  }
