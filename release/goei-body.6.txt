  var STATUS_HEAD = "[Previous status]";
  var NOTE = "This is the status line of the scene right before the next input; older status lines were removed from the history. Continue from it: advance the time, update the other fields, and output exactly one new [▣] line at the very end of your response.";
  function lastStatusLine(messages) {
    for (let i = messages.length - 1; i >= 0; i--) {
      const m = messages[i];
      if (m?.role !== "assistant" || typeof m.content !== "string") continue;
      const tags = findStatusTags(m.content);
      if (tags.length) return tags[tags.length - 1].raw.trim();
    }
    return null;
  }
  function stripStatusLines(messages) {
    return messages.map((m) => {
      if (m?.role !== "assistant" || typeof m.content !== "string") return m;
      const tags = findStatusTags(m.content);
      if (!tags.length) return m;
      let c = m.content;
      for (const t of tags) c = c.replace(t.raw, "");
      return { ...m, content: c.replace(/[ \t]+\n/g, "\n").replace(/\n{3,}/g, "\n\n").trim() };
    });
  }
  function statusBlock(line2) {
    return `${STATUS_HEAD}
${line2}
${NOTE}`;
  }
  function withStatusBlock(messages) {
    if (messages.some((m) => typeof m?.content === "string" && m.content.startsWith(STATUS_HEAD))) return messages;
    const line2 = lastStatusLine(messages);
    if (!line2) return messages;
    const out = stripStatusLines(messages);
    const at = out.map((m) => m?.role).lastIndexOf("user");
    const block2 = { role: "system", content: statusBlock(line2) };
    return at < 0 ? [...out, block2] : [...out.slice(0, at), block2, ...out.slice(at)];
  }
