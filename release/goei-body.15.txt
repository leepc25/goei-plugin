  async function handleWiki(action, value, ctx2, env3) {
    const v = value ?? "";
    if (action !== "wiki-delete") wiki.confirmDelete = null;
    if (action.startsWith("wiki-def-")) {
      try {
        const msg = await defaultsAction("wiki", action, v, ctx2, env3.refresh);
        if (msg) wiki.info = msg;
      } catch (error) {
        wiki.notice = `기본값을 처리하지 못했어요: ${error instanceof Error ? error.message : String(error)}`;
      }
      env3.render();
      return true;
    }
    switch (action) {
      case "wiki-home":
        go("home");
        wiki.history = [];
        wiki.listOpen = false;
        break;
      case "wiki-back": {
        const prev = wiki.history.pop();
        if (prev && prev.view !== "home") Object.assign(wiki, { view: prev.view, title: prev.title }, { confirmDelete: null, notice: "" });
        else Object.assign(wiki, { view: "home", title: "", notice: "" }, { history: [] });
        break;
      }
      case "wiki-list":
        wiki.listOpen = v === "open" ? true : v === "close" ? false : !wiki.listOpen;
        env3.render();
        if (wiki.listOpen && wide() && !v) document.querySelector(".nw .ph-search")?.focus();
        return true;
      case "wiki-find":
        wiki.listOpen = true;
        env3.render();
        document.querySelector(".nw .ph-search")?.focus();
        return true;
      case "wiki-tab":
        wiki.tab = TABS.some(([t]) => t === v) ? v : "all";
        break;
      case "wiki-browse":
        Object.assign(wiki, { tab: TABS.some(([t]) => t === v) ? v : "all", listOpen: true, query: "" });
        break;
      case "wiki-search":
        wiki.homeQuery = "";
        search(ctx2, v);
        if (wiki.view === "compose" || wiki.view === "doc") wiki.listOpen = false;
        break;
      case "wiki-pick":
        if ((wiki.view === "doc" || wiki.view === "compose") && canon(ctx2)(v) === wiki.title && wide()) {
          Object.assign(wiki, { view: "home", title: "", history: [] });
          break;
        }
        wiki.history = [];
        wiki.homeQuery = "";
        openTitle(ctx2, v, "search");
        if (wiki.view !== "home") wiki.listOpen = false;
        break;
      case "wiki-open":
        openTitle(ctx2, v, wiki.view === "doc" ? "link" : "search");
        break;
      case "wiki-write":
        if (!v.trim()) break;
        wiki.newTitle = "";
        openTitle(ctx2, v, "direct");
        if (wiki.view === "compose" || wiki.view === "doc") wiki.listOpen = false;
        break;
      case "wiki-new":
        if (wiki.view === "new") go("home");
        else go("new");
        wiki.listOpen = false;
        break;
      case "wiki-start":
        if (wiki.view === "compose" && wiki.title === v) void generate3(ctx2, env3, v, false, wiki.docReq);
        break;
      case "wiki-rewrite": {
        if (!v) break;
        if (wiki.view !== "doc" || wiki.title !== v) go("doc", v);
        if (wiki.rewrite === v) {
          wiki.rewrite = null;
          break;
        }
        const old = findDoc(ctx2.chat, v, ctx2.char.globalLore);
        Object.assign(wiki, { rewrite: v, docReq: old?.meta?.docRequest ?? "", rewritePrev: wikiSettings(ctx2.chat.scriptstate).refreshPrevious });
        break;
      }
      case "wiki-rw-prev":
        wiki.rewritePrev = v === "1";
        break;
      case "wiki-rw-cancel":
        Object.assign(wiki, { rewrite: null, docReq: "" });
        break;
      case "wiki-rw-go":
        if (!v || wiki.rewrite !== v) break;
        void generate3(ctx2, env3, v, true, wiki.docReq, wiki.rewritePrev);
        break;
      case "wiki-delete":
        if (wiki.confirmDelete !== v) {
          wiki.confirmDelete = v;
          break;
        }
        wiki.confirmDelete = null;
        if (wiki.title === v && wiki.view === "doc" && !baseDocs(ctx2.char.globalLore).some((d) => d.title === v)) Object.assign(wiki, { view: "home", title: "", history: [] });
        await updateChat(ctx2.charIndex, ctx2.chatIndex, (chat) => deleteDoc(chat, v));
        await env3.refresh();
        return true;
      case "wiki-jump": {
        const el = document.getElementById(v);
        const fold3 = el?.closest("details");
        if (fold3) fold3.open = true;
        el?.scrollIntoView({ behavior: "smooth", block: "start" });
        return true;
      }
      case "wiki-request":
        await setVar2(ctx2, env3, "$nw_writing_request", v.trim());
        return true;
      case "wiki-length":
      case "wiki-mode":
      case "wiki-model":
      case "wiki-refresh_previous":
      case "wiki-context":
        await setVar2(ctx2, env3, `$nw_${action.slice(5)}`, v);
        return true;
      case "wiki-settings":
        if (wiki.view === "settings") go("home");
        else go("settings");
        wiki.listOpen = false;
        break;
      case "wiki-people":
        resetPeopleUi();
        go("people");
        wiki.listOpen = false;
        break;
      case "wiki-person":
        if (!v) break;
        resetPeopleUi();
        go("person", v);
        wiki.listOpen = false;
        break;
      case "wiki-docs":
        resetPeopleUi();
        go("docs");
        wiki.listOpen = false;
        break;
      case "wiki-faction":
        if (!v) break;
        resetPeopleUi();
        go("faction", v);
        wiki.listOpen = false;
        break;
      default:
        return false;
    }
    env3.render();
    return true;
  }

  function nextColMode(cfg, v) {
    const [id, what, val] = v.split(":");
    const def = STATUS_COLS.find((c) => c.id === id);
    if (!def || what !== "use" && what !== "show" || val !== "1" && val !== "0") return null;
    const cur = cfg[def.id];
    let next = cur;
    if (what === "use") next = val === "1" ? cur === "off" ? "on" : cur : def.locked ? cur : "off";
    else next = cur === "off" ? cur : val === "1" ? "on" : "hide";
    return next === cur ? null : [def.id, next];
  }
  function nextWorld(w, v) {
    const [key2, op] = v.split(":");
    const f = WORLD_SPEC.find((x) => x.key === key2);
    if (!f || op === void 0) return null;
    const cur = w[f.key];
    const step = worldStep(f.key);
    const n2 = op === "+" ? cur + step : op === "-" ? cur - step : Number(op);
    if (!Number.isFinite(n2)) return null;
    const clamped = Math.min(f.max ?? 99, Math.max(0, f.fraction ? n2 : Math.round(n2)));
    return clamped === cur ? null : [f.key, clamped];
  }

  var N = 256;
  function lattice(seed, g) {
    const a = new Float32Array(g * g);
    let s5 = seed >>> 0;
    for (let i = 0; i < a.length; i++) {
      s5 = s5 * 1664525 + 1013904223 >>> 0;
      a[i] = s5 / 4294967296;
    }
    return a;
  }
  function noise(lat, g, x, y) {
    const xi = Math.floor(x), yi = Math.floor(y);
    const fx = x - xi, fy = y - yi;
    const sx = fx * fx * (3 - 2 * fx), sy = fy * fy * (3 - 2 * fy);
    const x0 = (xi % g + g) % g, y0 = (yi % g + g) % g, x1 = (x0 + 1) % g, y1 = (y0 + 1) % g;
    const a = lat[y0 * g + x0] + (lat[y0 * g + x1] - lat[y0 * g + x0]) * sx;
    const b = lat[y1 * g + x0] + (lat[y1 * g + x1] - lat[y1 * g + x0]) * sx;
    return a + (b - a) * sy;
  }
  function fbm(seed, base, oct) {
    const lats = Array.from({ length: oct }, (_, o) => lattice(seed + o * 101, base << o));
    return (u, v) => {
      let t = 0, amp = 0.5, sum = 0;
      for (let o = 0; o < oct; o++) {
        const g = base << o;
        t += noise(lats[o], g, u * g, v * g) * amp;
        sum += amp;
        amp *= 0.5;
      }
      return t / sum;
    };
  }
  function draw(seed) {
    const c = document.createElement("canvas");
    c.width = c.height = N;
    const x = c.getContext("2d");
    if (!x) return null;
    const main = fbm(seed, 4, 6);
    const warp = fbm(seed + 7, 3, 3);
    const img = x.createImageData(N, N);
    const sh = x.createImageData(N, N);
    for (let j = 0; j < N; j++) {
      for (let i = 0; i < N; i++) {
        const u = i / N, v = j / N;
        const w = warp(u, v) - 0.5;
        const n2 = main(u + w * 0.35, v + w * 0.2);
        const d = Math.min(1, Math.max(0, (n2 - 0.5) / 0.22));
        const a = d * d * (3 - 2 * d);
        const k = (j * N + i) * 4;
        const tone = 196 + 59 * Math.min(1, a * 1.3);
        img.data[k] = tone;
        img.data[k + 1] = tone + 2;
