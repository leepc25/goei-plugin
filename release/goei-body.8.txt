        } catch {
        }
      }
      return true;
    }
    if (action === "jr-step") {
      const [key2, op] = v.split(":");
      const s5 = journalSettings(ctx2.chat.scriptstate);
      const dir2 = op === "+" ? 1 : -1;

  function nextColMode(cfg, v) {
    const [id, what, val] = v.split(":");
    const def = STATUS_COLS.find((c) => c.id === id);
    if (!def || what !== "use" && what !== "show" || val !== "1" && val !== "0") return null;
    const cur = cfg[def.id];
    let next = cur;
    if (what === "use") next = val === "1" ? cur === "off" ? "on" : cur : def.locked ? cur : "off";
    else next = cur === "off" ? cur : val === "1" ? "on" : "hide";
    return next === cur ? null : [def.id, next];
  }
  function nextWorld(w, v) {
    const [key2, op] = v.split(":");
    const f = WORLD_SPEC.find((x) => x.key === key2);
    if (!f || op === void 0) return null;
    const cur = w[f.key];
    const step = worldStep(f.key);
    const n2 = op === "+" ? cur + step : op === "-" ? cur - step : Number(op);
    if (!Number.isFinite(n2)) return null;
    const clamped = Math.min(f.max ?? 99, Math.max(0, f.fraction ? n2 : Math.round(n2)));
    return clamped === cur ? null : [f.key, clamped];
  }

  var N = 256;
  function lattice(seed, g2) {
    const a = new Float32Array(g2 * g2);
    let s5 = seed >>> 0;
    for (let i = 0; i < a.length; i++) {
      s5 = s5 * 1664525 + 1013904223 >>> 0;
      a[i] = s5 / 4294967296;
    }
    return a;
  }
  function noise(lat, g2, x, y) {
    const xi = Math.floor(x), yi = Math.floor(y);
    const fx = x - xi, fy = y - yi;
    const sx = fx * fx * (3 - 2 * fx), sy = fy * fy * (3 - 2 * fy);
    const x0 = (xi % g2 + g2) % g2, y0 = (yi % g2 + g2) % g2, x1 = (x0 + 1) % g2, y1 = (y0 + 1) % g2;
    const a = lat[y0 * g2 + x0] + (lat[y0 * g2 + x1] - lat[y0 * g2 + x0]) * sx;
    const b = lat[y1 * g2 + x0] + (lat[y1 * g2 + x1] - lat[y1 * g2 + x0]) * sx;
    return a + (b - a) * sy;
  }
  function fbm(seed, base, oct) {
    const lats = Array.from({ length: oct }, (_, o) => lattice(seed + o * 101, base << o));
    return (u, v) => {
      let t = 0, amp = 0.5, sum = 0;
      for (let o = 0; o < oct; o++) {
        const g2 = base << o;
        t += noise(lats[o], g2, u * g2, v * g2) * amp;
        sum += amp;
        amp *= 0.5;
      }
      return t / sum;
    };
  }
