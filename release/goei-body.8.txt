  async function fetchParts(names4, get, onDone, limit = PART_CONCURRENCY) {
    const out = new Array(names4.length);
    let next = 0;
    let done3 = 0;
    let failed4 = false;
    const worker = async () => {
      while (next < names4.length && !failed4) {
        const i = next++;
        try {
          out[i] = await get(names4[i]);
        } catch (error) {
          failed4 = true;
          throw error;
        }
        onDone?.(++done3);
      }
    };
    await Promise.all(Array.from({ length: Math.min(limit, names4.length) }, worker));
    return out.join("");
  }
