  function findPublicSource(st, nick) {
    let found;

      }

  function parseJsonPayload(raw) {
    let text5 = String(raw ?? "").replace(/^﻿/, "").trim();
    for (let i = 0; i < 12; i++) {
      const before = text5;
      if (text5.startsWith("{")) break;
      const tag = /^<(think|thoughts|analysis|lb-process)(?:\s[^>]*)?>/i.exec(text5);
      if (tag) {
        const end = text5.toLowerCase().indexOf(`</${tag[1].toLowerCase()}>`);
        if (end >= 0) text5 = text5.slice(end + tag[1].length + 3).trim();
        else if (/\n\s*\{/.test(text5)) text5 = text5.slice(text5.search(/\n\s*\{/)).trim();
      } else {
        const fence = /^```[\w-]*\s*\n([\s\S]*?)\n?```\s*$/.exec(text5);
        if (fence) text5 = fence[1].trim();
        else {
          const wrap = /^<(output|json|lb-gn)(\s[^>]*)?>([\s\S]*)<\/\1>\s*$/i.exec(text5);
          if (wrap) text5 = wrap[3].trim();
        }
      }
      if (text5 === before) break;
    }
    try {
      return JSON.parse(text5);
    } catch {
      const dup = /^([\s\S]*\})\s*\{\s*"comments"\s*:\s*\[\s*\]\s*\}\s*$/.exec(text5);
      if (dup) {
        try {
          return JSON.parse(dup[1]);
        } catch {
        }
      }
      const a = text5.indexOf("{");
      const b = text5.lastIndexOf("}");
      if (a >= 0 && b > a) {
        const body5 = text5.slice(a, b + 1);
        try {
          return JSON.parse(body5);
        } catch {
          const fixed = body5.replace(/,(\s*[}\]])/g, "$1").replace(/[\u0000-\u001f]/g, (c) => c === "\n" || c === "\r" || c === "	" ? c : "");
          try {
            return JSON.parse(fixed.replace(/"((?:[^"\\]|\\.)*)"/g, (m) => m.replace(/\n/g, "\\n").replace(/\r/g, "\\r").replace(/\t/g, "\\t")));
          } catch {
          }
        }
      }
      throw new GnetError("AI 응답을 JSON으로 읽지 못했어요.", raw);
    }
  }
