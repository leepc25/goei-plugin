  var ALIAS_TEXT_MAX = 2e3;
  var ALIAS_WORD_MAX = 30;
  var SEP3 = /→|=>|->|=|:/;
  function aliasRows(text5) {
    const out = [];
    for (const line2 of String(text5 ?? "").split(/\r?\n/)) {
      const m = SEP3.exec(line2);
      if (!m) continue;
      out.push({ a: line2.slice(0, m.index).trim(), b: line2.slice(m.index + m[0].length).trim() });
    }
    return out;
  }
  var aliasText2 = (rows3) => rows3.map((r) => `${r.a} → ${r.b}`).join("\n");
  var aliasShown = (stored2) => {
    const s8 = String(stored2 ?? "");
    return aliasRows(s8.trim() ? s8 : ICON_ALIAS_DEFAULT);
  };
  var aliasChanged = (stored2) => {
    const s8 = String(stored2 ?? "").trim();
    return s8 !== "" && aliasText2(aliasRows(s8)) !== aliasText2(aliasRows(ICON_ALIAS_DEFAULT));
  };
  var aliasRowBad = (r) => !r.a ? "끝 말이 비어 있어 이 줄은 안 쓰여요." : !r.b ? "바꿀 말이 비어 있어 이 줄은 안 쓰여요." : r.a === r.b ? "두 말이 같아 이 줄은 안 쓰여요." : "";
  var clean7 = (v) => v.replace(/→|=>|->|[=:\n]/g, " ").replace(/\s+/g, " ").trim().slice(0, ALIAS_WORD_MAX);
  var store = (rows3) => {
    const t = aliasText2(rows3);
    return !rows3.length || t === aliasText2(aliasRows(ICON_ALIAS_DEFAULT)) ? "" : t;
  };
  var aliasCanAdd = (stored2) => aliasText2(aliasShown(stored2)).length + 4 <= ALIAS_TEXT_MAX;
  function aliasEdit(stored2, op, value = "") {
    const cur = String(stored2 ?? "").trim();
    const rows3 = aliasShown(cur);
    let next;
    if (op === "reset") next = "";
    else if (op === "add") {
      if (!aliasCanAdd(cur)) return null;
      next = store([...rows3, { a: "", b: "" }]);
    } else if (op === "del") {
      const i = Number(value);
      if (!/^\d+$/.test(value) || i >= rows3.length) return null;
      next = store(rows3.filter((_, k) => k !== i));
    } else {
      const m = /^(\d+):(a|b)=([\s\S]*)$/.exec(value);
      if (!m || Number(m[1]) >= rows3.length) return null;
      next = store(rows3.map((r, k) => k === Number(m[1]) ? { ...r, [m[2]]: clean7(m[3]) } : r));
      if (next.length > ALIAS_TEXT_MAX) return null;
    }
    return next === cur ? null : next;
  }
