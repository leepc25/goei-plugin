  async function handleField(action, value, ctx2, env3, opening) {
    const v = value ?? "";
    switch (action) {
      case "map-region":
        field3.region = v === "01" || MAP_CODES.includes(v) ? v : "01";
        field3.place = null;
        field3.confirmEnd = null;
        env3.render();
        return true;
      case "map-jump":
      case "map-place": {
        let v2 = value ?? "";
        if (action === "map-jump") {
          const [code2, key2] = v2.split("|");
          if (!MAP_CODES.includes(code2) || !key2) return true;
          field3.region = code2;
          field3.mapq = "";
          v2 = key2;
        }
        field3.confirmEnd = null;
        if (action === "map-place" && isPC() && field3.place === v2 && sameTarget(field3.target, v2)) {
          field3.place = null;
          field3.target = null;
          env3.render();
          return true;
        }
        field3.place = v2;
        const x = readCtx(ctx2);
        const here = typeof x === "string" || !x.on ? [] : activeEvents(x.world).filter((e) => placeKey(e, x.catalog) === v2);
        if (!here.length) field3.target = placeTarget(v2);
        env3.render();
        return true;
      }
      case "map-unplace":
        field3.place = null;
        field3.target = null;
        field3.confirmEnd = null;
        toStep("goal", env3);
        return true;
      case "map-filter":
        field3.filter = ["all", "an", "in", "ms"].includes(v) ? v : "all";
        env3.render();
        return true;
      case "pick-place":
        if (isPC() && sameTarget(field3.target, v)) {
          field3.target = null;
          env3.render();
          return true;
        }
        field3.target = placeTarget(v);
        toStep("team", env3);
        return true;
      case "field-step":
        toStep(v === "team" ? "team" : "goal", env3);
        return true;
      case "pick-event": {
        const x = readCtx(ctx2);
        if (typeof x === "string") return true;
        const ev = activeEvents(x.world).find((e) => e.key === v);
        if (!ev) return true;
        if (sameTarget(field3.target, ev.key)) {
          field3.target = null;
          field3.confirmEnd = null;
          env3.render();
          return true;
        }
        field3.target = { type: "event", key: ev.key, kind: ev.kind, name: ev.name, loc: ev.loc };
        field3.region = sidoOf(ev.loc);
        field3.confirmEnd = null;
        toStep("team", env3);
        return true;
      }
      case "end-event":
        if (field3.confirmEnd !== v) {
          field3.confirmEnd = v;
          env3.render();
          return true;
        }
        field3.confirmEnd = null;
        if (field3.target?.type === "event" && field3.target.key === v) field3.target = null;
        await saveWorld(ctx2, env3, (w) => endEvent(w, v) ? null : "이미 끝났거나 바뀐 상황이에요.");
        return true;
      case "team-toggle": {
        const tx = readCtx(ctx2);
        if (typeof tx !== "string" && lockedState(tx, v)) return true;
        const i = field3.team.indexOf(v);
        if (i >= 0) field3.team.splice(i, 1);
        else if (field3.team.length + field3.support < TEAM_MAX) field3.team.push(v);
        env3.render();
        return true;
      }
      case "support": {
        const next = field3.support + (Number(v) > 0 ? 1 : -1);
        if (next >= 0 && validTeam(field3.team.length, next)) field3.support = next;
        env3.render();
        return true;
      }
      case "depart":
        await depart(ctx2, env3);
        return true;
      case "mission-kind":
        if (v === "an" || v === "ms" || v === "lm") {
          field3.mkind = v;
          field3.query = "";
        }
        env3.render();
        return true;
      case "mission-open":
        field3.mopen = field3.mopen === v ? "" : v;
        env3.render();
        return true;
      case "mission-pick": {
        const m = v.match(/^(an|ms|lm):([\s\S]*)$/);
        const kind = m?.[1] ?? field3.mkind;
        const name = m ? m[2] : v;
        if (isPC() && field3.mname === name && field3.mpick === kind && field3.mopen !== `${kind}:${name}`) {
          field3.mname = null;
          env3.render();
          return true;
        }
        await aimFlash(v);
        field3.mname = name;
        field3.mpick = kind;
        toStep("team", env3);
        return true;
      }
      case "mission-cancel":
        if (v === "an" || v === "ms" || v === "in") await saveWorld(ctx2, env3, (w) => (cancelMission(w, v), null));
        return true;
      case "mission-unpick":
        await aimFlash(v);
        field3.mname = null;
        toStep("goal", env3);
        return true;
      case "mission-send":
        await sendMission(ctx2, env3, opening);
        return true;
    }
    return false;
  }
