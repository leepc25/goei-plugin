  async function handleField(action, value, ctx2, env3, opening) {
    const v = value ?? "";
    switch (action) {
      case "map-region":
        field3.region = v === "01" || MAP_CODES.includes(v) ? v : "01";
        field3.place = null;
        field3.confirmEnd = null;
        env3.render();
        return true;
      case "map-jump":
      case "map-place": {
        let v2 = value ?? "";
        if (action === "map-jump") {
          const [code2, key2] = v2.split("|");
          if (!MAP_CODES.includes(code2) || !key2) return true;
          field3.region = code2;
          field3.mapq = "";
          v2 = key2;
        }
        field3.confirmEnd = null;
        if (action === "map-place" && isPC() && field3.place === v2 && sameTarget(field3.target, v2)) {
          field3.place = null;
          field3.target = null;
          env3.render();
          return true;
        }
        field3.place = v2;
        const x = readCtx(ctx2);
        const here = typeof x === "string" || !x.on ? [] : activeEvents(x.world).filter((e) => placeKey(e, x.catalog) === v2);
        if (!here.length) field3.target = placeTarget(v2);
        env3.render();
        return true;
      }
      case "map-unplace":
        field3.place = null;
        field3.target = null;
        field3.confirmEnd = null;
        toStep("goal", env3);
        return true;
      case "map-filter":
        field3.filter = ["all", "an", "in", "ms"].includes(v) ? v : "all";
        env3.render();
        return true;
      case "pick-place":
        if (isPC() && sameTarget(field3.target, v)) {
          field3.target = null;
          env3.render();
          return true;
        }
        field3.target = placeTarget(v);
        toStep("team", env3);
