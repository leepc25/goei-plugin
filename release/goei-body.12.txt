  async function handleImgPrompt(action, value, env3) {
    const v = value ?? "";
    imgUi.notice = "";
    switch (action) {
      case "ip-template-save":
        await savePrefs((p) => ({ ...p, imgPrompt: v.trim() && v.trim() !== IMG_PROMPT_DEFAULT ? v.slice(0, IMG_TEMPLATE_MAX) : void 0 }));
        imgUi.template = null;
        imgUi.info = "지침 틀을 저장했어요.";
        env3.render();
        return true;
      case "ip-template-reset":
        await savePrefs((p) => ({ ...p, imgPrompt: void 0 }));
        imgUi.template = null;
        imgUi.info = "기본 틀로 되돌렸어요.";
        env3.render();
        return true;
      case "ip-preview":
        imgUi.preview = !imgUi.preview;
        env3.render();
        return true;
      case "ip-toggle": {
        const [name] = v.split("=");
        const key2 = name.trim();
        if (!key2) return true;
        await savePrefs((p) => {
          const off = new Set((p.imgNamesOff ?? []).map((n2) => n2.toLowerCase()));
          const list4 = off.has(key2.toLowerCase()) ? (p.imgNamesOff ?? []).filter((n2) => n2.toLowerCase() !== key2.toLowerCase()) : [...p.imgNamesOff ?? [], key2];
          return { ...p, imgNamesOff: list4.length ? list4 : void 0 };
        });
        env3.render();
        return true;
      }
      case "ip-extra-add": {
        const name = v.replace(/[<>"{}_\n]/g, "").trim().slice(0, 40);
        if (!name) {
          imgUi.notice = "영문명을 적어 주세요.";
          env3.render();
          return true;
        }
        await savePrefs((p) => ({ ...p, imgNamesExtra: [.../* @__PURE__ */ new Set([...p.imgNamesExtra ?? [], name])] }));
        imgUi.extra = "";
        imgUi.info = `「${name}」을 더했어요.`;
        env3.render();
        return true;
      }
      case "ip-extra-del":
        await savePrefs((p) => {
          const list4 = (p.imgNamesExtra ?? []).filter((n2) => n2 !== v);
          return { ...p, imgNamesExtra: list4.length ? list4 : void 0 };
        });
        env3.render();
        return true;
      case "ip-keywords-save": {
        const list4 = v.split(/[,\n]/).map((x) => x.trim().toLowerCase()).filter(Boolean);
        const same2 = list4.join(",") === IMG_KEYWORDS_DEFAULT.join(",");
        await savePrefs((p) => ({ ...p, imgKeywords: list4.length && !same2 ? list4 : void 0 }));
        imgUi.keywords = null;
        imgUi.info = `표정 ${list4.length ? list4.length : IMG_KEYWORDS_DEFAULT.length}종을 저장했어요.`;
        env3.render();
        return true;
      }
      case "ip-keywords-reset":
        await savePrefs((p) => ({ ...p, imgKeywords: void 0 }));
        imgUi.keywords = null;
        imgUi.info = "기본 표정 세트로 되돌렸어요.";
        env3.render();
        return true;
      default:
        return false;
    }
  }
