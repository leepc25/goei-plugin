      return;
    }
    u.cd -= dt;

    if (u.act === "hurt") {

      return;
    }
    if (u.act === "attack") {
      const target2 = nearest(u, foes);
      if (!u.hit && u.t >= ATTACK_TIME * AF.attackHit / AF.attack.length) {
        u.hit = true;
        if (target2 && Math.abs(target2.x - u.x) <= d.range + 12) damage(run3, u, target2, d.atk);
      }
      if (u.t >= ATTACK_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "skill") {
      if (!u.hit && u.t >= SKILL_TIME * AF.skillHit / AF.skill.length) {

        applySkill(run3, u, foes, team);
      }
      if (u.t >= SKILL_TIME) u.act = "idle", u.t = 0;
      return;
    }
    const target = nearest(u, foes);
    if (u.gauge >= 1) {

    }

    }

      u.x += step2;

      if (u.act !== "walk") u.act = "walk", u.t = 0;

    }

  function stepEnemy(run3, u, dt, team) {
    const d = ENEMIES[u.kind];
    u.t += dt;
    u.flash = Math.max(0, u.flash - dt);
    if (u.act === "dead") {
      u.gone += dt;
      return;
    }
    if (u.bound > 0) {
      u.bound = Math.max(0, u.bound - dt);
      if (u.act === "walk" || u.act === "attack") u.act = "idle", u.t = 0;
      return;
    }
    u.cd -= dt;
    if (u.act === "hurt") {
      if (u.t > HURT_TIME) u.act = "idle", u.t = 0;
      return;
    }
    const guardian = run3.guard > 0 ? team.find((a) => a.kind === "gorgo" && alive(a)) : void 0;
    const target = guardian ?? nearest(u, team);
    if (u.act === "attack") {
      const at = enemyAttackTime(u);
      if (!u.hit && u.t >= at * EF.attackHit / EF.attack.length) {
        u.hit = true;
        if (target && Math.abs(target.x - u.x) <= d.range + 14) damage(run3, u, target, scaled(run3, d.atk));
      }
      if (u.t >= at) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "skill") {
      if (!u.hit && u.t >= SKILL_TIME * EF.skillHit / EF.skill.length) {
        u.hit = true;
        run3.fx.push({ kind: "boom", x: u.x - 70, y: footY(u) - 30, t: 0, life: 0.5 });
        for (const a of team) if (alive(a) && u.x - a.x < 170) damage(run3, u, a, scaled(run3, d.atk) * 1.3, true);
      }
      if (u.t >= SKILL_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (!target) {
      if (u.act !== "idle") u.act = "idle", u.t = 0;
      return;
    }
    if (d.boss) {
      u.gauge -= dt;
      if (u.gauge <= 0 && u.x - target.x < 190) {
        u.gauge = 7 + rand(run3) * 3;
        u.act = "skill";
        u.t = 0;
        u.hit = false;
        return;
      }
    }
    const gap = u.x - target.x;
    if (gap > d.range) {
      const step2 = Math.min(gap - d.range, d.speed * dt);
      u.x -= step2;
      u.stride += step2;
      if (u.act !== "walk") u.act = "walk", u.t = 0;
      return;
    }
    if (u.cd <= 0) {
      u.act = "attack";
      u.t = 0;
      u.hit = false;
      u.cd = d.rate;
      return;
    }
    if (u.act !== "idle") u.act = "idle", u.t = 0;
  }
  function step(run3, meta, dt, viewW) {
    const res = { changed: false };
    run3.pt += dt;
    run3.guard = Math.max(0, run3.guard - dt);
    run3.rally = Math.max(0, run3.rally - dt);
    for (const f of run3.fx) f.t += dt;

    const team = agents(run3);
    const foes = enemies(run3);
    if (run3.phase === "camp") {
      for (const a of team) {
        a.t += dt;
        a.hp = Math.min(a.max, a.hp + a.max * 0.25 * dt);
        if (a.act !== "rest") a.act = "rest", a.t = 0;
      }
      if (run3.pt >= TUNE.campTime) {
        run3.phase = "fade";
        run3.pt = 0;
        run3.depth = meta.depth;
        run3.rooms = pickRooms(run3);
        run3.room = 0;
        run3.x = 180;
        run3.units = [];
        formSquad(run3, meta.squad, "walk");
        run3.waves = wavesFor(run3);
        run3.fx = [];
      }
      return res;
    }
    if (run3.phase === "fade" || run3.phase === "retreat") {
      if (run3.phase === "retreat" && run3.pt >= 1.2) {
        run3.phase = "camp";
        run3.pt = 0;
        run3.units = [];
        run3.x = 220;
        formSquad(run3, meta.squad, "rest");
        campLayout(run3);
        for (const a of agents(run3)) a.hp = Math.max(1, a.max * 0.3);
      }
      if (run3.phase === "fade" && run3.pt >= 0.6) run3.phase = "patrol", run3.pt = 0;
      return res;
    }
    for (const u of team) stepAgent(run3, u, dt, foes, team);
    for (const u of foes) stepEnemy(run3, u, dt, team);
    for (const f of foes) {
      if (f.act === "dead" && f.gone > 1.6) {
        meta.kills[f.kind] = (meta.kills[f.kind] ?? 0) + 1;
        res.changed = true;
      }
    }
    run3.units = run3.units.filter((u) => !(u.side === "e" && u.act === "dead" && u.gone > 1.6));
    const livingFoes = enemies(run3).some(alive);
    if (!team.some(alive)) {
      meta.wipes += 1;
      res.changed = true;
      run3.phase = "retreat";
      run3.pt = 0;
      return res;
    }
    const engaged = livingFoes && enemies(run3).some((e) => alive(e) && e.x - run3.x < 130);
    if (!engaged) {
      const lagging = team.some((a) => alive(a) && run3.x - SLOT_BACK[a.slot] - a.x > 30);
      if (!lagging) run3.x += TUNE.walkSpeed * dt;
    }
    if (run3.waves.length && run3.x >= run3.waves[0] - 140) {
      run3.waves.shift();
      spawnWave(run3, viewW);
    }
    if (!livingFoes && !run3.waves.length && run3.x >= TUNE.roomLen && !enemies(run3).length) {
      if (run3.room < 2) {
        run3.room += 1;
        run3.phase = "fade";
        run3.pt = 0;
        run3.x = 180;
        for (const a of team) {
          if (!alive(a)) a.act = "rise", a.t = 0, a.hp = a.max * 0.25;
          a.x = run3.x - SLOT_BACK[a.slot];
        }
        run3.waves = wavesFor(run3);
      } else {
        meta.patrols += 1;
        meta.best = Math.max(meta.best, run3.depth);
        meta.depth = run3.depth + 1;
        res.changed = true;
        run3.phase = "camp";
        run3.pt = 0;
        run3.units = [];
        run3.x = 220;
        formSquad(run3, meta.squad, "rest");
        campLayout(run3);
        for (const a of agents(run3)) a.hp = Math.max(1, a.max * 0.5);
      }
    }
    return res;
  }

  function bindFootnotes(root2) {
    const doc = root2.querySelector(".nw-doc");
    if (!doc?.querySelector(".nw-fn")) return;
    const hover = matchMedia("(hover: hover) and (pointer: fine)").matches;
    const pop = document.createElement("div");
    pop.className = "nw-fnpop";
    pop.hidden = true;
    doc.appendChild(pop);
    let cur = null;
    const hide = () => {
      pop.hidden = true;
      cur = null;
    };
    const show = (sup) => {
      const li = doc.querySelector(`.nw-notes li[data-fn="${sup.dataset.fn}"]`);
      if (!li) return;
      cur = sup;
      pop.innerHTML = `<b>[${sup.dataset.fn}]</b> ${li.innerHTML}`;
      pop.hidden = false;
      const box = doc.getBoundingClientRect();
      const r = sup.getBoundingClientRect();
      const w = Math.min(360, box.width - 24);
      pop.style.width = `${w}px`;
      const left = Math.max(12, Math.min(r.left - box.left - w / 2 + r.width / 2, box.width - w - 12));
      let top = r.bottom - box.top + doc.scrollTop + 6;
      if (r.bottom + pop.offsetHeight + 12 > box.bottom && r.top - pop.offsetHeight - 12 > box.top) top = r.top - box.top + doc.scrollTop - pop.offsetHeight - 6;
      pop.style.left = `${left}px`;
      pop.style.top = `${top}px`;
    };
    const supOf = (e) => e.target.closest(".nw-fn");
    if (hover) {
      doc.addEventListener("mouseover", (e) => {
        const sup = supOf(e);
        if (sup && sup !== cur) show(sup);
      });
      doc.addEventListener("mouseout", (e) => {
        if (supOf(e) && !e.relatedTarget?.closest?.(".nw-fn")) hide();
      });
    }
    doc.addEventListener("click", (e) => {
      const sup = supOf(e);
      if (!sup) {
        if (!hover && !e.target.closest(".nw-fnpop")) hide();
        return;
      }
      e.preventDefault();
      if (!hover) {
        if (cur === sup) hide();
        else show(sup);
        return;
      }
      hide();
      const li = doc.querySelector(`.nw-notes li[data-fn="${sup.dataset.fn}"]`);
      if (!li) return;
      li.scrollIntoView({ behavior: "smooth", block: "center" });
      li.classList.remove("hit");
      void li.offsetWidth;
      li.classList.add("hit");
    });
    doc.addEventListener("scroll", () => !hover && hide(), { passive: true });
  }

  function sendMission(ctx, env2, opening) {

    const support = field2.support;
    return compose2(ctx, env2, (latest, x) => {
      if (!name) throw new DirectiveError("목표가 정해지지 않았어요.");

      if (support > 0 && !x.supportOn) throw new DirectiveError("지원 투입이 꺼져 있어요. 설정 탭에서 켜거나 지원 인원을 빼 주세요.");

  async function handleField(action, value, ctx, env2, opening) {
    const v = value ?? "";
    switch (action) {
      case "map-region":
        field2.region = v === "01" || MAP_CODES.includes(v) ? v : "01";
        field2.place = null;
        field2.confirmEnd = null;
        env2.render();
        return true;
      case "map-jump":
      case "map-place": {
        let v2 = value ?? "";
        if (action === "map-jump") {
          const [code, key] = v2.split("|");
          if (!MAP_CODES.includes(code) || !key) return true;
          field2.region = code;
          field2.mapq = "";
          v2 = key;
        }
        field2.confirmEnd = null;
        if (action === "map-place" && isPC() && field2.place === v2 && sameTarget(field2.target, v2)) {
          field2.place = null;
          field2.target = null;
          env2.render();
          return true;
        }
        field2.place = v2;
        const x = readCtx(ctx);
        const here = typeof x === "string" || !x.on ? [] : activeEvents(x.world).filter((e) => placeKey(e, x.catalog) === v2);
        if (!here.length) field2.target = placeTarget(v2);
        env2.render();
        return true;
      }
      case "map-unplace":
        field2.place = null;
        field2.target = null;
        field2.confirmEnd = null;
        toStep("goal", env2);
        return true;
      case "map-filter":
        field2.filter = ["all", "an", "in", "ms"].includes(v) ? v : "all";
        env2.render();
        return true;
      case "pick-place":
        if (isPC() && sameTarget(field2.target, v)) {
          field2.target = null;
          env2.render();
          return true;
        }
        field2.target = placeTarget(v);
        toStep("team", env2);
        return true;
      case "field-step":
        toStep(v === "team" ? "team" : "goal", env2);
        return true;
      case "pick-event": {
        const x = readCtx(ctx);
        if (typeof x === "string") return true;
        const ev = activeEvents(x.world).find((e) => e.key === v);
        if (!ev) return true;
        if (isPC() && sameTarget(field2.target, ev.key)) {
          field2.target = null;
          field2.confirmEnd = null;
          env2.render();
          return true;
        }
        field2.target = { type: "event", key: ev.key, kind: ev.kind, name: ev.name, loc: ev.loc };
        field2.region = sidoOf(ev.loc);
        field2.confirmEnd = null;
        toStep("team", env2);
        return true;
      }
      case "end-event":
        if (field2.confirmEnd !== v) {
          field2.confirmEnd = v;
          env2.render();
          return true;
        }
        field2.confirmEnd = null;
        if (field2.target?.type === "event" && field2.target.key === v) field2.target = null;
        await saveWorld(ctx, env2, (w) => endEvent(w, v) ? null : "이미 끝났거나 바뀐 상황이에요.");
        return true;
      case "team-toggle": {
        const tx = readCtx(ctx);
        if (typeof tx !== "string" && lockedState(tx, v)) return true;
        const i = field2.team.indexOf(v);
        if (i >= 0) field2.team.splice(i, 1);
        else if (field2.team.length + field2.support < TEAM_MAX) field2.team.push(v);
        env2.render();
        return true;
      }
      case "support": {
        const next = field2.support + (Number(v) > 0 ? 1 : -1);
        if (next >= 0 && validTeam(field2.team.length, next)) field2.support = next;
        env2.render();
        return true;
      }
      case "depart":
        await depart(ctx, env2);
        return true;
      case "mission-kind":
        if (v === "an" || v === "ms" || v === "lm") {
          field2.mkind = v;
          field2.query = "";
        }
        env2.render();
        return true;
      case "mission-pick": {
        const m = v.match(/^(an|ms|lm):([\s\S]*)$/);
        const kind = m?.[1] ?? field2.mkind;
        const name = m ? m[2] : v;
        if (isPC() && field2.mname === name && field2.mpick === kind) {
          field2.mname = null;
          env2.render();
          return true;
        }
        field2.mname = name;
        field2.mpick = kind;
        toStep("team", env2);
        return true;
      }
      case "mission-cancel":
        if (v === "an" || v === "ms" || v === "in") await saveWorld(ctx, env2, (w) => (cancelMission(w, v), null));
        return true;
      case "mission-unpick":
        field2.mname = null;
        toStep("goal", env2);
        return true;
      case "mission-send":
        await sendMission(ctx, env2, opening);
        return true;
    }
    return false;
  }
