  function formSquad(run3, squad, act) {
    const order = [...squad].sort((a, b) => agentDef(a).range - agentDef(b).range);
    const keep = new Map(run3.units.filter(isAgent).map((u) => [u.kind, u]));
    run3.units = run3.units.filter((u) => !isAgent(u));
    order.forEach((id, slot) => {
      const u = makeAgent(run3, id, slot, run3.x);
      const old = keep.get(id);
      if (old) u.hp = Math.max(1, old.hp), u.gauge = old.gauge;
      u.act = act;
      run3.units.push(u);
    });

  function spawnWave(run3, viewW) {
    const edge = Math.max(run3.x + 160, run3.cam + viewW + 30);
    const last = run3.room === 2 && run3.waves.length === 0;
    const regular = ["corner", "watchman"];
    if (last) {
      const boss = run3.depth % 2 === 1 ? "statue" : "master";
      run3.units.push(makeEnemy(run3, boss, edge + 60, 1));
      run3.units.push(makeEnemy(run3, regular[Math.floor(rand(run3) * 2)], edge + 10, 0));
      return;
    }
    const n = Math.min(5, 2 + Math.floor(rand(run3) * 2) + Math.floor((run3.depth - 1) / 2));
    for (let i = 0; i < n; i++) {
      run3.units.push(makeEnemy(run3, regular[Math.floor(rand(run3) * 2)], edge + i * 46 + rand(run3) * 20, i % 2));
    }

  function addNum(run3, u, amount, heal = false, big = false) {
    const h = u.side === "e" && ENEMIES[u.kind].scale === 2 ? 170 : 96;
    let y = footY(u) - h;
    while (run3.fx.some((f) => (f.kind === "num" || f.kind === "heal") && f.t < 0.35 && Math.abs(f.x - u.x) < 20 && Math.abs(f.y - y) < 12)) y -= 13;
    run3.fx.push({ kind: heal ? "heal" : "num", x: u.x + (rand(run3) * 10 - 5), y, t: 0, life: 0.8, text: String(Math.round(amount)), big });
  }
  function footY(u) {
    return u.lane === 1 ? 336 : 318;

  function damage(run3, from, to, amount, big = false) {
    if (!alive(to)) return;
    let v = amount * (0.9 + rand(run3) * 0.2);
    if (to.side === "a" && run3.guard > 0) v *= 0.5;
    if (from.side === "a" && run3.rally > 0) v *= 1.4;
    to.hp = Math.max(0, to.hp - v);
    to.flash = 0.12;
    addNum(run3, to, v, false, big);
    if (to.hp <= 0) {
      to.act = to.side === "a" ? "down" : "dead";
      to.t = 0;
    } else if (to.act === "idle" || to.act === "walk") {
      to.act = "hurt";
      to.t = 0;
    }
  }
  function nearest(from, pool) {
    let best;
    for (const u of pool) if (alive(u) && (!best || Math.abs(u.x - from.x) < Math.abs(best.x - from.x))) best = u;
    return best;
  }
  var ATTACK_TIME = 0.5;
  var SKILL_TIME = 0.9;
  var HURT_TIME = 0.22;
  var DOWN_TIME = 0.6;
  var DIE_TIME = 0.8;
  function frameOf(u) {
    if (u.side === "a") {
      switch (u.act) {
        case "walk":
          return AF.walk[Math.floor(u.stride / TUNE.strideStep) % AF.walk.length];
        case "attack":
          return AF.attack[Math.min(AF.attack.length - 1, Math.floor(u.t / ATTACK_TIME * AF.attack.length))];
        case "skill":
          return AF.skill[Math.min(AF.skill.length - 1, Math.floor(u.t / SKILL_TIME * AF.skill.length))];
        case "hurt":
          return AF.hurt;
        case "down":
          return AF.down[Math.min(AF.down.length - 1, Math.floor(u.t / DOWN_TIME * AF.down.length))];
        case "rise":
          return AF.rise[Math.min(AF.rise.length - 1, Math.floor(u.t / 0.4 * AF.rise.length))];
        case "rest":
          return AF.rest[(Math.floor(u.t / 2.2) + u.slot) % AF.rest.length];
        default:
          return AF.idle;
      }
    }
    switch (u.act) {
      case "walk":
        return EF.move[Math.floor(u.stride / 7) % EF.move.length];
      case "attack":
        return EF.attack[Math.min(EF.attack.length - 1, Math.floor(u.t / enemyAttackTime(u) * EF.attack.length))];
      case "skill":
        return EF.skill[Math.min(EF.skill.length - 1, Math.floor(u.t / SKILL_TIME * EF.skill.length))];
      case "hurt":
        return EF.hurt;
      case "dead":
        return EF.die[Math.min(EF.die.length - 1, Math.floor(u.t / DIE_TIME * EF.die.length))];
      default:
        return EF.move[Math.floor(u.t * 4) % 2];
    }
  }
  var enemyAttackTime = (u) => ENEMIES[u.kind].boss ? 0.8 : 0.6;
  function trigger(run3, uid) {
    const u = run3.units.find((x) => x.uid === uid);
    if (!u || !alive(u) || u.gauge < 1 || run3.phase !== "patrol") return false;
    if (!enemies(run3).some(alive)) return false;
    startSkill(u);
    return true;
  }
  function startSkill(u) {
    u.act = "skill";
    u.t = 0;
    u.hit = false;
    u.gauge = 0;
    u.ready = 0;
  }
  function applySkill(run3, u, foes, team) {
    const d = agentDef(u.kind);
    const target = nearest(u, foes);
    switch (d.skill) {
      case "blast": {
        if (!target) return;
        run3.fx.push({ kind: "boom", x: target.x, y: footY(target) - 40, t: 0, life: 0.5 });
        for (const f of foes) if (alive(f) && Math.abs(f.x - target.x) < 110) damage(run3, u, f, d.atk * d.power, true);
        return;
      }
      case "heal":
        for (const a of team) {
          if (!alive(a)) continue;
          const v = Math.min(a.max - a.hp, a.max * d.power);
          a.hp += v;
          if (v > 0) addNum(run3, a, v, true);
        }
        run3.fx.push({ kind: "ring", x: u.x, y: footY(u) - 40, t: 0, life: 0.7 });
        return;
      case "guard":
        run3.guard = d.power;
        run3.fx.push({ kind: "ring", x: u.x, y: footY(u) - 44, t: 0, life: 0.6 });
        return;
      case "rally":
        run3.rally = d.power;
        run3.fx.push({ kind: "ring", x: u.x, y: footY(u) - 44, t: 0, life: 0.6 });
        return;
      case "bind":
        if (!target) return;
        for (const f of foes) {
          if (!alive(f) || Math.abs(f.x - target.x) > 70) continue;
          f.bound = ENEMIES[f.kind].boss ? d.power / 2 : d.power;
          damage(run3, u, f, d.atk * 1.5);
          run3.fx.push({ kind: "bind", x: f.x, y: footY(f) - 30, t: 0, life: f.bound });
        }
        return;
    }
  }
  function stepAgent(run3, u, dt, foes, team) {
    const d = agentDef(u.kind);
    u.t += dt;
    u.flash = Math.max(0, u.flash - dt);
    if (u.act === "down" || u.act === "rest") return;
    if (u.act === "rise") {
      if (u.t > 0.4) u.act = "idle", u.t = 0;
      return;
    }
    u.cd -= dt;
    if (foes.some(alive)) u.gauge = Math.min(1, u.gauge + dt / d.charge);
    if (u.act === "hurt") {
      if (u.t > HURT_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "attack") {
      const target2 = nearest(u, foes);
      if (!u.hit && u.t >= ATTACK_TIME * AF.attackHit / AF.attack.length) {
        u.hit = true;
        if (target2 && Math.abs(target2.x - u.x) <= d.range + 12) damage(run3, u, target2, d.atk);
      }
      if (u.t >= ATTACK_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "skill") {
      if (!u.hit && u.t >= SKILL_TIME * AF.skillHit / AF.skill.length) {
        u.hit = true;
        applySkill(run3, u, foes, team);
      }
      if (u.t >= SKILL_TIME) u.act = "idle", u.t = 0;
      return;
    }
    const target = nearest(u, foes);
    if (u.gauge >= 1) {
      u.ready += dt;
      if (target && u.ready > 1.8 && target.x - u.x < d.range + 40) return startSkill(u);
    }
    const home = run3.x - SLOT_BACK[u.slot];
    let goal = home;
    if (target && target.x - u.x < d.range + 140) {
      goal = Math.min(Math.max(home, target.x - d.range + 8), target.x - 18);
    }
    const dx = goal - u.x;
    if (Math.abs(dx) > 0.5) {
      const step2 = Math.sign(dx) * Math.min(Math.abs(dx), TUNE.walkSpeed * 1.25 * dt);
      u.x += step2;
      u.stride += Math.abs(step2);
      if (u.act !== "walk") u.act = "walk", u.t = 0;
      return;
    }
    if (target && Math.abs(target.x - u.x) <= d.range + 4 && u.cd <= 0) {
      u.act = "attack";
      u.t = 0;
      u.hit = false;
      u.cd = d.rate;
      return;
    }
    if (u.act !== "idle") u.act = "idle", u.t = 0;
  }
  function stepEnemy(run3, u, dt, team) {
    const d = ENEMIES[u.kind];
    u.t += dt;
    u.flash = Math.max(0, u.flash - dt);
    if (u.act === "dead") {
      u.gone += dt;
      return;
    }
    if (u.bound > 0) {
      u.bound = Math.max(0, u.bound - dt);
      if (u.act === "walk" || u.act === "attack") u.act = "idle", u.t = 0;
      return;
    }
    u.cd -= dt;
    if (u.act === "hurt") {
      if (u.t > HURT_TIME) u.act = "idle", u.t = 0;
      return;
    }
    const guardian = run3.guard > 0 ? team.find((a) => a.kind === "gorgo" && alive(a)) : void 0;
    const target = guardian ?? nearest(u, team);
    if (u.act === "attack") {
      const at = enemyAttackTime(u);
      if (!u.hit && u.t >= at * EF.attackHit / EF.attack.length) {
        u.hit = true;
        if (target && Math.abs(target.x - u.x) <= d.range + 14) damage(run3, u, target, scaled(run3, d.atk));
      }
      if (u.t >= at) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "skill") {
      if (!u.hit && u.t >= SKILL_TIME * EF.skillHit / EF.skill.length) {
        u.hit = true;
        run3.fx.push({ kind: "boom", x: u.x - 70, y: footY(u) - 30, t: 0, life: 0.5 });
        for (const a of team) if (alive(a) && u.x - a.x < 170) damage(run3, u, a, scaled(run3, d.atk) * 1.3, true);
      }
      if (u.t >= SKILL_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (!target) {
      if (u.act !== "idle") u.act = "idle", u.t = 0;
      return;
    }
    if (d.boss) {
      u.gauge -= dt;
      if (u.gauge <= 0 && u.x - target.x < 190) {
        u.gauge = 7 + rand(run3) * 3;
        u.act = "skill";
        u.t = 0;
        u.hit = false;
        return;
      }
    }
    const gap = u.x - target.x;
    if (gap > d.range) {
      const step2 = Math.min(gap - d.range, d.speed * dt);
      u.x -= step2;
      u.stride += step2;
      if (u.act !== "walk") u.act = "walk", u.t = 0;
      return;
    }
    if (u.cd <= 0) {
      u.act = "attack";
      u.t = 0;
      u.hit = false;
      u.cd = d.rate;
      return;
    }
    if (u.act !== "idle") u.act = "idle", u.t = 0;
  }
  function step(run3, meta, dt, viewW) {
    const res = { changed: false };
    run3.pt += dt;
    run3.guard = Math.max(0, run3.guard - dt);
    run3.rally = Math.max(0, run3.rally - dt);
    for (const f of run3.fx) f.t += dt;
    run3.fx = run3.fx.filter((f) => f.t < f.life);
    const team = agents(run3);
    const foes = enemies(run3);
    if (run3.phase === "camp") {
      for (const a of team) {
        a.t += dt;
        a.hp = Math.min(a.max, a.hp + a.max * 0.25 * dt);
        if (a.act !== "rest") a.act = "rest", a.t = 0;
      }
      if (run3.pt >= TUNE.campTime) {
        run3.phase = "fade";
        run3.pt = 0;
        run3.depth = meta.depth;
        run3.rooms = pickRooms(run3);
        run3.room = 0;
        run3.x = 180;
        run3.units = [];
        formSquad(run3, meta.squad, "walk");
        run3.waves = wavesFor(run3);
        run3.fx = [];
      }
      return res;
    }
    if (run3.phase === "fade" || run3.phase === "retreat") {
      if (run3.phase === "retreat" && run3.pt >= 1.2) {
        run3.phase = "camp";
        run3.pt = 0;
        run3.units = [];
        run3.x = 220;
        formSquad(run3, meta.squad, "rest");
        campLayout(run3);
        for (const a of agents(run3)) a.hp = Math.max(1, a.max * 0.3);
      }
      if (run3.phase === "fade" && run3.pt >= 0.6) run3.phase = "patrol", run3.pt = 0;
      return res;
    }
    for (const u of team) stepAgent(run3, u, dt, foes, team);
    for (const u of foes) stepEnemy(run3, u, dt, team);
    for (const f of foes) {
      if (f.act === "dead" && f.gone > 1.6) {
        meta.kills[f.kind] = (meta.kills[f.kind] ?? 0) + 1;
        res.changed = true;
      }
    }
    run3.units = run3.units.filter((u) => !(u.side === "e" && u.act === "dead" && u.gone > 1.6));
    const livingFoes = enemies(run3).some(alive);
    if (!team.some(alive)) {
      meta.wipes += 1;
      res.changed = true;
      run3.phase = "retreat";
      run3.pt = 0;
      return res;
    }
    const engaged = livingFoes && enemies(run3).some((e) => alive(e) && e.x - run3.x < 130);
    if (!engaged) {
      const lagging = team.some((a) => alive(a) && run3.x - SLOT_BACK[a.slot] - a.x > 30);
      if (!lagging) run3.x += TUNE.walkSpeed * dt;
    }
    if (run3.waves.length && run3.x >= run3.waves[0] - 140) {
      run3.waves.shift();
      spawnWave(run3, viewW);
    }
    if (!livingFoes && !run3.waves.length && run3.x >= TUNE.roomLen && !enemies(run3).length) {
      if (run3.room < 2) {
        run3.room += 1;
        run3.phase = "fade";
        run3.pt = 0;
        run3.x = 180;
        for (const a of team) {
          if (!alive(a)) a.act = "rise", a.t = 0, a.hp = a.max * 0.25;
          a.x = run3.x - SLOT_BACK[a.slot];
        }
        run3.waves = wavesFor(run3);
      } else {
        meta.patrols += 1;
        meta.best = Math.max(meta.best, run3.depth);
        meta.depth = run3.depth + 1;
        res.changed = true;
        run3.phase = "camp";
        run3.pt = 0;
        run3.units = [];
        run3.x = 220;
        formSquad(run3, meta.squad, "rest");
        campLayout(run3);
        for (const a of agents(run3)) a.hp = Math.max(1, a.max * 0.5);
      }
    }
    return res;
  }

  async function handleWiki(action, value, ctx2, env2) {
    const v = value ?? "";
    if (action !== "wiki-delete") wiki.confirmDelete = null;
    if (action.startsWith("wiki-def-")) {
      try {
        const msg = await defaultsAction("wiki", action, v, ctx2, env2.refresh);
        if (msg) wiki.info = msg;
      } catch (error) {
        wiki.notice = `기본값을 처리하지 못했어요: ${error instanceof Error ? error.message : String(error)}`;
      }
      env2.render();
      return true;
    }
    switch (action) {
      case "wiki-home":
        go("home");
        wiki.history = [];
        wiki.listOpen = false;
        break;
      case "wiki-back": {
        const prev = wiki.history.pop();
        if (prev && prev.view !== "home") Object.assign(wiki, { view: prev.view, title: prev.title }, { confirmDelete: null, notice: "" });
        else Object.assign(wiki, { view: "home", title: "", notice: "" }, { history: [] });
        break;
      }
      case "wiki-list":
        wiki.listOpen = v === "open" ? true : v === "close" ? false : !wiki.listOpen;
        env2.render();
        if (wiki.listOpen && wide() && !v) document.querySelector(".nw .ph-search")?.focus();
        return true;
      case "wiki-find":
        wiki.listOpen = true;
        env2.render();
        document.querySelector(".nw .ph-search")?.focus();
        return true;
      case "wiki-tab":
        wiki.tab = TABS.some(([t]) => t === v) ? v : "all";
        break;
      case "wiki-browse":
        Object.assign(wiki, { tab: TABS.some(([t]) => t === v) ? v : "all", listOpen: true, query: "" });
        break;
      case "wiki-search":
        wiki.homeQuery = "";
        search(ctx2, v);
        if (wiki.view === "compose" || wiki.view === "doc") wiki.listOpen = false;
        break;
      case "wiki-pick":
        if ((wiki.view === "doc" || wiki.view === "compose") && canon(ctx2)(v) === wiki.title && wide()) {
          Object.assign(wiki, { view: "home", title: "", history: [] });
          break;
        }
        wiki.history = [];
        wiki.homeQuery = "";
        openTitle(ctx2, v, "search");
        if (wiki.view !== "home") wiki.listOpen = false;
        break;
      case "wiki-open":
        openTitle(ctx2, v, wiki.view === "doc" ? "link" : "search");
        break;
      case "wiki-write":
        if (!v.trim()) break;
        wiki.newTitle = "";
        openTitle(ctx2, v, "direct");
        if (wiki.view === "compose" || wiki.view === "doc") wiki.listOpen = false;
        break;
      case "wiki-new":
        if (wiki.view === "new") go("home");
        else go("new");
        wiki.listOpen = false;
        break;
      case "wiki-start":
        if (wiki.view === "compose" && wiki.title === v) void generate3(ctx2, env2, v, false);
        break;
      case "wiki-rewrite":
        if (!v) break;
        if (wiki.view !== "doc") go("doc", v);
        void generate3(ctx2, env2, v, true);
        break;
      case "wiki-delete":
        if (wiki.confirmDelete !== v) {
          wiki.confirmDelete = v;
          break;
        }
        wiki.confirmDelete = null;
        if (wiki.title === v && wiki.view === "doc" && !baseDocs(ctx2.char.globalLore).some((d) => d.title === v)) Object.assign(wiki, { view: "home", title: "", history: [] });
        await updateChat(ctx2.charIndex, ctx2.chatIndex, (chat) => deleteDoc(chat, v));
        await env2.refresh();
        return true;
      case "wiki-jump": {
        document.getElementById(v)?.scrollIntoView({ behavior: "smooth", block: "start" });
        return true;
      }
      case "wiki-request":
        await setVar2(ctx2, env2, "$nw_writing_request", v.trim());
        return true;
      case "wiki-length":
      case "wiki-mode":
      case "wiki-model":
      case "wiki-refresh_previous":
      case "wiki-context":
        await setVar2(ctx2, env2, `$nw_${action.slice(5)}`, v);
        return true;
      case "wiki-settings":
        if (wiki.view === "settings") go("home");
        else go("settings");
        wiki.listOpen = false;
        break;
      default:
        return false;
    }
    env2.render();
    return true;
  }
