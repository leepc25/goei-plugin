  function stepAgent(run3, u, dt, foes, team) {
    const d = agentDef(u.kind);
    u.t += dt;
    u.flash = Math.max(0, u.flash - dt);
    if (u.act === "down" || u.act === "rest") return;
    if (u.act === "rise") {
      if (u.t > 0.4) u.act = "idle", u.t = 0;
      return;
    }
    u.cd -= dt;
    if (foes.some(alive)) u.gauge = Math.min(1, u.gauge + dt / d.charge);
    if (u.act === "hurt") {
      if (u.t > HURT_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "attack") {
      const target2 = nearest(u, foes);
      if (!u.hit && u.t >= ATTACK_TIME * AF.attackHit / AF.attack.length) {
        u.hit = true;
        if (target2 && Math.abs(target2.x - u.x) <= d.range + 12) damage(run3, u, target2, d.atk);
      }
      if (u.t >= ATTACK_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "skill") {
      if (!u.hit && u.t >= SKILL_TIME * AF.skillHit / AF.skill.length) {
        u.hit = true;
        applySkill(run3, u, foes, team);
      }
      if (u.t >= SKILL_TIME) u.act = "idle", u.t = 0;
      return;
    }
    const target = nearest(u, foes);
    if (u.gauge >= 1) {
      u.ready += dt;
      if (target && u.ready > 1.8 && target.x - u.x < d.range + 40) return startSkill(u);
    }
    const home = run3.x - SLOT_BACK[u.slot];
    let goal = home;
    if (target && target.x - u.x < d.range + 140) {
      goal = Math.min(Math.max(home, target.x - d.range + 8), target.x - 18);
    }
    const dx = goal - u.x;
    if (Math.abs(dx) > 0.5) {
      const step3 = Math.sign(dx) * Math.min(Math.abs(dx), TUNE.walkSpeed * 1.25 * dt);
      u.x += step3;
      u.stride += Math.abs(step3);
      if (u.act !== "walk") u.act = "walk", u.t = 0;
      return;
    }
    if (target && Math.abs(target.x - u.x) <= d.range + 4 && u.cd <= 0) {
      u.act = "attack";
      u.t = 0;
      u.hit = false;
      u.cd = d.rate;
      return;
    }
    if (u.act !== "idle") u.act = "idle", u.t = 0;
  }
  function stepEnemy(run3, u, dt, team) {
    const d = ENEMIES[u.kind];
    u.t += dt;
    u.flash = Math.max(0, u.flash - dt);
    if (u.act === "dead") {
      u.gone += dt;
      return;
    }
    if (u.bound > 0) {
      u.bound = Math.max(0, u.bound - dt);
      if (u.act === "walk" || u.act === "attack") u.act = "idle", u.t = 0;
      return;
    }
    u.cd -= dt;
    if (u.act === "hurt") {
      if (u.t > HURT_TIME) u.act = "idle", u.t = 0;
      return;
    }
    const guardian = run3.guard > 0 ? team.find((a) => a.kind === "gorgo" && alive(a)) : void 0;
    const target = guardian ?? nearest(u, team);
    if (u.act === "attack") {
      const at = enemyAttackTime(u);
      if (!u.hit && u.t >= at * EF.attackHit / EF.attack.length) {
        u.hit = true;
        if (target && Math.abs(target.x - u.x) <= d.range + 14) damage(run3, u, target, scaled(run3, d.atk));
      }
      if (u.t >= at) u.act = "idle", u.t = 0;
      return;
    }
    if (u.act === "skill") {
      if (!u.hit && u.t >= SKILL_TIME * EF.skillHit / EF.skill.length) {
        u.hit = true;
        run3.fx.push({ kind: "boom", x: u.x - 70, y: footY(u) - 30, t: 0, life: 0.5 });
        for (const a of team) if (alive(a) && u.x - a.x < 170) damage(run3, u, a, scaled(run3, d.atk) * 1.3, true);
      }
      if (u.t >= SKILL_TIME) u.act = "idle", u.t = 0;
      return;
    }
    if (!target) {
      if (u.act !== "idle") u.act = "idle", u.t = 0;
      return;
    }
    if (d.boss) {
      u.gauge -= dt;
      if (u.gauge <= 0 && u.x - target.x < 190) {
        u.gauge = 7 + rand(run3) * 3;
        u.act = "skill";
        u.t = 0;
        u.hit = false;
        return;
      }
    }
    const gap = u.x - target.x;
    if (gap > d.range) {
      const step3 = Math.min(gap - d.range, d.speed * dt);
      u.x -= step3;
      u.stride += step3;
      if (u.act !== "walk") u.act = "walk", u.t = 0;
      return;
    }
    if (u.cd <= 0) {
      u.act = "attack";
      u.t = 0;
      u.hit = false;
      u.cd = d.rate;
      return;
    }
    if (u.act !== "idle") u.act = "idle", u.t = 0;
  }
  function step(run3, meta, dt, viewW) {
    const res = { changed: false };

    run3.guard = Math.max(0, run3.guard - dt);
    run3.rally = Math.max(0, run3.rally - dt);

    if (run3.phase === "camp") {
      for (const a of team) {
        a.t += dt;

        if (a.act !== "rest") a.act = "rest", a.t = 0;
      }
      if (run3.pt >= TUNE.campTime) {
        run3.phase = "fade";
        run3.pt = 0;
        run3.depth = meta.depth;
        run3.rooms = pickRooms(run3);
        run3.room = 0;
        run3.x = 180;
        run3.units = [];
        formSquad(run3, meta.squad, "walk");
        run3.waves = wavesFor(run3);
        run3.fx = [];
      }
      return res;
    }
    if (run3.phase === "fade" || run3.phase === "retreat") {
      if (run3.phase === "retreat" && run3.pt >= 1.2) {
        run3.phase = "camp";
        run3.pt = 0;
        run3.units = [];
        run3.x = 220;
        formSquad(run3, meta.squad, "rest");
        campLayout(run3);
        for (const a of agents(run3)) a.hp = Math.max(1, a.max * 0.3);
      }
      if (run3.phase === "fade" && run3.pt >= 0.6) run3.phase = "patrol", run3.pt = 0;
      return res;
    }
    for (const u of team) stepAgent(run3, u, dt, foes, team);
    for (const u of foes) stepEnemy(run3, u, dt, team);
    for (const f of foes) {
      if (f.act === "dead" && f.gone > 1.6) {
        meta.kills[f.kind] = (meta.kills[f.kind] ?? 0) + 1;
        res.changed = true;
      }
    }
    run3.units = run3.units.filter((u) => !(u.side === "e" && u.act === "dead" && u.gone > 1.6));
    const livingFoes = enemies(run3).some(alive);
    if (!team.some(alive)) {
      meta.wipes += 1;
      res.changed = true;
      run3.phase = "retreat";
      run3.pt = 0;
      return res;
    }
    const engaged = livingFoes && enemies(run3).some((e) => alive(e) && e.x - run3.x < 130);
    if (!engaged) {
      const lagging = team.some((a) => alive(a) && run3.x - SLOT_BACK[a.slot] - a.x > 30);
      if (!lagging) run3.x += TUNE.walkSpeed * dt;
    }
    if (run3.waves.length && run3.x >= run3.waves[0] - 140) {
      run3.waves.shift();
      spawnWave(run3, viewW);
    }
    if (!livingFoes && !run3.waves.length && run3.x >= TUNE.roomLen && !enemies(run3).length) {
      if (run3.room < 2) {
        run3.room += 1;
        run3.phase = "fade";
        run3.pt = 0;
        run3.x = 180;
        for (const a of team) {
          if (!alive(a)) a.act = "rise", a.t = 0, a.hp = a.max * 0.25;
          a.x = run3.x - SLOT_BACK[a.slot];
        }
        run3.waves = wavesFor(run3);
      } else {
        meta.patrols += 1;
        meta.best = Math.max(meta.best, run3.depth);
        meta.depth = run3.depth + 1;
        res.changed = true;
        run3.phase = "camp";
        run3.pt = 0;
        run3.units = [];
        run3.x = 220;
        formSquad(run3, meta.squad, "rest");
        campLayout(run3);
        for (const a of agents(run3)) a.hp = Math.max(1, a.max * 0.5);
      }
    }
    return res;
  }

  function clipIndex(phase, frames, times) {
    if (!frames.length) return 0;
    const p = Math.max(0, Math.min(0.999999, phase));
    if (!times || times.length !== frames.length) return Math.floor(p * frames.length);
    const duration = times.reduce((sum, value) => sum + value, 0);
    if (duration <= 0) return 0;
    let cursor = p * duration;
    for (let i = 0; i < times.length; i++) {
      if (cursor < times[i]) return i;
      cursor -= times[i];
    }
    return frames.length - 1;
  }
