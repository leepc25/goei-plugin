  async function handleDirectory(action, value, ctx2, env2) {
    if (!action.startsWith("dir-")) return false;

          dirUi.q = "";
          dirUi.adding = false;
        }
        break;
      case "dir-open":
        dirUi.open = dirUi.open === v ? "" : v;
        break;
      case "dir-move": {
        const [how, lore, target] = v.split("|");
        if ((how === "up" || how === "down" || how === "before" || how === "after") && moveAgent(d, lore ?? "", how, target ?? "")) dirUi.done = "";
        break;
      }
      case "dir-jump": {
        const cut2 = v.indexOf("|");
        const kind = v.slice(0, cut2);
        if (kind === "anomaly" || kind === "site" || kind === "agent" || kind === "org" || kind === "base") {
          Object.assign(dirUi, { kind, open: v.slice(cut2 + 1), q: "", adding: false });
          dirUi.scrollTo = v.slice(cut2 + 1);
        }
        break;
      }
      case "dir-set":
      case "dir-toggle": {
        const [field4, arg] = v.split("|", 2);
        const row5 = findRow2(d, v.slice(field4.length + arg.length + 2));
        if (!row5) break;
        if (field4 === "preset") row5.preset = arg === "1";
        else if (field4 === "affiliation") row5.affiliation = arg;
        else if (field4 === "rosterFor" || field4 === "phoneTabs") {
          const cur = row5[field4];
          row5[field4] = cur.includes(arg) ? cur.filter((x) => x !== arg) : [...cur, arg];
        }
        dirUi.done = "";
        break;
      }
      case "dir-org-pick": {
        const at = v.indexOf("|");

        if (!row5) break;
        assignOrg(row5, d.orgs.find((o) => o.name === v.slice(0, at)));
        dirUi.done = "";
        break;
      }
      case "dir-org-toggle": {
        const [field4, arg] = v.split("|", 2);
        const o = d.orgs.find((x) => x.name === v.slice(field4.length + arg.length + 2));
        if (!o || field4 !== "rosterFor" && field4 !== "phoneTabs") break;
        o[field4] = o[field4].includes(arg) ? o[field4].filter((x) => x !== arg) : [...o[field4], arg];

        if (!here.length) field2.target = placeTarget(v2);
        env2.render();

      }
      case "map-unplace":
        field2.place = null;
        field2.target = null;
        field2.confirmEnd = null;
        toStep("goal", env2);
        return true;
      case "map-filter":
        field2.filter = ["all", "an", "in", "ms"].includes(v) ? v : "all";
        env2.render();
        return true;
      case "pick-place":
        if (isPC() && sameTarget(field2.target, v)) {

          env2.render();
          return true;
        }
        field2.target = placeTarget(v);
        toStep("team", env2);
        return true;
      case "field-step":
        toStep(v === "team" ? "team" : "goal", env2);
        return true;
      case "pick-event": {
        const x = readCtx(ctx2);
        if (typeof x === "string") return true;

  function nextWorld(w, v) {
    const [key, op] = v.split(":");
    const f = WORLD_SPEC.find((x) => x.key === key);
    if (!f || op === void 0) return null;
    const cur = w[f.key];
    const step3 = worldStep(f.key);
    const n = op === "+" ? cur + step3 : op === "-" ? cur - step3 : Number(op);
    if (!Number.isFinite(n)) return null;
    const clamped = Math.min(f.max ?? 99, Math.max(0, f.fraction ? n : Math.round(n)));
    return clamped === cur ? null : [f.key, clamped];
  }
  async function handleSettings(action, value, ctx2, env2, scope = "chat") {
    const v = value ?? "";
    const lore = ctx2.char.globalLore;
    if (action === "set-bar") {
      settingsUi.confirm = v === "save" || v === "load" ? v : "";
      env2.render();
      return true;
    }
    if (action === "set-bar-no") {
      settingsUi.confirm = "";
      env2.render();
      return true;
    }
    if (action === "set-bar-yes") {
      const what = settingsUi.confirm;
      settingsUi.confirm = "";
      if (what === "save") {
        const cur = current2(ctx2, "chat");
        await saveDefaults(env2, (p) => ({ ...p, basic: { ...cur.basic }, world: { ...cur.world }, theme: cur.theme }));
        if (!settingsUi.notice) await saveVarDefaults(["$gen_"], ctx2.chat.scriptstate).catch((error) => settingsUi.notice = `기본값을 저장하지 못했어요: ${error instanceof Error ? error.message : String(error)}`);

        env2.render();
      } else if (what === "load") {
        const d = current2(ctx2, "defaults");
        await save(ctx2, env2, (chat) => {
          const vars = { ...basicVars(d.basic), [THEME_KEY]: d.theme, ...loadVarDefaults(["$gen_"], chat.scriptstate) };
          for (const f of WORLD_SPEC) vars[wgKey(f.key)] = String(d.world[f.key]);
          if (worldOn(chat)) {
            const ss = { ...chat.scriptstate ?? {}, ...vars };
            Object.assign(vars, worldVars(readWorld(chat), parseCatalog(lore), worldSettings(lore, ss), String(ss["$st_faction"] ?? "")));
          }

        env2.render();

      return true;
    }
    if (action === "set-status" && ["auto", "dark", "light", "paper", "off"].includes(v ?? "")) {
      await saveDefaults(env2, (p) => ({ ...p, status: v === "auto" ? void 0 : v }));
      repaintTheme();
      return true;
    }
    if (action === "set-nav" || action === "set-dim") {
      const n = Number(v);
      if (action === "set-nav" && (v === "left" || v === "right")) await saveDefaults(env2, (p) => ({ ...p, nav: v === "right" ? "right" : void 0 }));

        case "set-agent": {

          if (next !== d.basic.agentLimit) await basicPatch(env2, { agentLimit: next });
          return true;
        }
        case "set-theme":

        case "set-support":

          return true;
        case "set-asset-limit": {

          return true;
        }
        case "set-world": {
          const hit2 = nextWorld(d.world, v);
          if (hit2) await saveDefaults(env2, (p) => ({ ...p, world: { ...p.world, [hit2[0]]: hit2[1] } }));
          return true;
        }
      }
      return false;
    }
    switch (action) {
      case "set-world":
        await save(ctx2, env2, (chat) => {
          const hit2 = nextWorld(worldSettings(lore, chat.scriptstate), v);
          return hit2 ? { [wgKey(hit2[0])]: String(hit2[1]) } : null;
        });

        }
        if (v === "off-no") {

          return true;
        }
        if (v === "on") {

            const faction = String(chat.scriptstate?.["$st_faction"] ?? "");

          });

          const cur = basicSettings(lore, chat.scriptstate).agentLimit;
          const next = Math.min(AGENT_MAX, Math.max(0, cur + (v.endsWith("+") ? 1 : -1)));
          if (next === cur) return null;
          const vars = withBasic(ctx2, chat, { agentLimit: next });
          if (worldOn(chat)) {
            const ss = { ...chat.scriptstate ?? {}, ...vars };
            Object.assign(vars, worldVars(readWorld(chat), parseCatalog(lore), worldSettings(lore, ss), String(ss["$st_faction"] ?? "")));
          }
          return vars;
        });
        return true;
      case "set-theme":
        if (v !== "light" && v !== "dark" && v !== "paper") return true;
        await save(ctx2, env2, (chat) => chat.scriptstate?.[THEME_KEY] === v ? null : { [THEME_KEY]: v });
        return true;
      case "set-support":
        await save(ctx2, env2, (chat) => withBasic(ctx2, chat, { supportOn: v === "1" }));
        return true;
      case "set-asset":
        await save(ctx2, env2, (chat) => withBasic(ctx2, chat, { assetOn: v === "1" }));
        return true;
      case "set-asset-model":

        await save(ctx2, env2, (chat) => withBasic(ctx2, chat, { assetLimit: n }));
        return true;
      }
    }
    return false;
  }
