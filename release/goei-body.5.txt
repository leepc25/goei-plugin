  async function shrink(url, crop = null) {

      const im = new Image();
      im.src = url;
      await im.decode();
      const W3 = im.naturalWidth;
      const H2 = im.naturalHeight;
      let [sx, sy, sw, sh] = [0, 0, W3, H2];
      if (crop) {
        const side = Math.min(W3, H2, crop[2] / 100 * W3);
        sx = Math.min(W3 - side, Math.max(0, crop[0] / 100 * W3 - side / 2));
        sy = Math.min(H2 - side, Math.max(0, crop[1] / 100 * H2 - side / 2));
        sw = sh = side;
      }
      const scale = Math.min(1, (crop ? FACE_SIDE : MAX_SIDE) / Math.max(sw, sh));
      if (!crop && !(scale < 1)) return url;
      const canvas = document.createElement("canvas");
      canvas.width = Math.max(1, Math.round(sw * scale));
      canvas.height = Math.max(1, Math.round(sh * scale));
      const g2 = canvas.getContext("2d");
      if (!g2) return url;
      g2.imageSmoothingQuality = "high";
      g2.drawImage(im, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
      const blob = await new Promise((res) => canvas.toBlob(res, "image/webp", 0.88));
      return blob ? URL.createObjectURL(blob) : url;
    } catch {
      return url;
    }
  }
