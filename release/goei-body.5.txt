    [/가짜\s*관리국|fake/i, "B00"],

  function factionCode(org) {
    const t = org.trim();
    if (/^[A-Z]00$/.test(t)) return t;
    for (const [re, code2] of ORG_CODES) if (re.test(t)) return code2;
    return "";
  }
  function rankLevel(faction, rank3) {
    const r = rank3.replace(/\s+/g, "");
    if (!r || /^(-|없음|미정)$/.test(r)) return null;
    if (/신입|교육생|인턴|수습|trainee|rookie/i.test(r)) return 0;
    if (faction === "A00") {
      if (/부장|팀장|chief/i.test(r)) return 3;
      if (/선임|책임자|senior|lead/i.test(r)) return 2;
      if (/요원|연구원|agent|researcher/i.test(r)) return 1;
      return null;
    }
    if (faction === "B00" || faction === "C00") {
      const table = [[/사장|ceo/i, 9], [/전무/, 8], [/국장|이사|director/i, 7], [/부장/, 6], [/차장/, 5], [/과장|manager/i, 4], [/대리/, 3], [/주임/, 2], [/사원|직원|staff/i, 1]];
      for (const [re, n2] of table) if (re.test(r)) return n2;
      return null;
    }
    if (faction === "E00") {
      if (/특급/.test(r)) return 5;
      const m = /([1-5])\s*급/.exec(r);
      if (m) return 5 - Number(m[1]);
      if (/정예/.test(r)) return 4;
      if (/선임/.test(r)) return 3;
      if (/숙련/.test(r)) return 2;
      if (/일반/.test(r)) return 1;
      if (/임시/.test(r)) return 0;
    }
    return null;
  }
  function parseIdentity(text4, anchor, from = "status") {
    const t = String(text4 ?? "").trim();
    if (!t || t === "없음" || t === "-") return null;
    const [org, ...rest] = t.split(/[\/|·,]/).map((x) => x.trim());
    const rank3 = rest.join(" ").replace(/^-$/, "");
    const faction = factionCode(org);
    return { faction: faction || org, rank: rank3, level: rankLevel(faction, rank3), from, anchor };
  }

  function relativeRank(user, personFaction, personLevel) {
    if (!user || !user.faction || user.faction !== personFaction) return "outsider";
    const u = user.level ?? 0;
    if (personLevel === null) return "peer";
    return u > personLevel ? "up" : u < personLevel ? "down" : "peer";
  }
  function factionCodesIn(text4, extra = {}) {
    const out = [];
    let t = text4;
    for (const [re, code2] of ORG_CODES) {
      if (re.test(t)) {
        out.push(code2);
        t = t.replace(new RegExp(re.source, re.flags.includes("g") ? re.flags : re.flags + "g"), " ");
      }
    }
    for (const [code2, name] of Object.entries(extra)) if (!out.includes(code2) && name && t.includes(name)) out.push(code2);
    return out;
  }
