        if (!here.length) field2.target = placeTarget(v2);

      }
      case "map-unplace":

        field2.target = null;
        field2.confirmEnd = null;
        toStep("goal", env2);
        return true;
      case "map-filter":
        field2.filter = ["all", "an", "in", "ms"].includes(v) ? v : "all";

      case "pick-place":
        if (isPC() && sameTarget(field2.target, v)) {

          env2.render();
          return true;

        toStep("team", env2);

      case "field-step":

        return true;
      case "pick-event": {
        const x = readCtx(ctx2);
        if (typeof x === "string") return true;
        const ev = activeEvents(x.world).find((e) => e.key === v);
        if (!ev) return true;

      case "end-event":

      case "team-toggle": {

        const i = field2.team.indexOf(v);
        if (i >= 0) field2.team.splice(i, 1);

        env2.render();
        return true;

        env2.render();
        return true;
      }
      case "depart":
        await depart(ctx2, env2);
        return true;
      case "mission-kind":
        if (v === "an" || v === "ms" || v === "lm") {
          field2.mkind = v;
          field2.query = "";
        }
        env2.render();
        return true;

        toStep("goal", env2);
        return true;
      case "mission-send":
