  var liveEvidence = (c, live) => c.evidence.filter((e) => !e.void && isAlive({ anchor: e.anchor, quote: e.quote, src: "story" }, live));
  function candidateStats(c, live) {
    const ev = liveEvidence(c, live);
    const n2 = ev.length;
    const scenes = new Set(ev.map((e) => e.anchor)).size;
    const confidence = n2 ? Math.round(ev.reduce((s8, e) => s8 + e.confidence, 0) / n2) : 0;
    const strength = n2 ? Math.round(ev.reduce((s8, e) => s8 + e.strength, 0) / n2) : 0;
    const sign = c.direction === "down" || c.direction === "weaken" ? -1 : 1;
    const delta = c.axis === "pattern" || isSexualAxis(c.axis) ? 0 : sign * clamp2(Math.round(strength / 15), 2, 8);
    const need = isSexualAxis(c.axis) ? SEXUAL_READY : CANDIDATE_READY;
    const ready = n2 >= need.evidence && scenes >= need.scenes && confidence >= need.confidence;
    return { n: n2, scenes, confidence, strength, delta, ready };
  }
  function approveCandidate(p2, id2, live, now) {
    const c = p2.candidates[id2];
    if (!c || c.status !== "collecting") return "승인할 수 있는 후보가 아니에요";
    const st = candidateStats(c, live);
    if (!st.ready) return "아직 관찰이 모자라요";
    if (isSexualAxis(c.axis)) {
      const sx = ensureSexual(p2);
      const f = SEXUAL_FIELD_OF[c.axis];
      c.applied = { sexual: c.axis, before: sx[f].text, after: c.pattern };
      sx[f] = { text: c.pattern, at: now };
    } else if (c.axis === "pattern") {
      if (c.direction === "weaken") {
        const had = p2.patterns.includes(c.pattern);
        p2.patterns = p2.patterns.filter((t) => t !== c.pattern);
        c.applied = { pattern: c.pattern, removed: had };
      } else {
        if (!p2.patterns.includes(c.pattern)) p2.patterns = [...p2.patterns, c.pattern].slice(-30);
        c.applied = { pattern: c.pattern };
      }
    } else {
      const before = p2.axes[c.axis];
      const after = axisValue(before + st.delta);
      c.applied = { axis: c.axis, before, after };
      setAxes(p2, { ...p2.axes, [c.axis]: after }, "manual");
    }
    c.status = "approved";
    c.at = now;
    return null;
  }
  function rejectCandidate(p2, id2, now) {
    const c = p2.candidates[id2];
    if (!c || c.status !== "collecting") return "거절할 수 있는 후보가 아니에요";
    c.status = "rejected";
    c.at = now;
    return null;
  }
  function revertCandidate(p2, id2, now) {
    const c = p2.candidates[id2];
    if (!c || c.status !== "approved" || !c.applied) return "되돌릴 수 있는 후보가 아니에요";
    const ap = c.applied;
    if ("sexual" in ap) {
      const sx = ensureSexual(p2);
      sx[SEXUAL_FIELD_OF[ap.sexual]] = { text: ap.before, at: now, manual: true };
    } else if ("axis" in ap) setAxes(p2, { ...p2.axes, [ap.axis]: ap.before }, "manual");
    else if (ap.removed) p2.patterns = [...p2.patterns, ap.pattern].slice(-30);
    else p2.patterns = p2.patterns.filter((t) => t !== ap.pattern);
    c.status = "reverted";
    c.at = now;
    return null;
  }
  function pruneCandidates(p2, live) {
    ensureAxes(p2);
    let n2 = 0;
    for (const [id2, c] of Object.entries(p2.candidates)) {
      if (c.status === "collecting" && !liveEvidence(c, live).length) {
        delete p2.candidates[id2];
        n2++;
      }
    }
    return n2;
  }

  function parseSheet(text4) {
    const lines3 = String(text4 ?? "").replace(/\r/g, "").split("\n");
    const out = { title: "", sections: {} };
    let cur = null;
    for (const line2 of lines3) {
      const h3 = /^###\s+(.+?)\s*$/.exec(line2);
      const h4 = /^####\s+(.+?)\s*$/.exec(line2);
      if (h3 && !h4) {
        out.title = h3[1];
        cur = null;
        continue;
      }
      if (h4) {
        cur = h4[1];
        out.sections[cur] = "";
        continue;
      }
      if (cur !== null) out.sections[cur] += (out.sections[cur] ? "\n" : "") + line2;
    }
    for (const k of Object.keys(out.sections)) out.sections[k] = out.sections[k].trim();
    return out;
  }
  function bullets(section3) {
    const out = [];
    let label2 = "";
    for (const raw of section3.split("\n")) {
      const line2 = raw.trim();
      if (!line2) continue;
      const tag = /^\[([^\]]+)\]$/.exec(line2);
      if (tag) {
        label2 = tag[1];
        continue;
      }
      const b = /^[-*•]\s+(.+)$/.exec(line2);
      const text4 = b ? b[1] : line2;
      out.push(label2 ? `[${label2}] ${text4}` : text4);
    }
    return out;
  }
