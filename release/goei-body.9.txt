  async function mallSettingsAction(action, value, ctx2, env3) {
    if (await deskSettingsAction(action, value, ctx2, env3)) return true;
    if (await cosExtraAction(action, value, ctx2, env3)) return true;
    if (action !== "ms-cfg" && action !== "ms-step" && action !== "ms-text" && action !== "ms-num") return false;
    let site = "";
    let k = "";
    let v = "";
    if (action === "ms-cfg" || action === "ms-text") [, site, k, v] = /^(\w+):(\w+)=([\s\S]*)$/.exec(value ?? "") ?? [];
    if (action === "ms-text") v = String(v ?? "").replace(/\s+/g, " ").trim().slice(0, 80);
    else if (action === "ms-num") {
      [, site, k, v] = /^(\w+):(\w+)=(.*)$/.exec(value ?? "") ?? [];
      const n3 = Math.floor(Number(v));
      const r = MS_RANGE[k];
      if (!r || !Number.isFinite(n3) || String(v).trim() === "") return env3.render(), true;
      v = String(Math.max(r[0], Math.min(r[1], n3)));
    } else if (action === "ms-step") {
      const m = /^(\w+):(\w+):([+-])$/.exec(value ?? "");
      if (!m) return true;
      [, site, k] = m;
      const s8 = mallSettings(ctx2.chat.scriptstate, site);
      const cur = k === "min" ? s8.size[0] : k === "max" ? s8.size[1] : k === "turns" ? s8.turns : s8.refs;
      const [lo, hi] = MS_RANGE[k] ?? [0, 0];
      v = String(Math.max(lo, Math.min(hi, cur + (m[3] === "+" ? 1 : -1))));
    }
    if (site !== "cos" && site !== "bm") return true;
    const varName = k === "auto" ? mallAutoKey(site) : MS_KEY[k] ? mallKey(MS_KEY[k], site) : "";
    if (!varName) return true;
    const extra2 = {};
    if ((action === "ms-step" || action === "ms-num") && (k === "min" || k === "max")) {
      const s8 = mallSettings(ctx2.chat.scriptstate, site);
      const n3 = Number(v);
      if (k === "min" && n3 > s8.size[1]) extra2[mallKey("$sh_max", site)] = v;
      if (k === "max" && n3 < s8.size[0]) extra2[mallKey("$sh_min", site)] = v;
    }
    await updateChat(ctx2.charIndex, ctx2.chatIndex, (chat) => String(chat.scriptstate?.[varName] ?? "") === v && !Object.keys(extra2).length ? null : { ...chat, scriptstate: { ...chat.scriptstate ?? {}, [varName]: v, ...extra2 } });
    await env3.refresh();
    return true;
  }
  async function deskSettingsAction(action, value, ctx2, env3) {
    const set2 = async (vars) => {
      await updateChat(ctx2.charIndex, ctx2.chatIndex, (chat) => ({ ...chat, scriptstate: { ...chat.scriptstate ?? {}, ...vars } }));
      await env3.refresh();
      return true;
    };
    if (action === "ms-cfg") {
      const m = /^bm:(desk_on|desk_again)=([01])$/.exec(value ?? "");
      if (!m) return false;
      return m[1] === "desk_on" ? set2({ [DESK_KEY.on]: m[2] === "0" ? "0" : "" }) : set2({ [DESK_KEY.again]: m[2] === "1" ? "1" : "" });
    }
    if (action === "ms-desk") {
      const m = /^axes=([\s\S]*)$/.exec(value ?? "");
      return m ? set2({ [DESK_KEY.axes]: m[1].trim().slice(0, 4e3) }) : true;
    }
    if (action === "ms-desk-step") {
      const d = deskSettings(ctx2.chat.scriptstate);
      const n3 = Math.max(DESK_AVOID_RANGE[0], Math.min(DESK_AVOID_RANGE[1], d.avoid + (value === "+" ? 1 : -1)));
      return set2({ [DESK_KEY.avoid]: String(n3) });
    }
    if (action === "ms-desk-reset") return set2({ [DESK_KEY.axes]: "" });
