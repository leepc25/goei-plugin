  var STATUS_HEAD = "[Previous status]";
  var NOTE = "This is the status line of the scene right before the next input; older status lines were removed from the history. Continue from it: advance the time, update the other fields, and output exactly one new [▣] line at the very end of your response.";
  function lastStatusLine(messages) {
    for (let i = messages.length - 1; i >= 0; i--) {
      const m = messages[i];
      if (m?.role !== "assistant" || typeof m.content !== "string") continue;
      const tags = findStatusTags(m.content);
      if (tags.length) return tags[tags.length - 1].raw.trim();
    }
    return null;
  }
  function stripStatusLines(messages) {
    return messages.map((m) => {
      if (m?.role !== "assistant" || typeof m.content !== "string") return m;
      const tags = findStatusTags(m.content);
      if (!tags.length) return m;
      let c = m.content;
      for (const t of tags) c = c.replace(t.raw, "");
      return { ...m, content: c.replace(/[ \t]+\n/g, "\n").replace(/\n{3,}/g, "\n\n").trim() };
    });
  }
  function statusBlock(line2, cols) {
    const fields3 = cols?.length ? `Fields: ${cols.map((c) => colDef(c).label).join(" | ")}
` : "";
    return `${STATUS_HEAD}
${line2}
${fields3}${NOTE}`;
  }
  function rebuildStatusLine(line2, cols, fill3 = {}) {
    const t = findStatusTags(line2)[0];
    if (!t || !t.date) return line2;
    const values = {
      date: t.date,
      time: t.time,
      weather: t.weather,
      temp: t.temp,
      place: t.place,
      target: t.target,
      companions: t.companions.join(","),
      deployed: t.deployed ? "투입" : "",
      identity: t.identity || fill3.identity || "",
      wallet: fill3.wallet !== void 0 && fill3.wallet !== null ? walletCell(fill3.wallet) : t.wallet
    };
    return formatStatusLine(values, cols);
  }
  function withIdentity(line2, identity) {
    if (!identity) return line2;
    const tags = findStatusTags(line2);
    const t = tags[0];
    if (!t || t.identity) return line2;
    const cols = line2.replace(/^[[【［]\s*▣\s*[|｜]/, "").replace(/[\]】］]\s*$/, "").split(/[|｜]/).length;
    if (cols !== 8) return line2;
    return line2.replace(/[\]】］]\s*$/, `|${identity}]`);
  }
  function withStatusBlock(messages, identity = "", shape) {

  var squash4 = (s5) => s5.replace(/\s+/g, " ").trim();
  function stripSheets(messages, sheets) {
    if (!sheets.length) return messages;
    const targets2 = sheets.map(squash4).filter(Boolean);
    return messages.map((m) => {
      if (typeof m?.content !== "string") return m;
      let text = m.content;
      let changed = false;
      for (const t of targets2) {
        const i = text.indexOf(t);
        if (i >= 0) {
          text = text.slice(0, i) + text.slice(i + t.length);
          changed = true;
          continue;
        }
        const flat = squash4(text);
        const j = flat.indexOf(t);
        if (j >= 0 && flat === squash4(text)) {
          const lines3 = t.split(" ");
          const first = lines3[0];
          const start3 = text.indexOf(first);
          if (start3 >= 0) {
            const end = text.indexOf(lines3[lines3.length - 1], start3);
            if (end > start3) {
              text = text.slice(0, start3) + text.slice(end + lines3[lines3.length - 1].length);
              changed = true;
            }
          }
        }
      }
      return changed ? { ...m, content: text.replace(/\n{3,}/g, "\n\n") } : m;
    });
  }
