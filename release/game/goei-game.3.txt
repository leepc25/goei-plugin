  var STATUS = {
    bleed: { label: "출혈", text: "출혈 · 피가 멎지 않는다", max: 3, sec: 8 },
    taint: { label: "오염", text: "오염 · 숨이 탁해진다", max: 3, sec: Infinity },
    headache: { label: "두통", text: "두통 · 머리가 울린다", max: 1, sec: 10 },
    confuse: { label: "혼란", text: "혼란 · 방향이 뒤집힌다", max: 1, sec: 4 },
    fear: { label: "공포", text: "공포 · 몸이 굳는다", max: 1, sec: 5 },
    slow: { label: "둔화", text: "느려졌다", max: 1, sec: 5 },
    silence: { label: "침묵", text: "목소리가 안 나온다", max: 1, sec: 6 }
  };
  var STATUS_NUM = {
    /** 출혈: 초당 최대 체력 비율 (겹침마다), 달리면 배 */
    bleedPerSec: 0.01,
    /** 출혈: 막기를 이만큼 이어 하면 멎는다 (초) */
    bleedBlockStop: 3,
    /** 오염: 단계마다 최대 체력 감소 */
    taintPerStack: 0.2,
    headacheSp: 0.5,
    /** 두통 걸린 AI 적의 공격 간격 배수 (10-30 결정) */
    headacheGap: 1.25,
    fearSpeed: 0.75,
    slowSpeed: 0.6,
    maxDebuffs: 3
  };
  var StatusSet = class {
    list = [];
    blockT = 0;
    /** 걸리지 않는 상태 (퍽 「정신 방벽」: 공포·혼란) */
    immune;
    /** 오염 단계마다 줄어드는 최대 체력 (퍽 「내성」으로 줄어듦) */
    taintPer;
    has(k) {
      return this.list.some((s) => s.kind === k);
    }
    stacks(k) {
      return this.list.find((s) => s.kind === k)?.stacks ?? 0;
    }
    /** 걸기. sec 생략이면 기본 시간. 새로 걸렸으면 true (글귀를 띄울지) */
    apply(kind, sec, mul) {
      if (this.immune?.has(kind)) return false;
      const def = STATUS[kind];
      const dur = def.sec === Infinity ? Infinity : Math.max(0.5, sec ?? def.sec);
      const cur = this.list.find((s) => s.kind === kind);
      if (cur) {
        cur.stacks = Math.min(def.max, cur.stacks + (def.max > 1 ? 1 : 0));
        cur.t = Math.max(cur.t, dur);
        if (mul) cur.mul = Math.max(cur.mul ?? 1, mul);
        return false;
      }
      if (this.list.length >= STATUS_NUM.maxDebuffs) {
        const out = [...this.list].sort((a, b) => a.t - b.t)[0];
        this.list = this.list.filter((s) => s !== out);
      }
      this.list.push({ kind, stacks: 1, t: dur, ...mul ? { mul } : {} });
      return true;
    }
    /** 해제. 오염은 한 단계씩(정화제), 나머지는 전부. 풀린 게 있으면 true */
    cure(kind, stages = 1) {
      const cur = this.list.find((s) => s.kind === kind);
      if (!cur) return false;
      if (kind === "taint" && cur.stacks > stages) cur.stacks -= stages;
      else this.list = this.list.filter((s) => s !== cur);
      return true;
    }
    clear() {
      this.list = [];
    }
    /** 시간 흐름. 이 동안의 체력 피해(최대 체력 기준)를 돌려준다 */
    step(dt, o) {
      let dmg = 0;
      const bleed = this.stacks("bleed");
      if (bleed) dmg += o.maxHp * STATUS_NUM.bleedPerSec * bleed * (o.running ? 2 : 1) * dt * (this.list.find((s) => s.kind === "bleed")?.mul ?? 1);
      this.blockT = o.blocking ? this.blockT + dt : 0;
      if (bleed && this.blockT >= STATUS_NUM.bleedBlockStop) this.cure("bleed");
      for (const s of this.list) s.t -= dt;
      this.list = this.list.filter((s) => s.t > 0);
      return dmg;
    }
    /** 맞으면 혼란이 풀린다 */
    onHit() {
      this.cure("confuse");
    }
    speedMul() {
      return (this.has("slow") ? STATUS_NUM.slowSpeed : 1) * (this.has("fear") ? STATUS_NUM.fearSpeed : 1);
    }
    spMul() {
      return this.has("headache") ? STATUS_NUM.headacheSp : 1;
    }
    /** AI 두뇌의 공격 간격 배수: 두통이면 느려진다 */
    attackGap() {
      return this.has("headache") ? STATUS_NUM.headacheGap : 1;
    }
    canBlock() {
      return !this.has("fear");
    }
    canSkill() {
      return !this.has("silence");
    }
    flipped() {
      return this.has("confuse");
    }
    /** 오염으로 줄어든 최대 체력 비율 (1 = 그대로) */
    maxHpRatio() {
      return Math.max(0.2, 1 - (this.taintPer ?? STATUS_NUM.taintPerStack) * this.stacks("taint"));
    }
    /** 화면 표시용 */
    view() {
      return this.list.map((s) => ({ kind: s.kind, label: STATUS[s.kind].label, stacks: s.stacks, t: s.t === Infinity ? null : s.t }));
    }
  };

    rerollsLeft() {

      };
      put2("연속 처치", "chain", 5);
      put2("무피격", "clean", 10);
      put2("사냥 기록", "hunt", 60);
      put2("피의 저금통", "bank", 8);
      put2("아슬아슬", "close", 10);
      put2("콤보 고조", "combo", 50);
      put2("쌓인 울분", "rage", 5);
      put2("현장 경험", "rooms", 10);
      put2("탄피 모으기", "shells", 20);
      put2("되갚기", "payback", 5);

    /* ───────── 대상별 기억 ───────── */

    /* ───────── 맞히기 전: 피해 보너스·치명·반응 바꾸기 ───────── */
    pre(c) {
      const h = this.host;
      if (!h) return;
      const s = this.st(c.body);
      const melee = MELEE.has(c.src), shot = c.src === "shot", tech = TECH.has(c.src);
      const hpR = c.body.hpMax ? c.body.hp / c.body.hpMax : 1;
      const faceDir2 = c.body.dir ?? c.body.core?.P?.face ?? 0;
      let back = faceDir2 !== 0 && Math.sign(c.t.x - h.P.x) === faceDir2;
      if (this.has("겁먹은 등") && s?.has("fear") || this.has("혼란 틈") && s?.has("confuse") || this.has("사냥꾼의 눈") && c.t.mark > 0) back = true;
      let add = 0;
      const R = (n) => this.r(n);
      if (c.src === "light") add += 0.08 * R("단련된 주먹") + (c.last ? 0.08 * R("단련된 주먹") : 0);
      if (c.src === "slide") add += 0.2 * R("미끄럼 숙련");
      if (c.src === "heavy") add += 0.12 * R("무거운 한 방");
      if (melee && c.t.z > 4) add += 0.15 * R("공중 사냥");
      if (melee && hpR <= 0.2) add += 0.25 * R("마무리 일격");
      if (shot) add += 0.08 * R("정조준") + (c.thrown ? 0.12 * R("투척 강화") : 0);
      if (shot && R("거리 감각")) {
        const d3 = Math.abs(c.t.x - h.P.x);
        if (d3 < 200 || d3 > 640) add += 0.12 * R("거리 감각");
      }
      if (back) add += 0.15 * R("그림자 습격");
      if (c.hpBefore != null && c.hpBefore >= (c.body.hpMax ?? 1)) add += 0.2 * R("첫 일격");
      if (this.has("배수진") && h.me.hp <= h.me.hpMax * 0.3) add += 0.25;
      if (h.isElite(c.body)) add += 0.25 * R("정예 사냥");
      if (this.has("일대일") && h.foes().length === 1) add += 0.2;
      if (h.isDown(c.body)) add += 0.2 * R("앉게");
      if (R("약점 찌르기") && this.ailCount(c) > 0) add += 0.1 * R("약점 찌르기");
      if (this.has("겁먹은 등") && s?.has("fear")) add += 0.15;
      if (c.src === "skill") add += 0.12 * R("고유기 강화") + (this.has("피의 저금통") ? 0.05 * Math.min(8, this.n.bank ?? 0) : 0);
      if (c.src === "charge") add += 0.12 * R("충전 강화");
      if (c.src === "super") add += 0.15 * R("필살 강화");
      if (c.src === "heavy" && this.has("쌓인 울분")) add += 0.12 * Math.min(5, this.n.rage ?? 0);
      if (this.has("연속 처치")) add += 0.05 * Math.min(5, this.n.chain ?? 0);
      if (this.has("무피격")) add += 0.02 * Math.min(10, this.n.clean ?? 0);
      if (this.has("현장 경험")) add += 0.015 * Math.min(10, this.n.rooms ?? 0);
      if (melee && this.has("되갚기")) add += 0.03 * Math.min(5, this.n.payback ?? 0);
      if (melee && this.has("일부러 비좁게")) {
        const [b0, b1] = h.bounds();
        const wd = Math.min(h.P.x - b0, b1 - h.P.x);
        add += wd < 150 ? 0.3 : wd > 300 ? -0.1 : 0;
      }
      if (this.traits.has("무모함")) add += 0.2;
      if (this.traits.has("유리 대포")) add += 0.35;
      if (this.traits.has("한 방 주의")) add += c.src === "heavy" || c.src === "charge" ? 0.4 : c.src === "light" ? -0.3 : 0;
      if (this.traits.has("광전사") && h.me.hp <= h.me.hpMax * 0.5) add += 0.3;
      if (this.traits.has("되감는 시계") && c.src === "skill") add -= 0.2;
      if (this.traits.has("피의 대가") && tech) add += 0.35;
      if (this.traits.has("겁쟁이") && melee) add -= 0.15;
      if (this.traits.has("총 버림") && melee) add += 0.3;
      if (this.traits.has("난사") && shot) add -= 0.2;
      let dmg = c.dmg * Math.max(0.1, 1 + add);
      if (this.has("혼돈 변수") && tech) dmg *= 0.7 + Math.random() * 0.8;
      if (c.src === "light" && this.has("되받아치기") && this.T - this.blockAt < 0.5) {
        dmg *= 2.2;
        c.kind = "heavy";
        this.blockAt = -99;
      }
      if (c.src === "slide" && this.has("발목 걸기")) c.kind = "down";
      if (shot && this.has("자세 무너뜨리기") && (c.t.stunT > 0 || c.body.state === "hurt")) c.kind = "down";
      if (c.src === "heavy" && this.has("두통 울림") && s?.has("headache") && this.ready(c.body, "ring", 6)) {
        c.kind = "stun";
        h.later(0, () => h.stun(c.t, 0.8));
      }
      if (c.src === "heavy" && this.has("입막음 처형") && s?.has("silence") && hpR <= 0.25) {
        if (h.isElite(c.body)) dmg *= 1.5;
        else c.kill = true;
      }
      c.critAdd += this.critBonus() + (shot ? 0.05 * R("급소 겨냥") : 0) + (shot && this.has("묶인 과녁") && c.t.bindT > 0 ? 0.4 : 0);
      c.critDmgAdd += this.critDmgAdd();
      if (this.has("기절 추격") && h.isStunned(c.t) && this.ready(c.body, "stunCrit", 1.5)) c.forceCrit = true;
      if (shot && this.has("틈새 사격") && this.T - this.dodgeEndAt < 1 && !this.n.gapUsed) {
        c.forceCrit = true;
        this.n.gapUsed = 1;
      }
      if (this.traits.has("도박꾼")) dmg *= 0.9;
      c.dmg = dmg;
    }
    /* ───────── 맞힌 뒤: 상태 걸기·터뜨리기·계수 ───────── */
    post(c) {
      const h = this.host;
      if (!h || c.src === "perk") {
        if (c.killed) this.kill(c);
        return;
      }
      if (c.guarded) return;
      const s = this.st(c.body);
      const put2 = (k, sec, mul) => h.status(c.body, k, sec * this.outDur(k), mul);
      const melee = MELEE.has(c.src);
      if (melee) {
        this.n.combo = (this.n.combo ?? 0) + 1;
        this.killWindow = 0;
      }
      if (c.src === "light") {
        this.comboHits[c.step] = true;
        if (c.last) {
          if (this.has("두통 유발")) put2("headache", 4);
          if (this.traits.has("저주받은 손")) {
            const ks = ["bleed", "taint", "headache", "confuse", "fear", "slow", "silence"];
            put2(ks[Math.floor(Math.random() * ks.length)], 3);
          }
          if (this.has("오염 손톱")) {
            const m = this.mem(c.body);
            m.claw = (m.claw ?? 0) + 1;
            if (m.claw === 3) put2("taint", 1);
          }
          if (this.has("따라 치는 그림자") && this.comboHits[1] && this.comboHits[2]) h.later(0.4, () => h.ghost(["9", "10", "11"], 0.3));
          if (this.has("그림자 분대") && this.comboHits[1] && this.comboHits[2]) this.n.squadUntil = this.T + 3.4;
          if (this.has("충격 주먹")) this.shockFist(c);
          this.comboHits = {};
        }
      }
      if (c.src === "slide" && this.has("발목 걸기")) put2("slow", 2);
      if (c.src === "heavy") {
        if (this.has("출혈 일격") && this.ready(c.body, "bleedHit", 2)) put2("bleed", 6, this.bleedMul());
        const faceDir2 = c.body.dir ?? c.body.core?.P?.face ?? 0;
        if (this.has("족쇄 차기") && faceDir2 && Math.sign(c.t.x - h.P.x) === faceDir2 && this.ready(c.body, "shackle", 5)) h.bind(c.t, 1.5 * this.outDur("slow"));
        if (this.has("쌓인 울분")) this.n.rage = 0;
      }
      if (c.src === "shot") {
        this.shotHits++;
        if (this.has("표식탄") && this.shotHits % 4 === 0) h.mark(c.t, 3 + (this.has("사냥꾼의 눈") ? 2 : 0));
        if (this.has("겁주기") && Math.random() < 0.2 && this.ready(c.body, "scare", 6)) put2("fear", 2);
        if (this.has("폭발탄") && this.shotHits % 5 === 0) h.boom({ x: c.t.x, gy: c.t.y, r: 150, dmg: 66, kind: "heavy", tag: "boom", depth: 0, style: "blast" });
      }
      if (c.src === "charge" && c.maxLv && this.has("혼란 충전")) put2("confuse", 2);
      if (c.src === "skill") {
        if (this.has("둔화 고유기")) put2("slow", 2);
        if (this.has("진조의 피")) h.me.hp = Math.min(h.me.hpMax, h.me.hp + c.dmg * 0.08 * this.r("진조의 피"));
        if (this.has("피의 저금통") && (this.n.bank ?? 0) > 0 && !this.n.bankSpend) {
          this.n.bankSpend = 1;
          h.later(1, () => {
            this.n.bank = 0;
            this.n.bankSpend = 0;
          });
        }
      }
      if (c.src === "super" && this.has("침묵의 필살")) put2("silence", 4);
      if (s?.has("bleed") && this.has("피 터뜨리기") && (c.src === "heavy" || c.src === "charge" && c.maxLv)) this.burstBleed(c, s);
      if (s?.has("slow") && this.has("얼어붙은 발") && ["launch", "down", "blow", "spike"].includes(c.kind)) {
        h.hit(c.t, (c.body.hpMax ?? 1e3) * 0.06, "light", "frozen");
        this.wakeLate(c.body, 0.5);
      }
      if (s?.stacks("taint") && c.src === "super" && this.has("오염 연소") && this.ready(c.body, "burn", 1.5)) {
        const n = s.stacks("taint");
        h.hit(c.t, (c.body.hpMax ?? 1e3) * 0.1 * n, "heavy", "burn");
        s.cure("taint");
      }
      if (this.has("바닥 차갑지?") && ["launch", "down", "blow", "spike"].includes(c.kind)) this.wakeLate(c.body, 0.6);
      if (this.count("공중 사냥") >= 2) c.body.maxLaunch = 3;
      if (this.has("벽꽝 강화")) c.body.wallDmg = 0.15;
      if (this.has("보이지 않는 사슬") && c.t.bindT > 0) {
        for (const q of h.foes()) if (q !== c.t && q.bindT > 0) h.hit(q, c.dmg * 0.25, "light", "chain");
      }
      if (c.crit) {
        if (this.has("급소 간파")) h.mark(c.t, 3 + (this.has("사냥꾼의 눈") ? 2 : 0));
        if (this.has("표식 전염") && c.t.mark > 0) {
          const q = this.around(c.t, 300)[0];
          if (q) h.mark(q, 3);
        }
      }
      if (this.has("혼란 틈") && s?.has("confuse") && this.ready(c.body, "confTip", 4)) put2("slow", 1);
      if (c.killed) this.kill(c);

    wakeLate(body, sec) {
      if (body.mem) body.wakeExtra = (body.wakeExtra ?? 0) + Math.round(sec * 30);
      else if (body.core?.P) body.core.P.wakeExtra = (body.core.P.wakeExtra ?? 0) + sec;
    }
    /** 피 터뜨리기: 남은 출혈 피해를 한꺼번에 ×1.3, 출혈 풀림. 피바다면 둘레 2m에 출혈이 튐 */
    burstBleed(c, s) {
      const h = this.host;
      const b = s.list.find((x) => x.kind === "bleed");
      if (!b) return;
      const left = (c.body.hpMax ?? 1e3) * STATUS_NUM.bleedPerSec * b.stacks * Math.min(b.t, 12) * (b.mul ?? 1) * 1.3;
      s.cure("bleed");
      h.hit(c.t, left, "heavy", "burst");
      if (this.has("피바다")) for (const q of this.around(c.t, 200)) h.status(q.body, "bleed", 6 * this.outDur("bleed"), this.bleedMul());
    }
    /** 충격 주먹: 3타 끝에 앞으로 충격파(관통, 3타 피해 60% + 움찔) */
    shockFist(c) {
      const h = this.host;
      const dir = h.P.face, x0 = h.P.x;
      for (const q of h.foes()) {
        const dx = (q.x - x0) * dir;
        if (dx > 0 && dx < 420 && Math.abs(q.y - h.P.y) < 70 && q !== c.t) h.hit(q, c.dmg * 0.6, "light", "fist");
      }
      h.boom({ x: x0 + dir * 200, gy: h.P.y, r: 0, dmg: 0, kind: "light", tag: "fist", style: "shock" });
    }
    /** 처치 */
    kill(c) {
      const h = this.host;
      if (!h) return;
      this.lastKill = { x: c.t.x, y: c.t.y };
      const s = this.st(c.body);
      const elite = h.isElite(c.body);
      this.n.chain = Math.min(5, (this.n.chain ?? 0) + 1);
      this.lastKillT = this.T;
      this.n.hunt = (this.n.hunt ?? 0) + 1;
      if (this.n.hunt % 10 === 0) this.applyHpMax();
      if (this.has("피의 저금통")) this.n.bank = Math.min(8, (this.n.bank ?? 0) + 1);
      if (this.traits.has("피에 굶주림")) h.me.hp = Math.min(h.me.hpMax, h.me.hp + h.me.hpMax * 0.03);
      const isBoom = c.tag === "corpse" || c.tag === "boom" || c.tag === "dodgebomb" || c.tag === "heat";
      const depth = c.depth ?? 0;
      if (this.has("시체 폭발") && (!isBoom || this.has("연쇄 폭파") && depth < 3)) {
        const mx = c.body.hpMax ?? 1e3, x = c.t.x, gy = c.t.y;
        h.later(0.4, () => h.boom({ x, gy, r: 200, dmg: mx * (elite ? 0.06 : 0.12), kind: "heavy", tag: "corpse", depth: depth + 1, style: "blast", skip: c.body }));
      }
      if (this.has("위협")) {
        for (const q of this.around(c.t, 300)) if (!h.isElite(q.body)) h.status(q.body, "fear", 1.5 * this.outDur("fear"));
      }
      if (s?.stacks("taint") && this.has("오염 전파")) for (const q of this.around(c.t, this.has("역병") ? 400 : 200)) {
        h.status(q.body, "taint", 1);
        if (this.has("역병")) h.status(q.body, "slow", 2 * this.outDur("slow"));
      }
      if (s?.has("bleed") && this.has("피바다")) for (const q of this.around(c.t, 200)) h.status(q.body, "bleed", 6 * this.outDur("bleed"), this.bleedMul());
      if (s && s.list.length >= 2 && this.has("상처 옮기기")) {
        const q = this.around(c.t, 250)[0];
        if (q) for (const a of s.list) h.status(q.body, a.kind, Math.max(0.5, (a.t === Infinity ? 6 : a.t) / 2));
      }
    }
    /* ───────── 회피 ───────── */
    dodge() {
      const h = this.host;
      if (!h) return;
      this.dodgeAt = this.T;
      this.dodgeHit.clear();
      this.n.gapUsed = 0;
      this.dodgeSide.clear();
      for (const q of h.foes()) this.dodgeSide.set(q.body, Math.sign(q.x - h.P.x));
      if (this.has("회피 폭탄")) {
        const x = h.P.x, gy = h.P.y;
        h.later(0.6, () => h.boom({ x, gy, r: 150, dmg: 60, kind: "heavy", tag: "dodgebomb", style: "blast" }));
      }
      if (this.has("미끼 잔상")) {
        perkDecoy.x = h.P.x;
        perkDecoy.y = h.P.y;
        perkDecoy.until = performance.now() + 1500;
      }
    }
    /** 회피 중 매 장: 몸을 통과한 적 */
    dodgeTick() {
      const h = this.host;
      if (!h || !this.has("꿰뚫는 회피")) return;
      const vx = h.P.dodgeV?.x ?? 0;
      for (const q of h.foes()) if (!this.dodgeHit.has(q.body) && Math.abs(q.y - h.P.y) < 50 && (Math.abs(q.x - h.P.x) < 120 && Math.sign(q.x - h.P.x) === Math.sign(vx) || this.dodgeSide.has(q.body) && Math.sign(q.x - h.P.x) !== this.dodgeSide.get(q.body))) {
        this.dodgeHit.add(q.body);
        h.hit(q, 60, "light", "pierce");
        h.status(q.body, "slow", 1.5 * this.outDur("slow"));
      }
    }
    dodgeEnd() {
      this.dodgeEndAt = this.T;
    }
    /** 회피 무적으로 피함. 시작 0.15초 안이면 아슬아슬 */
    evaded() {
      if (this.T - this.dodgeAt < 0.15 && this.has("아슬아슬")) this.n.close = Math.min(10, (this.n.close ?? 0) + 1);
    }
    /** 회피 무적 중 투사체 → 되돌림(빈틈 없음) */
    reflectOnDodge() {
      return this.has("빈틈 없음");
    }
    /* ───────── 막기 ───────── */
    blockPressed() {
      this.blockPress = this.T;
    }
    /** 막기 성공. melee = 근접(때린 적 attacker), dmg = 막은 공격 피해. 돌려줄 투사체면 true */
    blocked(o) {
      const h = this.host;

      const p = at2 ?? this.perks.lastKill ?? { x: this.x + this.dir * 160, y: this.y };
      const [b0, b1] = this.world?.bounds ?? [0, 99999];
      this.reward = { x: Math.max(b0 + 60, Math.min(b1 - 60, p.x)), y: p.y, t: 0, offers, color: pair ? PAIR_COLOR : offers.every((f) => f.type === "trait") ? TRAIT_COLOR : GRADE_COLOR[top] };
      fighterHooks.sfx({ type: "pickup", x: p.x });

      const st = body.status;
      if (!(st instanceof StatusSet) || !body.alive || body.hp <= 0) return;
      if (st.apply(k, k === "taint" ? void 0 : s * (body.stat?.statusDur ?? 1), mul)) this.core.perkApi.text(body.x, body.y - (body.z || 0) - 250, STATUS[k].label);

    /** 기절: 엔진 적은 stunned, 인물은 스턴 동작 */
    stunBody(body, s) {
      if (!body.alive || body.hp <= 0) return;
      if (body.core?.P) {
        if (!["down", "belly", "dead"].includes(body.core.P.mode)) {
          body.core.stun();
          body.core.P.stunT = s;
        }
        return;
      }
      if (typeof body.set === "function" && body.state !== "dead" && body.state !== "spawn") {
        body.move = null;
        body.set("stunned");
        body.stunT = Math.round(s * 30);
      }
    }
    /** 묶기: 엔진 적은 bound(못 움직임), 인물은 스턴 */

      if (!body.alive || body.hp <= 0) return;
      if (body.core?.P) return this.stunBody(body, s);
      if (typeof body.set === "function" && body.state !== "dead" && body.state !== "spawn" && body.state !== "air") {
        body.move = null;
        body.set("bound");
        body.bindT = Math.round(s * 30);
      }
    }
    /** 발 자리 아래 올라설 물건 (데스크 등) */

      const ps = this.world?.floor?.platforms;
      let best = null;
      for (const p of ps ?? []) if (x >= p.x0 && x <= p.x1 && y >= p.y0 - 4 && y <= p.y1 + 4 && (!best || p.h > best.h)) best = p;
      return best;
    }

    skillMul = 1;

    /** 상태 효과 (출혈·오염·두통·혼란·공포·둔화·침묵, src/game/combat/status.ts). 적 공격의 debuff 이벤트가 건다 */
    status = new StatusSet();

    hurtTarget(t, dmg, kind, face, hitstop, o) {
      const w = this.world;
      if (!w || !t.alive) return;
      const r = REACT[kind] ?? REACT.light;
      const st = this.progression?.stats;
      const form = o.form ?? (o.src === "ranged" ? "gun" : "body");
      let scale = form === "gun" ? this.stat.aim * (st?.aim ?? 1) : form === "psi" ? this.stat.psi : this.stat.power * (st?.power ?? 1);
      if (o.src === "blast" && !o.form) scale = 1;
      if (o.src === "ranged" && !this.firstShotUsed && this.label === "에일린" && this.progression?.relics.includes("own_eileen")) {
        this.firstShotUsed = true;
        scale *= 2;
        w.say("치명!", 0.8);
      }
      const hit2 = { x0: t.x - 1, x1: t.x + 1, z0: 0, z1: 400, y: t.y, dy: 999, attackId: newAttackId(), dmg: dmg * this.dmgMul * scale * (o.src === "skill" ? this.skillMul : 1), react: r.react, dir: face, push: o.push ?? r.push, lift: r.lift, stop: 0, strength: r.strength, blow: kind === "blow", lethal: !!o.src && ["skill", "ranged", "blast"].includes(o.src) || form !== "body" || !!this.core.P.armed };
      const res = t.applyHit(hit2, w);
      if (res.kind === "immune") return;
      if (o.debuff && res.kind !== "guard") this.putStatus(t, o.debuff.k, (o.debuff.s > 0 ? o.debuff.s : STATUS[o.debuff.k].sec) * this.perks.outDur(o.debuff.k), o.debuff.k === "bleed" ? this.perks.bleedMul() : void 0);
      t.stop = Math.max(t.stop, Math.round(Math.max(hitstop, kind === "light" ? 0.085 : 0.16) * 30));
      this.combo = this.comboClock > 0 ? this.combo + 1 : 1;
      this.comboClock = 40;
      this.maxCombo = Math.max(this.maxCombo, this.combo);
      if (res.killed) {
        w.kills++;
        if (t.team === "enemy" && !(t instanceof Fighter)) w.emit({ type: "enemyDie", x: t.x, kind: t.kind });
      }
      return { killed: !!res.killed, guard: res.kind === "guard" };
    }
    /* ───────── 엔진 쪽 ───────── */
    frame() {
      const k = spr(this.y) * CHAR_K;
      return { hit: [], hurt: this.alive ? [{ x0: this.x - 26 * k, x1: this.x + 26 * k, z0: this.z, z1: this.z + this.height(), y: this.y, dy: 16 }] : [] };
    }
    /** 화면 위 키(서 있는 그림 기준) */
    height() {
      const m = this.art.A["1"];
      return (m ? Math.min(m.h - 4, m.floor) - (m.top[0] ?? 14) : 210) * spr(this.y) * CHAR_K;
    }
    applyHit(hit2, world) {
      const P2 = this.core.P;
      if (P2.inv > 0 || this.invuln > 0) {
        this.core.takeHit("hit");
        if (P2.dodging && this.perks.reflectOnDodge() && !this.attackerOf(hit2)) this.core.perkApi.reflect(hit2.dmg, this.perks.reflectSplit());
        return { kind: "immune" };
      }
      if (this.phase === "down") return { kind: "immune" };
      const lying = P2.mode === "down" || P2.mode === "belly";
      const front = hit2.dir === -P2.face;
      if (P2.mode === "block" && front) {
        const attacker = this.attackerOf(hit2);
        const pr = this.perks.blocked({ melee: !!attacker, attacker, dmg: hit2.dmg });
        if (!pr.noDrain) this.spendGuard((14 + hit2.dmg * 0.6) / this.perks.guardMul());
        if (pr.reflect) this.core.perkApi.reflect(hit2.dmg, this.perks.reflectSplit());
        if (this.guardBroken && this.perks.guardBreak()) {
          this.core.onRelease("block");
          world.emit({ type: "guard", x: this.x });
          return { kind: "guard" };
        }
        if (this.guardBroken) {
          this.core.onRelease("block");
          this.core.stun();
          P2.stunT = 1.1;
          world.emit({ type: "guard", x: this.x });
          return { kind: "hit" };
        }
        this.core.takeHit("hit");
        this.hp = Math.max(1, this.hp - Math.round(hit2.dmg * this.perks.blockChip()));
        world.emit({ type: "guard", x: this.x });
        return { kind: "guard" };
      }
      if (this.ward > 0 && hit2.dmg > 0) {
        this.ward--;
        this.invuln = 20;
        world.say("부적이 대신 맞았다", 1);
        return { kind: "immune" };
      }
      const strength = hit2.strength ?? (hit2.react === "down" || hit2.react === "launch" ? 3 : hit2.react === "stagger" ? 2 : 1);
      const armored = this.core.armored() || strength <= (this.progression?.stats.poise ?? 0) || P2.mode === "charge" && this.perks.keepCharge();
      let taken = hit2.dmg * this.stat.taken * (1 - (this.progression?.stats.defense ?? 0)) * (this.core.armored() ? 0.5 : 1) * this.perks.takenMul();
      if (this.shield > 0) {
        const cut = Math.min(this.shield, Math.round(taken));
        this.shield -= cut;
        taken -= cut;
        world.emit({ type: "hurt", x: this.x, dmg: cut, hp: this.hp });
        if (this.shield > 0) {
          P2.armorT = 0.28;
          P2.armorC = "#5ab8ff";
          return { kind: "armor" };
        }
        world.emit({ type: "guard", x: this.x });
        if (this.team === "player" && !this.allySide) {
          this.perks.shieldBroke();
          this.shieldMax = 0;
        } else {
          this.core.stun();
          P2.stunT = 0.8;
        }
        this.hp = Math.max(0, this.hp - Math.round(taken));
        return { kind: "hit" };
      }
      if (this.hp - Math.round(taken) <= 0) {
        const r = this.perks.lethal();
        if (r === "hold") {
          world.emit({ type: "hurt", x: this.x, dmg: taken, hp: this.hp });
          this.core.takeHit("hit");
          return { kind: "hit" };
        }
        if (r === "revive") {
          world.emit({ type: "hurt", x: this.x, dmg: taken, hp: this.hp });
          this.core.knock();
          return { kind: "hit" };
        }
      }
      this.hp = Math.max(0, this.hp - Math.round(taken));
      this.perks.hurt(taken);
      world.emit({ type: "hurt", x: this.x, dmg: taken, hp: this.hp });
      if (!this.lowWarned && this.hp < this.hpMax * 0.3) {
        this.lowWarned = true;
        world.emit({ type: "hp:low" });
      }
      if (this.hp <= 0 && this.team === "player" && !this.allySide && (fighterHooks.permaDeath?.() ?? false)) {
        this.dieNow(lying);
        return { kind: "hit", killed: true };
      }
      if (this.hp <= 0) {
        if (lying) this.core.takeHit("hit");
        else this.core.knock();
        this.phase = "down";
        return { kind: "hit", killed: true };
      }
      this.status.onHit();
      if (lying) {
        this.core.takeHit("hit");
        return { kind: "hit" };
      }
      if (armored) {
        this.core.takeHit("hit");
        return { kind: "armor" };
      }
      P2.face = Math.sign(hit2.dir) === 1 ? -1 : 1;
      const noKnock = this.perks.noKnock();

      }
      this.status.immune = k.mindGuard() ? /* @__PURE__ */ new Set(["fear", "confuse"]) : void 0;
      this.status.taintPer = k.taintPer();
      if (this.team !== "player") return k.tickNpc(dt);
      k.tick(dt);
      const w = this.world;
      if (!w) return;
      const gap = k.enemyGap();
      const foes = w.bodies.filter((b) => b.alive && b.team === "enemy" && b.hp > 0 && b.kind !== "bullet");
      for (const e of foes) e.perkGap = gap;
      if (perkWorld.friendlyFire) for (const e of foes) {

        if (!o) continue;

        for (const b of [e, o]) b.applyHit({ x0: b.x - 1, x1: b.x + 1, z0: 0, z1: 400, y: b.y, dy: 999, attackId: newAttackId(), dmg: pain(b), react: "down", dir: b.x >= e.x ? 1 : -1, push: 8, lift: 8, stop: 4, strength: 3 }, w);

  function floorHall(id, name, floor, doors, extra = {}) {
    const p = PAINTED[id];
    const chairBoxes = (p.chairs ?? []).map((ch) => ({ kind: "chair", x: ch.x, w: 54, y: ch.y, d: 20, h: 150, solid: false, face: ch.face, set: ch.set }));
    if (MAPEDIT_SPOTS[id]?.length) extra = { ...extra, spots: MAPEDIT_SPOTS[id] };
    if (p.stations?.length && !extra.stations) extra = { ...extra, stations: p.stations };
    const sc = floorHallBase(id, name, floor, doors, p, extra);
    if (chairBoxes.length) {
      const base2 = sc.boxes.length;
      sc.boxes = [...sc.boxes, ...chairBoxes];
      sc.spots = sc.spots.map((s) => s.chair !== void 0 ? { ...s, chair: base2 + s.chair } : s);
      sc.seats = chairBoxes.map((b, i) => ({ x: b.x, y: b.y, face: b.face, box: base2 + i, set: b.set }));
    }
    return sc;
  }
