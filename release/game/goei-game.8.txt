  function loadBadges(urlOf) {
    sheets = [];
    cache5.clear();
    for (let k = 1; k <= 6; k++) {
      const i = k - 1;
      void urlOf(`sheet${k}.png`).then((src) => {
        const im = new Image();
        im.onload = () => {
          sheets[i] = im;
        };
        im.src = src;
      }).catch(() => {
      });
    }
  }
  function badgeOf(name, grade = "희귀") {
    const n = CW_BADGE[name];
    if (n === void 0) return null;
    const key = `${n}|${grade}`;
    const hit2 = cache5.get(key);
    if (hit2) return hit2;
    const sh = sheets[Math.floor(n / 32)];
    if (!sh) return null;
    const cv = document.createElement("canvas");
    cv.width = cv.height = 64;
    const g = cv.getContext("2d");
    g.imageSmoothingEnabled = false;
    g.drawImage(sh, n % 8 * 64, Math.floor(n % 32 / 8) * 64, 64, 64, 0, 0, 64, 64);
    const ramp = RAMP[grade];
    if (ramp) {
      const d3 = g.getImageData(0, 0, 64, 64), px = d3.data;
      for (let i = 0; i < px.length; i += 4) {
        if (!px[i + 3]) continue;
        const t = Math.round(0.299 * px[i] + 0.587 * px[i + 1] + 0.114 * px[i + 2]) / 255;
        const [a, b, u] = t < 0.5 ? [ramp[0], ramp[1], t * 2] : [ramp[1], ramp[2], (t - 0.5) * 2];
        for (let c = 0; c < 3; c++) px[i + c] = Math.round(a[c] + (b[c] - a[c]) * u);
      }
      g.putImageData(d3, 0, 0);
    }
    cache5.set(key, cv);
    return cv;
  }

  function filt(o, type, f, q = 0.7, dest) {
    const n = ownNode(o, o.ctx.createBiquadFilter());
    n.type = type;
    n.frequency.value = f;
    n.Q.value = q;
    if (dest) n.connect(dest);
    return n;
  }
  function gainNode(o, v, dest) {
    const g = ownNode(o, o.ctx.createGain());
    g.gain.value = v;
    if (dest) g.connect(dest);
    return g;
  }
  function shaper(o, k, dest) {
    const n = ownNode(o, o.ctx.createWaveShaper());
    n.curve = driveCurve(k);
    n.oversample = "2x";
    if (dest) n.connect(dest);
    return n;
  }
  function buildGraph(ctx, dest) {
    const comp2 = ctx.createDynamicsCompressor();
    comp2.threshold.value = -10;
    comp2.knee.value = 8;
    comp2.ratio.value = 6;
    comp2.attack.value = 3e-3;
    comp2.release.value = 0.18;
    const master = ctx.createGain();
    master.connect(comp2).connect(dest);
    const bus = (v) => {
      const g = ctx.createGain();
      g.gain.value = v;
      g.connect(master);
      return g;
    };
    const music = bus(1);
    const sfx = bus(1);
    const amb = bus(1);
    const ui = bus(1);
    const duck = ctx.createGain();
    duck.connect(music);
    const longImpulse = makeImpulse(ctx, 2.6, 2.4, 0.7);
    const shortImpulse = makeImpulse(ctx, 0.45, 3, 0.3);
    const effects = {};
    for (const [name, target] of Object.entries({ music: duck, sfx, amb, ui })) {
      const verb = ctx.createConvolver();
      verb.buffer = longImpulse;
      const verbOut = ctx.createGain();
      verbOut.gain.value = 0.55;
      verb.connect(verbOut).connect(target);
      const room2 = ctx.createConvolver();
      room2.buffer = shortImpulse;
      const roomOut = ctx.createGain();
      roomOut.gain.value = 0.5;
      room2.connect(roomOut).connect(target);
      effects[name] = { verb, room: room2 };
    }
    const out = {
      ctx,
      sfx,
      ui,
      music: duck,
      amb,
      ...effects.sfx,
      effects,
      noise: makeNoise(ctx, 2),
      brown: makeNoise(ctx, 4, true)
    };
    return { out, gains: { master, music, sfx, amb, ui }, duck, comp: comp2 };
  }

  var rnd = (a, b) => a + Math.random() * (b - a);
  function thump(o, t, dest, f0, f1, dur, vol) {
    const g = gainNode(o, 0, dest);
    perc(g.gain, t, 2e-3, vol, dur);

  function bell(o, t, dest, f, dur, vol, ratio = 3.5, index = 3) {

  var PC = { c: 0, d: 2, e: 4, f: 5, g: 7, a: 9, b: 11 };
  function noteToMidi(s) {
    const m = /^([a-gA-G])([#b]?)(-?\d)$/.exec(s);
    if (!m) throw new Error(`음 표기 오류: ${s}`);
    const acc = m[2] === "#" ? 1 : m[2] === "b" ? -1 : 0;
    return 12 * (Number(m[3]) + 1) + PC[m[1].toLowerCase()] + acc;
  }
  function parseNotes(src) {
    const toks = [];
    for (const raw of src.split(/\s+/)) {
      if (!raw || raw === "|") continue;
      const m = /^(.*)\*(\d+)$/.exec(raw);
      if (m) {
        const n = Number(m[2]);
        if (m[1] === "." || m[1] === "-") for (let i = 0; i < n; i++) toks.push(m[1]);
        else {
          toks.push(m[1]);
          for (let i = 1; i < n; i++) toks.push("-");
        }
      } else toks.push(raw);
    }
    const evs = [];
    let cur = null;
    toks.forEach((t, i) => {
      if (t === "-") {
        if (cur) cur.len++;
        return;
      }
      cur = null;
      if (t === ".") return;
      let vel = 0.8;
      let body = t;
      if (body.endsWith("!")) vel = 1, body = body.slice(0, -1);
      else if (body.endsWith("?")) vel = 0.5, body = body.slice(0, -1);
      cur = { step: i, notes: body.split("+").map(noteToMidi), len: 1, vel };
      evs.push(cur);
    });
    return { evs, steps: toks.length };
  }
  function parseDrums(src) {
    const s = src.replace(/[\s|]/g, "");
    const evs = [];
    for (let i = 0; i < s.length; i++) if (s[i] !== ".") evs.push({ step: i, kind: s[i] });
    return { evs, steps: s.length };
  }
  var QUAL = {
    "": [0, 4, 7],
    m: [0, 3, 7],
    "7": [0, 4, 7, 10],
    maj7: [0, 4, 7, 11],
    m7: [0, 3, 7, 10],

  function chord(name) {
    const m = /^([A-G])([#b]?)(.*)$/.exec(name);
    if (!m || !(m[3] in QUAL)) throw new Error(`화음 표기 오류: ${name}`);
    const root = PC[m[1].toLowerCase()] + (m[2] === "#" ? 1 : m[2] === "b" ? -1 : 0);
    const tones = QUAL[m[3]].map((iv) => {
      let n = 48 + (root + iv) % 12;
      while (n < 52) n += 12;
      if (iv >= 12) n += 12;
      return n;
    });
    tones.sort((a, b) => a - b);
    return { root: (root + 12) % 12, tones };
  }
  var NAMES = ["c", "c#", "d", "d#", "e", "f", "f#", "g", "g#", "a", "a#", "b"];
  var tok = (notes) => notes.map((n) => `${NAMES[n % 12]}${Math.floor(n / 12) - 1}`).join("+");
  function comp(chords, rhythm, shift = 0) {
    return chords.split(/\s+/).filter(Boolean).map((c) => {
      const ns = chord(c).tones.map((n) => n + shift);
      return [...rhythm].map((ch) => ch === "x" ? tok(ns) : ch === "X" ? tok(ns) + "!" : ch).join(" ");
    }).join(" ");
  }
  function bass(chords, rhythm, oct = 2) {
    return chords.split(/\s+/).filter(Boolean).map((c) => {
      const { root } = chord(c);
      const r = 12 * (oct + 1) + root;
      const pick3 = (ch) => ch === "r" ? r : ch === "o" ? r + 12 : ch === "f" ? r + 7 : ch === "b" ? r + 10 : -1;
      return [...rhythm].map((ch) => pick3(ch) >= 0 ? tok([pick3(ch)]) : ch).join(" ");
    }).join(" ");
  }
  function arp(chords, shape, shift = 12) {
    return chords.split(/\s+/).filter(Boolean).map((c) => {
      const ts = chord(c).tones;
      return [...shape].map((ch) => {
        if (!/\d/.test(ch)) return ch;
        const i = Number(ch);
        return tok([ts[i % ts.length] + 12 * Math.floor(i / ts.length) + shift]);
      }).join(" ");
    }).join(" ");
  }

  function compileSong(def) {
    const chans = {};
    for (const [name, c] of Object.entries(def.chans)) chans[name] = { def: c, at: /* @__PURE__ */ new Map() };
    let base2 = 0;
    for (const entry of def.order) {
      const m = /^([A-Za-z0-9_]+)([+-]\d+)?$/.exec(entry);
      if (!m || !def.pats[m[1]]) throw new Error(`곡 순서 오류: ${entry}`);
      const pat = def.pats[m[1]];
      const tr = Number(m[2] ?? 0);
      let len = 0;
      const parsed = {};
      for (const [name, src] of Object.entries(pat)) {
        const c = def.chans[name];
        if (!c) throw new Error(`없는 악기 줄: ${name}`);
        parsed[name] = c.inst === "drums" ? parseDrums(src) : parseNotes(src);
        len = Math.max(len, parsed[name].steps);
      }
      for (const [name, p] of Object.entries(parsed)) {
        for (const e of p.evs) {
          const ev = "notes" in e ? { ...e, notes: e.notes.map((n) => n + tr) } : e;
          const at2 = chans[name].at;
          const k = base2 + e.step;
          (at2.get(k) ?? at2.set(k, []).get(k)).push(ev);
        }
      }
      base2 += len;
    }
    return { def, steps: base2, chans };
  }

        this.chanNodes.set(name, g);

    schedule(until, now = -Infinity) {
      const s = this.song;
      const sec = stepSec(s.def.bpm);
      while (!this.ended && this.nextTime < now - 0.05) {
        this.step++;
        this.nextTime += sec;
        if (!s.def.loop && this.step >= s.steps) this.ended = true;
      }
      while (!this.ended && this.nextTime < until) {
        const i = this.step % s.steps;
        const swing = i % 2 === 1 ? (s.def.swing ?? 0) * sec : 0;
        const t = this.nextTime + swing;
        for (const [name, c] of Object.entries(s.chans)) {
          const evs = c.at.get(i);
          if (!evs) continue;
          const dest = this.chanNodes.get(name);
          for (const e of evs) {
            if ("kind" in e) oneShot(this.o, (o) => drum(o, dest, t, e.kind, 1));
            else {

                oneShot(this.o, (o) => inst(o, dest, t, mtof(n + sour), e.len * sec * 0.95, e.vel));
              }
            }
          }
        }
        this.step++;
        this.nextTime += sec;
        if (!s.def.loop && this.step >= s.steps) this.ended = true;
      }
    }

        const g = node.gain;
        g.cancelScheduledValues(t);
        g.setValueAtTime(g.value, t);
        g.linearRampToValueAtTime(0, t + sec);
      }
      this.ended = true;
      this.scope.stopAt(t + sec);
    }
