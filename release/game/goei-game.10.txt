  var empty = () => ({ schema: "goei-session/1", revision: 0, hq: null, meta: null, checkpoint: null });
  var parse = (raw) => {
    if (!raw) return empty();
    const value = JSON.parse(raw);
    if (value?.schema === "goei-session/1") {
      if (!Number.isFinite(value.revision) || value.revision < 0 || value.hq !== null && typeof value.hq !== "string") throw new Error("손상된 저장 봉투");
      return { ...empty(), ...value };
    }
    if (!parseSave(raw)) throw new Error("지원하지 않는 저장");
    return { ...empty(), hq: raw };
  };
  var SessionStore = class {
    constructor(saves, slot, journal) {
      this.saves = saves;
      this.slot = slot;
      this.journal = journal;
      let cached = null;
      try {
        const raw = journal?.getItem(this.key);
        if (raw) cached = parse(raw);
      } catch {
      }
      if (cached) this.value = cached;
      this.ready = saves.read(slot).then((raw) => {
        const remote = parse(raw);
        if (!cached || remote.revision > cached.revision) this.value = remote;
        if (cached && cached.revision > remote.revision) this.persist();
      }).catch(() => {
        this.error = "저장 칸을 읽지 못했습니다. 이 기기의 복구 기록을 사용합니다.";
      }).then(() => {
        this.loaded = true;
      });
      this.hqSaves = {
        list: () => saves.list(),
        read: async (n) => {
          await this.ready;
          return n === slot ? this.value.hq : parse(await saves.read(n)).hq;
        },
        write: async (n, hq2, info) => {
          if (!this.loaded) await this.ready;
          if (n !== slot) {
            let oldRevision = 0;
            try {
              oldRevision = parse(await saves.read(n)).revision;
            } catch {
            }
            try {
              oldRevision = Math.max(oldRevision, parse(this.journal?.getItem(this.keyFor(n)) ?? null).revision);
            } catch {
            }
            const snapshot = { ...this.value, hq: hq2, revision: Math.max(Date.now(), this.value.revision + 1, oldRevision + 1) };
            const text2 = JSON.stringify(snapshot);
            try {
              this.journal?.setItem(this.keyFor(n), text2);
            } catch {
            }
            return saves.write(n, text2, { ...info, schema: "goei-session/1" });
          }
          this.value.hq = hq2;
          const completed = parseSave(hq2);
          const run = this.value.checkpoint?.run;
          if (!completed?.active && completed?.last && completed.last.runId === (run?.launch?.runId ?? run?.runId)) this.value.checkpoint = null;
          this.persist();
          await this.flush();
        },
        restore: async (n) => {
          await this.ready;
          await this.flush();
          let remote = null, cached2 = null;
          try {
            remote = parse(await saves.read(n));
          } catch {
          }
          try {
            cached2 = parse(this.journal?.getItem(this.keyFor(n)) ?? null);
          } catch {
          }
          const next = cached2 && (!remote || cached2.revision > remote.revision) ? cached2 : remote;
          if (!next?.hq || !parseSave(next.hq)) return null;
          this.value = { ...next, revision: Math.max(next.revision, this.value.revision) };
          this.persist();
          await this.flush();
          return this.value.hq;
        }
      };
    }
    value = empty();
    queue = Promise.resolve();
    loaded = false;
    ready;
    error = null;
    hqSaves;
    keyFor(slot) {
      return `goei-session-journal:${slot}`;
    }
    get key() {
      return this.keyFor(this.slot);
    }
    get checkpoint() {
      return this.value.checkpoint;
    }
    get meta() {
      return this.value.meta;
    }
    get hasSave() {
      return !!(this.value.hq || this.value.checkpoint);
    }
    setMeta(meta) {
      this.value.meta = JSON.parse(JSON.stringify(meta));
    }
    setCheckpoint(checkpoint) {
      this.value.checkpoint = checkpoint;
      this.persist();
    }
    persist() {
      this.value.revision = Math.max(Date.now(), this.value.revision + 1);
      const text2 = JSON.stringify(this.value);
      try {
        this.journal?.setItem(this.key, text2);
      } catch {
        this.error = "기기 복구 기록을 쓰지 못했습니다.";
      }
      this.queue = this.queue.catch(() => {
      }).then(() => this.saves.write(this.slot, text2, {
        title: this.value.checkpoint ? "출동 중 · 안전 지점" : "자동 저장",
        summary: "본부 · 규칙서 · 출동 기록",
        schema: "goei-session/1"
      })).catch(() => {
        this.error = "자동 저장에 실패했습니다. 기기 복구 기록을 유지합니다.";
      });
    }
    flush() {
      return this.queue;
    }
  };
