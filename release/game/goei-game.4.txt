      if (move3.frames[0]?.tele) w.emit({ type: this.boss ? "bossTell" : "enemyTell", x: this.x, kind: this.kind });
    }
    frameNow() {
      return this.state === "attack" && this.move ? this.move.frames[this.mf] ?? null : null;
    }
    /** 지금 몸 기준 피격 상자 */

      if (f) return f.inv ? [] : f.hurt;
      if (this.state === "air") return this.def.body.map((b) => ({ ...b, z1: b.z0 + (b.z1 - b.z0) * 0.7 }));
      return this.def.body;
    }

      const hurt2 = this.localHurt().map((b) => toWorld2(b, this.x, this.y, this.z, this.dir));
      const hit2 = [];

        const spec = this.move.specs[f.spec ?? 0];
        for (const b of f.hit) {
          const wb = toWorld2(b, this.x, this.y, this.z, this.dir);
          const h = {
            ...wb,
            attackId: this.atk + (f.spec ?? 0),
            dmg: Math.round(spec.dmg * this.power),
            react: spec.react,
            dir: this.dir,
            push: spec.push,
            lift: spec.lift,
            stop: spec.stop,
            low: spec.low,
            grab: !!spec.grab,
            armorBreak: spec.unblockable,
            strength: spec.strength,
            noise: spec.noise
          };
          const debuff = spec.debuff ?? (this.has("bleed") && spec.dmg > 0 ? { kind: "bleed", ticks: 120 } : void 0);
          if (debuff) {
            h.debuff = debuff;
            armDebuff(h, debuff, this.id);
          }
          hit2.push(h);
        }
      }
      const extra = this.brain.hits?.(this);
      if (extra) hit2.push(...extra);
      if (this.state === "grabbing" && this.grabHitId) {
        hit2.push({ ...this.grabBox, attackId: this.grabHitId, dmg: this.grabDmg, react: "none", dir: this.dir, grab: true, armorBreak: true, stop: 1, strength: 1 });

      for (const k in this.cds) if (this.cds[k] > 0) this.cds[k]--;

      if (this.weakOpen > 0) this.weakOpen--;
      if (this.bubbleT > 0 && --this.bubbleT === 0) this.bubble = "";
      if (this.state === "dead") {

        return;

      if (this.state === "spawn" && this.st >= 12) this.set("idle");

      switch (this.state) {
        case "spawn":
          if (this.st >= 12) this.set("idle");
          break;
        case "hurt":
        case "stunned":
          if (--this.stunT <= 0) this.set("idle");
          break;
        case "down":
          if (this.st >= (this.mem.downFor ?? 30)) this.set("getup"), this.launches = 0;
          break;
        case "getup":
          if (this.st >= 12) this.inv = 10, this.set("idle");
          break;
        case "bound":
          if (--this.bindT <= 0) this.set("idle");
          break;
        case "grabbing":
          this.holdGrab(w);
          break;
        case "attack":
          this.runMove(w);
          break;
        case "idle":
        case "walk":
          this.brain.think?.(this, w);
          break;
      }
    }

      if (!m) return this.set("idle");
      const f = m.frames[this.mf];
      if (!f) return this.endMove();

      if (f.vz) this.vz = f.vz, this.mem.cling = false;
      if (f.fire && this.mf === firstIndex(m, this.mf, "fire")) this.fire(f.fire, w);
      if (f.ev && this.mf === firstIndex(m, this.mf, "ev")) this.brain.ev?.(this, f.ev, w);
      const spec = f.hit.length ? m.specs[f.spec ?? 0] : null;
      if (spec?.grab && this.state === "attack") {

        const free = p.alive && !p.airborne && !/evade|down|dead|grabbed|getup/.test(p.state);
        if (free && Math.abs(p.y - this.y) < 26) {
          const hit2 = this.frame().hit;
          const hurt2 = p.frame().hurt;
          if (hit2.some((h) => hurt2.some((b) => h.x0 < b.x1 && b.x0 < h.x1 && h.z0 < b.z1 && b.z0 < h.z1))) return this.beginGrab(spec.grab, w);
        }
      }
      if (this.state !== "attack") return;
      this.mf++;
      if (this.mf >= m.frames.length) this.endMove();
    }
    endMove() {
      this.move = null;
      this.mf = 0;
      this.set("idle");
    }

    fire(name, w, opts = {}) {

    }
    physics(w) {

      const [ly0, ly1] = w.lane;

              w.emit({ type: "down", x: this.x, who: "enemy" });
            }
          }
        }
      }
      if (this.state !== "flee" && this.state !== "hidden" && this.age > 30) this.x = Math.max(bx0 + 10, Math.min(bx1 - 10, this.x));
      this.y = Math.max(ly0, Math.min(ly1, this.y));
    }
    applyHit(hit2, w) {
      if (!this.alive || this.state === "dead" || this.state === "spawn" || this.inv > 0) return { kind: "immune" };
      if (this.hitBy.has(hit2.attackId)) return { kind: "immune" };
      this.hitBy.add(hit2.attackId);
      if (this.hitBy.size > 24) this.hitBy.delete(this.hitBy.values().next().value);
      const hurt2 = this.localHurt(w).map((b) => toWorld2(b, this.x, this.y, this.z, this.dir));
      const part = pickPart(hurt2, hit2);
      const mod2 = this.brain.hit?.(this, hit2, part, w) ?? {};
      if (mod2.immune) {
        if (mod2.text) w.fx("text", this.x, this.y, this.def.h + 20, mod2.text);
        return { kind: "immune" };
      }
      const front = hit2.dir === -this.dir;
      let mult = mod2.mult ?? 1;
      let guard3 = !!mod2.guard;
      if (!guard3 && this.def.role === "shield" && front && !hit2.armorBreak && !hit2.grab && this.state !== "hurt" && this.state !== "air") {
        guard3 = true;
        mult *= this.kind === "doorframe" ? 0.5 : 0.1;
      }
      const weak = !!mod2.weak || this.weakOpen > 0;
      if (weak) mult *= 2;
      const dmg = Math.max(1, Math.round(hit2.dmg * mult));
      this.hp -= dmg;
      this.lastHurt = this.age;
      this.flash = 6;
      if (weak) w.fx("text", this.x, this.y, this.def.h + 30, mod2.text ?? "약점");
      this.brain.after?.(this, w, dmg);
      if (this.groggyMax && this.state !== "stunned" && this.hp > 0) {
        this.groggy += dmg;
        if (this.groggy >= this.groggyMax) {
          this.groggy = 0;
          if (this.state === "grabbing") this.releaseGrab(w);
          this.move = null;
          this.set("stunned");
          this.stunT = 60;
          w.fx("text", this.x, this.y, this.def.h + 40, "그로기");
          w.emit({ type: "custom", name: "groggy", data: { id: this.id, kind: this.kind } });
          return { kind: "hit" };
        }
      }
      if (this.hp <= 0) {
        this.die(w, hit2);
        return { kind: "hit", killed: true };
      }
      if (guard3) {
        this.x += hit2.dir * 3;
        w.fx("spark", this.x - this.dir * -30, this.y, 100);
        return { kind: "guard" };
      }
      if (mod2.noReact) return { kind: hit2.react === "none" ? "hit" : "armor" };
      const f = this.frameNow();
      let armor2 = Math.max(this.armorBase, f?.armor ?? 0, this.has("armor") && f?.hit.length ? 1 : 0);
      if (weak || hit2.armorBreak || this.state === "stunned" || this.state === "bound") armor2 = 0;
      if (REACT_RANK[hit2.react] <= armor2) return { kind: hit2.react === "none" ? "hit" : "armor" };
      this.react(hit2, w);
      return { kind: "hit" };
    }
    react(hit2, w) {
      if (this.state === "grabbing") this.releaseGrab(w);
      this.move = null;
      const push = hit2.push ?? 4;
      switch (hit2.react) {
        case "none":
          return;
        case "flinch":
          if (this.state === "air") return void (this.vz = Math.max(this.vz, 3));
          this.set("hurt");
          this.stunT = 10;
          this.vx = hit2.dir * push * 0.4;
          return;
        case "stagger":
          if (this.state === "air") return void (this.vz = Math.max(this.vz, 4), this.vx = hit2.dir * push * 0.5);
          this.set("hurt");
          this.stunT = 16;
          this.vx = hit2.dir * push;
          return;
        case "launch":
          if (this.launches >= 2) return this.knockDown(hit2);
          this.launches++;
          this.set("air");
          this.vz = hit2.lift ?? 12;
          this.vx = hit2.dir * push * 0.5;
          this.z = Math.max(this.z, (this.def.hover ?? 0) + 1);
          return;
        case "down":
          return this.knockDown(hit2);
        case "bind":
          this.set("bound");
          this.bindT = 75;
          return;
      }
    }
    knockDown(hit2) {
      this.set("air");
      this.vz = Math.max(this.vz, hit2.lift ?? 8);
      this.vx = hit2.dir * (hit2.push ?? 6);
      this.z = Math.max(this.z, (this.def.hover ?? 0) + 1);

        fireDropHook({ enemy: this, kind: this.kind, x: this.x, y: this.y, drops, world: w });
      }
    }

      drawEnemy(this, ctx);

  var PART_ORDER = ["corner", "mouth", "back", "head", "shield", "core", "weapon", "arm", "torso", "legs"];
  function pickPart(hurt2, hit2) {
    const hits = hurt2.filter((b) => hit2.x0 < b.x1 && b.x0 < hit2.x1 && hit2.z0 < b.z1 && b.z0 < hit2.z1);
    if (!hits.length) return hurt2[0]?.part;
    hits.sort((a, b) => PART_ORDER.indexOf(a.part ?? "torso") - PART_ORDER.indexOf(b.part ?? "torso"));
    const shield2 = hits.find((b) => b.part === "shield");
    return shield2 ? "shield" : hits[0].part;
  }

    const f = (c) => Math.max(0, Math.min(255, Math.round(k >= 0 ? c + (255 - c) * k : c * (1 + k))));
    return `rgb(${f(n >> 16)},${f(n >> 8 & 255)},${f(n & 255)})`;
  }

  var lerp = (a, b, t) => a + (b - a) * t;
  function lerpBox(a, b, t) {
    return { fx0: lerp(a.fx0, b.fx0, t), fx1: lerp(a.fx1, b.fx1, t), z0: lerp(a.z0, b.z0, t), z1: lerp(a.z1, b.z1, t), dy: lerp(a.dy, b.dy, t), part: a.part };
  }
  var at = (b, t) => b ? typeof b === "function" ? b(t) : b : [];
  function bake(id, label3, body2, phases, specs, extra = {}) {
    const frames = [];
    for (const p of phases) {
      for (let i = 0; i < p.n; i++) {
        const t = p.n <= 1 ? 1 : i / (p.n - 1);
        const f = { hit: at(p.hit, t), hurt: p.hurt ? at(p.hurt, t) : body2, armor: p.armor ?? 0 };
        if (f.hit.length) f.spec = p.spec ?? 0;
        const vx = typeof p.vx === "function" ? p.vx(t) : p.vx;
        if (vx) f.vx = vx;
        if (p.vz && i === 0) f.vz = p.vz;
        if (p.tele) f.tele = true;
        if (p.fire && i === 0) f.fire = p.fire;
        if (p.inv) f.inv = true;
        if (p.ev && i === 0) f.ev = p.ev;
        frames.push(f);
      }
    }
    return { id, label: label3, frames, specs, ...extra };
  }

  function poolFor(floor, intrusion = null, allowMid = false) {
    if (floor === "b12") return [];
    let kinds;
    if (intrusion === "bureau") kinds = [...INTRUDERS.bureau, ...allowMid ? INTRUDERS.bureauMid : []];
    else if (intrusion === "daydream") kinds = INTRUDERS.daydream;
    else kinds = ROSTER.filter((d3) => d3.types.includes(floor) && d3.price > 0 && (allowMid || !d3.mid)).map((d3) => d3.kind);
    return kinds.map((k) => ROSTER_BY_KIND[k]).filter(Boolean);
  }
  function place(d3, i, area, r) {
    const inner = ["fixed", "floorgrab", "ambush", "ceiling", "rule", "stain"].includes(d3.ai);
    const y = d3.kind === "wallface" ? area.y0 : area.y0 + 10 + r() * (area.y1 - area.y0 - 20);
    if (inner) return { x: area.x0 + 160 + r() * (area.x1 - area.x0 - 320), y };
    const left = i % 2 === 0 ? r() < 0.7 : r() < 0.3;
    const gap = Math.floor(i / 2) * 120;
    return { x: left ? area.x0 - 40 - gap - r() * 60 : area.x1 + 40 + gap + r() * 60, y };
  }
  function planEncounter(o) {
    let plan = buy3(o);
    for (let i = 0; i < 3 && o.recent?.slice(-3).includes(plan.signature); i++) plan = buy3(o);
    return plan;
  }
  function buy3(o) {
    const r = o.rand;
    const area = o.area ?? DEFAULT_AREA;
    const intrusion = o.intrusion ?? null;
    const pool = poolFor(o.floor, intrusion, o.allowMid);
    const plan = { budget: o.budget, spent: 0, waves: [], elites: 0, intrusion, signature: "" };
    if (!pool.length || o.budget <= 0) return plan;
    const cheapest = Math.min(...pool.map((d3) => d3.price));
    const nWaves = o.budget < 30 ? 1 : o.budget < 75 ? 2 : 3;
    const reserve = o.depth >= 8 ? Math.round(o.budget * 0.15) : 0;
    let left = o.budget - reserve;
    const mids = pool.filter((d3) => d3.mid && d3.price <= left * 0.6);
    let mid = null;
    if (o.allowMid && mids.length && r() < 0.5) {
      mid = mids[Math.floor(r() * mids.length)];
      left -= mid.price;
    }
    const normal = pool.filter((d3) => !d3.mid);
    const softCap = Math.min(SCREEN_CAP, 3 + Math.floor(o.depth / 2));
    for (let wv = 0; wv < nWaves; wv++) {
      const share = wv === nWaves - 1 ? left : Math.round(left / (nWaves - wv));
      let wl = share;
      const wave = [];
      if (mid && wv === nWaves - 1) wave.push({ kind: mid.kind, ...place(mid, 0, area, r), wave: wv });
      let guard3 = 0;
      while (wl >= cheapest && wave.length < softCap && guard3++ < 40) {
        const ok = normal.filter((d4) => d4.price <= wl && wave.filter((s2) => s2.kind === d4.kind).length < (d4.cap ?? SCREEN_CAP));
        if (!ok.length) break;
        const weights = ok.map((d4) => 1 / (1 + wave.filter((s2) => s2.kind === d4.kind).length * 1.5));
        let pick6 = r() * weights.reduce((a, b) => a + b, 0);
        let d3 = ok[ok.length - 1];
        for (let i = 0; i < ok.length; i++) if ((pick6 -= weights[i]) <= 0) d3 = ok[i], i = ok.length;
        wave.push({ kind: d3.kind, ...place(d3, wave.length, area, r), wave: wv });
        wl -= d3.price;
      }
      left -= share - wl;
      plan.waves.push(wave);
    }
    left += reserve;
    const all = plan.waves.flat().filter((s2) => !ROSTER_BY_KIND[s2.kind]?.mid);
    for (const s2 of shuffled(all, r)) {
      if (left < ELITE_COST) break;
      s2.elite = true;
      s2.affixes = [AFFIXES[Math.floor(r() * AFFIXES.length)]];
      if (left >= ELITE_COST * 3 && r() < 0.4) {
        const second = AFFIXES.filter((a) => a !== s2.affixes[0]);
        s2.affixes.push(second[Math.floor(r() * second.length)]);
        left -= ELITE_COST;
      }
      left -= ELITE_COST;
      plan.elites++;
    }
    plan.waves = plan.waves.filter((w) => w.length);
    plan.spent = o.budget - left;
    plan.signature = plan.waves.map((w) => w.map((s2) => s2.kind + (s2.elite ? "*" : "")).sort().join(",")).join("|");
    return plan;
  }
  function shuffled(a, r) {
    const b = a.slice();
    for (let i = b.length - 1; i > 0; i--) {
      const j = Math.floor(r() * (i + 1));
      [b[i], b[j]] = [b[j], b[i]];
    }
    return b;
  }
