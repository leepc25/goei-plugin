setStick(nx, ny);
    }
    setStick(x, y) {
      const dialog = this.mode === "dialog";
      this.hub.setAnalog("touch", dialog ? 0 : x, dialog ? 0 : y);
      const dir = (code, v) => {
        if (this.dirOn[code] === v) return;
        this.dirOn[code] = v;
        if (v) this.hub.press(code, "touch");
        else this.hub.release(code, "touch");
      };
      const th = (on) => on ? 0.3 : 0.42;
      if (dialog) {
        const vert = Math.abs(y) >= Math.abs(x);
        dir("ArrowUp", vert && y < -th(this.dirOn.ArrowUp));
        dir("ArrowDown", vert && y > th(this.dirOn.ArrowDown));
        dir("ArrowLeft", !vert && x < -th(this.dirOn.ArrowLeft));
        dir("ArrowRight", !vert && x > th(this.dirOn.ArrowRight));
        return;
      }
      dir("ArrowLeft", x < -th(this.dirOn.ArrowLeft));
      dir("ArrowRight", x > th(this.dirOn.ArrowRight));
      dir("ArrowUp", y < -th(this.dirOn.ArrowUp) * 1.15);
      dir("ArrowDown", y > th(this.dirOn.ArrowDown) * 1.15);
      const wantRun = this.st().stickRun && Math.abs(x) > 0.9;
      if (wantRun && !this.runOn) {
        this.runOn = true;
        this.runSince = performance.now();
        const pending = this.runTimer !== 0;
        clearTimeout(this.runTimer);
        this.runTimer = 0;
        if (!pending) this.key(CODE.evade, true);
      } else if (!wantRun && this.runOn) {
        this.runOn = false;
        const left = 250 - (performance.now() - this.runSince);
        clearTimeout(this.runTimer);
        this.runTimer = 0;
        if (left > 0)
          this.runTimer = window.setTimeout(() => {
            this.runTimer = 0;
            this.key(CODE.evade, false);
          }, left);
        else this.key(CODE.evade, false);
      }
    }
    releaseAll() {
      const ids = [...this.ptr.keys()];
      this.ptr.clear();
      for (const id of ids) {
        try {
          if (this.surface.hasPointerCapture(id)) this.surface.releasePointerCapture(id);
        } catch {
        }
      }
      clearTimeout(this.runTimer);
      this.runTimer = 0;
      this.runOn = false;
      for (const t of this.timers) clearTimeout(t);
      this.timers.clear();
      this.uniq = { timer: 0, charging: false, down: false };
      this.stickPid = null;
      this.setStick(0, 0);
      this.count.clear();
      for (const b of this.btns) b.el.classList.remove("on");
      this.hub.releaseSource("touch");
      this.restBase();
    }
    dispose() {
      const s = this.surface;
      s.removeEventListener("pointerdown", this.onDown);
      s.removeEventListener("pointermove", this.onMove);
      s.removeEventListener("pointerup", this.onUp);
      s.removeEventListener("pointercancel", this.onUp);
      s.removeEventListener("lostpointercapture", this.onUp);
      s.removeEventListener("contextmenu", this.onContext);
      s.removeEventListener("gesturestart", this.onGesture);
      s.removeEventListener("dblclick", this.onContext);
      s.removeEventListener("touchend", this.onTouchEnd);
      window.removeEventListener("pointerup", this.onUp);
      window.removeEventListener("pointercancel", this.onUp);
      window.removeEventListener("blur", this.onBlur);
      document.removeEventListener("focusin", this.onIme);
      document.removeEventListener("focusout", this.onIme);
      if (typeof visualViewport !== "undefined") visualViewport?.removeEventListener("resize", this.onIme);
      document.removeEventListener("visibilitychange", this.onVisibility);
      window.removeEventListener("touchstart", this.onAnyTouch, { capture: true });
      this.releaseAll();
      this.root.remove();
    }
  };

  function navigationTarget(props2, x) {
    let best = null;
    for (const p of props2) if (Math.abs(p.x - x) < 60 && (!best || Math.abs(p.x - x) < Math.abs(best.x - x))) best = p;
    return best;

  function navigationFeedback(w) {
    if (w.player.seat) return { objective: "본부 · 의자에서 휴식 중", target: { id: "seat:stand", text: "일어나기", kind: "interact" } };
    const stage = w.stage;
    const objective = w.floor.type === "hq" ? "본부 · 표시된 출입구와 자리 앞에서 조사" : "표시된 장치를 조사하세요";
    const result = { objective, target: null };
    if (w.ended || w.player.hp <= 0 || w.player.carry) return result;
    const p = navigationTarget(w.props, w.player.x);
    if (!p) return result;
    const set = (text, kind = "interact", reason) => {
      result.target = { id: p.id, text, kind: reason ? "locked" : kind, reason };
      return result;
    };
    const scene = w.floor.type === "hq" ? stage?.current?.() : null;
    if (scene && p.kind === "hubdoor") {
      const d3 = scene.doors.find((d4) => `door:${d4.id}` === p.id);
      if (!d3) return result;
      if (!d3.open) return set("닫힌 문", "locked", d3.needs ? "본부 시설 해금 필요" : "출입할 수 없는 문");
      if (d3.kind === "elevator") return set("승강기 · 층 선택");
      return d3.to ? set(d3.side === "S" ? "출구 · 복도로 이동" : "출입구 · 이동", "exit") : set("배경 문 · 출입 불가", "decoration");

    fill(text) {
      const f = this.ctx.flags;
      return text.replace(/\{([\w.가-힣]+)\}/g, (_, k) => k === "name" ? this.ctx.speakerName("agent") : String(f[k] ?? ""));
    }
    jump(to) {
      const top = this.stack[this.stack.length - 1];
      if (!top) return;
      if (to === "end") top.pc = top.script.nodes.length;
      else top.pc = top.script.labels[to] ?? top.script.nodes.length;
    }
    say(n) {
      const who = n.who === "agent" ? this.ctx.agent : n.who;
      return { kind: "say", who, name: this.ctx.speakerName(who), text: this.fill(n.text), mood: n.mood ?? "base" };
    }
    run() {
      const { flags, rng: rng3 } = this.ctx;
      for (let guard2 = 0; guard2 < 2e3; guard2++) {
        const top = this.stack[this.stack.length - 1];
        if (!top) {
          delete flags.d100;
          return { kind: "end" };
        }
        flags.d100 = Math.floor(rng3() * 100);
        const n = top.script.nodes[top.pc];
        if (!n) {
          this.stack.pop();
          continue;
        }
        top.pc++;
        switch (n.t) {
          case "say":
            return this.say(n);
          case "pick":
            return this.say(n.lines[Math.floor(rng3() * n.lines.length) % n.lines.length]);
          case "agentline": {
            const text = this.ctx.agentLine(this.ctx.agent, n.tag, rng3);
            if (!text) break;
            const who = n.who === "agent" ? this.ctx.agent : n.who;
            return { kind: "say", who, name: this.ctx.speakerName(who), text: this.fill(text), mood: "base" };
          }
          case "choice": {
            const visible = n.options.map((o, index) => ({ o, index })).filter(({ o }) => test(o.if, flags));
            if (!visible.length) break;
            this.pending = n.options;
            return { kind: "choice", options: visible.map(({ o, index }) => ({ text: this.fill(o.text), index })) };
          }
          case "set":
            applySet(n.key, n.op, n.expr, flags);
            break;
          case "goto":
            if (test(n.if, flags)) this.jump(n.to);
            break;
          case "emit":
            this.ctx.emit?.(n.event, this.fill(n.arg));
            break;
          case "call":
            this.stack.push({ script: this.lib.get(n.script), pc: 0 });
            break;
          case "end":
            this.stack = [];
            delete flags.d100;
            return { kind: "end" };
        }
      }
      throw new ScriptError("대본이 끝없이 돌아요");
    }
  };

  var Cutscene = class {
    constructor(def, io2) {
      this.def = def;
      this.io = io2;
    }
    i = 0;
    cur = null;
    done = false;
    elapsed = 0;
    begin(step2) {
      const io2 = this.io;
      let talk = false;
      switch (step2.k) {
        case "bg":
          io2.bg?.(step2.scene);
          break;
        case "fade":
          io2.fade?.(step2.to, step2.ms);
          break;
        case "caption":
          io2.caption?.(step2.text, step2.ms);
          break;
        case "say":
          io2.dialogue(step2.script);
          talk = true;
          break;
        case "line":
          io2.line(step2.who, step2.text, step2.mood ?? "base");
          talk = true;
          break;
        case "agentline":
          io2.agentLine(step2.tag);
          talk = true;
          break;
        case "bark":
          io2.bark?.(step2.who, step2.text, step2.ms ?? 2200);
          break;
        case "sfx":
          io2.sfx?.(step2.name);
          break;
        case "shake":
          io2.shake?.(step2.ms, step2.amp ?? 4);
          break;
        case "actor":
          io2.actor?.(step2.id, { x: step2.x, y: step2.y, face: step2.face, anim: step2.anim, show: step2.show }, step2.ms ?? 0);
          break;
        case "camera":
          io2.camera?.({ x: step2.x, y: step2.y, zoom: step2.zoom }, step2.ms ?? 0);
          break;
        case "panel":
          io2.panel?.(step2.text, step2.ms ?? 0);
          break;
        case "emit":
          io2.emit?.(step2.event, step2.arg ?? "");
          break;
        case "par":
          return { step: step2, left: 0, talk: false, children: step2.steps.map((s) => this.begin(s)) };
      }
      return { step: step2, left: durationOf(step2), talk };
    }
    tick(r, ms) {
      if (r.children) return r.children.map((c) => this.tick(c, ms)).every(Boolean);
      r.left -= ms;
      if (r.talk && this.io.talking()) return false;
      return r.left <= 0;
    }
    /** dt: 초 */
    update(dt) {
      if (this.done) return;
      const ms = dt * 1e3;
      this.elapsed += ms;
      for (let guard2 = 0; guard2 < 64; guard2++) {
        if (!this.cur) {
          const s = this.def.steps[this.i++];
          if (!s) {
            this.done = true;
            return;
          }
          this.cur = this.begin(s);
        }
        if (!this.tick(this.cur, guard2 === 0 ? ms : 0)) return;
        this.cur = null;
      }
    }

        else if (s.k === "camera") this.io.camera?.({ x: s.x, y: s.y, zoom: s.zoom }, 0);
        else if (s.k === "panel") this.io.panel?.(s.text, 0);
        else if (s.k === "emit") this.io.emit?.(s.event, s.arg ?? "");
      }
      this.i = this.def.steps.length;
      this.cur = null;
      this.done = true;
    }
  };

    constructor(opts = {}) {

      this.opts.model = model;
    }

    }

    }

    async reply(npc, text, agent, flags) {
      const clean = text.trim().slice(0, 200);
      if (!clean) return { text: "…", source: "offline" };
      const log = this.history(npc).slice();
      let out;
      const model = this.opts.model;
      if (model && !this.busy) {
        this.busy = true;
        const ctrl = new AbortController();
        const timer = setTimeout(() => ctrl.abort(), this.opts.timeoutMs ?? 25e3);
        try {
          const raw = await Promise.race([
            model(buildMessages(npc, agent, flags, log, clean), { signal: ctrl.signal }),
            new Promise((_, rej) => ctrl.signal.addEventListener("abort", () => rej(new Error("응답이 늦어요"))))
          ]);
          const t = sanitize(raw, PERSONAS[npc].name);
          out = t ? { text: t, source: "model" } : { text: offlineReply(npc, clean, agent, this.opts.rng), source: "offline", error: "빈 응답" };
        } catch (e) {
          out = { text: offlineReply(npc, clean, agent, this.opts.rng), source: "offline", error: e instanceof Error ? e.message : String(e) };
        } finally {
          clearTimeout(timer);
          this.busy = false;
        }
      } else out = { text: offlineReply(npc, clean, agent, this.opts.rng), source: "offline" };
      log.push({ role: "user", content: clean }, { role: "assistant", content: out.text });
      this.logs.set(npc, log.slice(-16));
      return out;
    }

  function choose(o) {
    const t = TEMPER[o.name];
    let best = null;
    for (const a of ACTIVITIES) {
      if (!o.has(a.clip) && !(ACT_OPTS[a.id] ?? []).some((x) => o.has(`line:${x.line}`))) continue;
      if (a.need === "train" && !o.fighter) continue;
      const lack = 100 - o.needs[a.need];
      if (lack < 25) continue;
      const kinds = a.id === "hobby" && t?.hobbyAt ? t.hobbyAt : a.where;
      const places = o.roam.filter((s) => kinds.includes(o.graph.kind(s)));
      for (const s of places) {
        if (a.partner && !o.partnerIn(s)) continue;
        const hops = s === o.scene ? 0 : o.graph.route(o.scene, s)?.length ?? 99;
        if (hops >= 99) continue;
        let w = (lack / 100) ** 2 * 100;
        if (a.need === "work") w *= o.duty ? 1.6 : 0;
        if (a.need === "bladder" || a.need === "bowel") w *= lack > 70 ? 2.2 : 0.7;
        if (o.last === a.id) w *= 0.5;
        const score = w / (1 + hops * 0.5) * (0.9 + o.rand() * 0.2);
        if (!best || score > best.score) best = { act: a, scene: s, score };
      }
    }
    return best && best.score > 8 ? best : null;
  }
