id)) throw new ScriptError(`대본 id 중복: ${s2.id}`);
        this.scripts.set(s2.id, s2);
      }
      return this;
    }
    get(id) {
      const s2 = this.scripts.get(id);
      if (!s2) throw new ScriptError(`없는 대본: ${id}`);
      return s2;
    }
    /** 모든 goto·선택지·call이 실제로 있는 곳을 가리키는지 */
    validate() {
      const errs = [];
      for (const s2 of this.scripts.values())
        for (const n of s2.nodes) {
          const target = (to) => to === "end" || s2.labels[to] !== void 0 || errs.push(`${s2.id}:${n.line} 없는 표지 ${to}`);
          if (n.t === "goto") target(n.to);
          if (n.t === "choice") n.options.forEach((o) => target(o.to));
          if (n.t === "call" && !this.scripts.has(n.script)) errs.push(`${s2.id}:${n.line} 없는 대본 ${n.script}`);
          try {
            if (n.t === "goto" && n.if) evaluate(n.if, {});
            if (n.t === "set") evaluate(n.expr, {});
            if (n.t === "choice") n.options.forEach((o) => o.if && evaluate(o.if, {}));
          } catch (e) {
            errs.push(`${s2.id}:${n.line} ${e.message}`);
          }
        }
      return errs;
    }
  };
  var StoryRunner = class {
    constructor(lib, ctx) {
      this.lib = lib;
      this.ctx = ctx;
    }
    stack = [];
    pending = null;
    /** 지금 보이는 걸음 */
    current = { kind: "end" };
    get done() {
      return this.current.kind === "end";
    }
    start(id) {
      this.stack = [{ script: this.lib.get(id), pc: 0 }];
      this.pending = null;
      this.ctx.flags.agent = this.ctx.agent;
      this.ctx.flags[`seen.${id}`] = (Number(this.ctx.flags[`seen.${id}`]) || 0) + 1;
      return this.current = this.run();
    }
    next() {
      if (this.pending) return this.current;
      return this.current = this.run();
    }
    choose(index) {
      const opts = this.pending;
      if (!opts) return this.current;
      const opt = opts[index];
      if (!opt) return this.current;
      this.pending = null;
      applySets(opt.set, this.ctx.flags);
      this.jump(opt.to);
      return this.current = this.run();
    }
    /** 바로 끝까지 넘김. 선택지가 나오면 거기서 멈춘다. */
    skip() {
      let s2 = this.current;
      for (let i = 0; i < 500 && s2.kind === "say"; i++) s2 = this.next();
      return s2;
    }
    fill(text2) {
      const f = this.ctx.flags;
      return text2.replace(/\{([\w.가-힣]+)\}/g, (_, k) => k === "name" ? this.ctx.speakerName("agent") : String(f[k] ?? ""));
    }
    jump(to) {
      const top = this.stack[this.stack.length - 1];
      if (!top) return;
      if (to === "end") top.pc = top.script.nodes.length;
      else top.pc = top.script.labels[to] ?? top.script.nodes.length;
    }
    say(n) {
      const who = n.who === "agent" ? this.ctx.agent : n.who;
      return { kind: "say", who, name: this.ctx.speakerName(who), text: this.fill(n.text), mood: n.mood ?? "base" };
    }
    run() {
      const { flags, rng: rng2 } = this.ctx;
      for (let guard3 = 0; guard3 < 2e3; guard3++) {
        const top = this.stack[this.stack.length - 1];
        if (!top) {
          delete flags.d100;
          return { kind: "end" };
        }
        flags.d100 = Math.floor(rng2() * 100);
        const n = top.script.nodes[top.pc];
        if (!n) {
          this.stack.pop();
          continue;
        }
        top.pc++;
        switch (n.t) {
          case "say":
            return this.say(n);
          case "pick":
            return this.say(n.lines[Math.floor(rng2() * n.lines.length) % n.lines.length]);
          case "agentline": {
            const text2 = this.ctx.agentLine(this.ctx.agent, n.tag, rng2);
            if (!text2) break;
            const who = n.who === "agent" ? this.ctx.agent : n.who;
            return { kind: "say", who, name: this.ctx.speakerName(who), text: this.fill(text2), mood: "base" };
          }
          case "choice": {
            const visible = n.options.map((o, index) => ({ o, index })).filter(({ o }) => test(o.if, flags));
            if (!visible.length) break;
            this.pending = n.options;
            return { kind: "choice", options: visible.map(({ o, index }) => ({ text: this.fill(o.text), index })) };
          }
          case "set":
            applySet(n.key, n.op, n.expr, flags);
            break;
          case "goto":
            if (test(n.if, flags)) this.jump(n.to);
            break;
          case "emit":
            this.ctx.emit?.(n.event, this.fill(n.arg));
            break;
          case "call":
            this.stack.push({ script: this.lib.get(n.script), pc: 0 });
            break;
          case "end":
            this.stack = [];
            delete flags.d100;
            return { kind: "end" };
        }
      }
      throw new ScriptError("대본이 끝없이 돌아요");
    }
  };

  var pick4 = (list, rng2) => list?.length ? list[Math.floor(rng2() * list.length) % list.length] : void 0;
  function lookupLine(agent, tag2, flags, rng2) {
    if (!tag2.startsWith("el:")) return agentLine(agent, tag2, rng2);
    const key2 = tag2.slice(3);
    const favor = Math.max(-3, Math.min(3, Number(flags.favor) || 0));
    switch (key2) {
      case "board":
        return pick4(BOARD_BY_FAVOR[String(favor)], rng2);
      case "call": {
        const k = `calls.${agent}`;
        const n = Number(flags[k]) || 0;
        flags[k] = n + 1;
        return CALLS[agent][n % CALLS[agent].length];
      }
      case "favor_high":
        return pick4(FAVOR_LINES[agent].high, rng2);
      case "favor_low":
        return pick4(FAVOR_LINES[agent].low, rng2);
      case "announce":
        return announce2(Number(flags.depth) || 1, rng2);
      case "hint": {
        const list = TYPE_HINTS[flags.ftype || "lobby"];
        return favor >= 2 ? list?.[0] : pick4(list, rng2);
      }
      default:
        return pick4(ELEVATOR2[key2], rng2);
    }
  }

  function sanitize(raw, name) {
    let t = raw.replace(/<[^>]+>/g, "").trim();
    t = t.replace(new RegExp(`^\\s*(?:${name}|${name.slice(0, 2)})\\s*[:：]\\s*`), "");
    t = t.replace(/^[「"“'](.*)[」"”']$/s, "$1");
    t = t.replace(/\*[^*]*\*/g, "").replace(/[ \t]{2,}/g, " ").replace(/\n{2,}/g, "\n").trim();
    t = t.split(/\n\s*[^\s:]{1,8}\s*[:：]/)[0].trim();
    if (t.length > 160) {
      const cut = t.slice(0, 160);
      const end = Math.max(cut.lastIndexOf("."), cut.lastIndexOf("?"), cut.lastIndexOf("!"), cut.lastIndexOf("…"));
      t = end > 40 ? cut.slice(0, end + 1) : cut + "…";
    }
    return t;
  }

          out = { text: offlineReply(npc, clean, agent, this.opts.rng), source: "offline", error: e instanceof Error ? e.message : String(e) };
        } finally {

      }

    }
    /**
     * 주인공 공격 판정이 행위자에 닿았나 (필드에는 전투 몸이 아닌 행위자도 맞는다).
     * 몹: 쓰러졌다 일어나 도망(겁) 또는 맞섬, 소란 +15, 범죄로 경찰·경비 출동. 이름 인물: 화내고 사건으로 알림.
     */
    hitBoxes(scene, boxes, dir) {
      const out = [];
      for (const a of this.inScene(scene)) {
        for (const b of boxes) {
          if (a.hitBy === b.attackId) continue;
          if (a.x + 30 < b.x0 || a.x - 30 > b.x1 || Math.abs(a.y - b.y) > (b.dy ?? 18) + 14 || b.z0 > 200) continue;
          a.hitBy = b.attackId;
          out.push(a);
          this.struck(a, dir);
          break;
        }
      }
      if (out.length) this.crime(scene, out[0].x, 15 * out.length);
      return out;
    }
    /** 행위자 하나를 맞힘 (에일린 코어 표적 경로). 같은 틱 여러 번은 hitBy로 거른다 */
    hitActor(id, dir, attackId) {
      const a = this.actors.find((q) => q.def.id === id);
      if (!a || a.act === "gone" || a.act === "door" || a.hitBy === attackId) return false;
      a.hitBy = attackId;
      this.struck(a, dir);
      this.crime(a.scene, a.x, 15);
      return true;
    }
    struck(a, dir) {
      a.x += dir * 36;
      a.face = -dir;
      a.goal = null;
      a.route = [];
      const down2 = a.def.kind === "mob" && this.rng() < 0.5;
      this.events.push({ kind: "hurt", scene: a.scene, who: a.def.name, named: a.def.kind === "named", x: a.x, down: down2 });
      if (a.def.kind === "named") {
        a.act = "argue";
        a.doing = "맞고 화남";
        a.wait = 3;
        this.setClip(a, this.canClip(a, "angry") ? "angry" : "idle");
        this.iconOn(a, "#", 3);
        return;
      }
      a.act = "down";
      a.doing = down2 ? "쓰러짐" : "휘청";
      a.wait = down2 ? 1.8 : 0.6;
      this.setClip(a, down2 && this.canClip(a, "down") ? "down" : "idle");
      this.iconOn(a, "!", 1.5);
    }

      const s2 = this.graph.scene(scene);

    /** 생활 시간을 dt초 진행. player = 보이는 장면과 주인공 발, hour = 필드 시(모르면 null) */
    step(dt, player) {
      this.time += dt;
      this.hour = player?.hour ?? null;
      this.player = player ? { scene: player.scene, x: player.x, y: player.y ?? 0 } : null;
      for (const [s2, u] of this.uproar) {
        u.v = Math.max(0, u.v - (player?.scene === s2 ? 1.5 : 5) * dt);
        if (u.v <= 0) this.uproar.delete(s2);
      }
      for (const a of [...this.actors]) this.stepActor(a, dt);
      for (let i = this.actors.length - 1; i >= 0; i--) if (this.actors[i].act === "gone") this.actors.splice(i, 1);
      for (const r of this.spawns) this.stepSpawn(r, dt);
      for (const p of this.posts) this.stepPost(p);
    }

      if (a.clip !== clip) {

    stepActor(a, dt) {
      if (a.iconT > 0 && (a.iconT -= dt) <= 0) a.icon = void 0;
      if (a.sayT > 0 && (a.sayT -= dt) <= 0) a.say = void 0;
      if (a.noticeCd > 0) a.noticeCd -= dt;
      if (a.needs) decayNeeds(a.needs, dt, TEMPER[a.def.name], onDuty(this.hour, TEMPER[a.def.name]));
      if (a.def.kind === "mob") this.mobThink(a, dt);
      if (a.def.kind === "named") this.namedNotice(a);
      switch (a.act) {
        case "door": {
          a.wait -= dt;
          if (a.wait > 0) return;
          const hop = a.route.shift();
          const at2 = this.graph.doorSpot(hop.to, hop.enter) ?? this.randomSpot(hop.to);
          a.scene = hop.to;
          a.x = at2.x;
          a.y = at2.y;
          return this.nextLeg(a);
        }
        case "walk":
          return this.move(a, dt);
        case "do":
          if (a.task && a.needs) a.needs[a.task.need] = Math.min(100, a.needs[a.task.need] + a.task.fill * dt);
          a.wait -= dt;
          if (a.wait <= 0 || a.task && a.needs && a.needs[a.task.need] >= 100) {
            a.last = a.task?.id;
            a.task = void 0;
            this.idle(a, 1, 3);
          }
          return;
        case "notice":
        case "watch":
        case "argue":
          a.wait -= dt;
          if (a.wait <= 0) this.resume(a);
          return;
        case "down":
          a.wait -= dt;
          if (a.wait > 0) return;
          if (a.def.kind === "mob" && a.temper && !a.post) {
            if (a.temper.courage > 0.75 && ARCHETYPES[a.def.archetype ?? ""]?.role === "thug") {
              a.act = "argue";
              a.doing = "맞섬";
              a.wait = 3;
              this.iconOn(a, "#", 3);
              return;
            }
            return this.flee(a, this.player?.x ?? a.x);
          }
          return this.resume(a);
        case "post":
          return this.postStand(a, dt);
        case "idle":
          a.wait -= dt;
          if (a.wait <= 0) this.decide(a);
          return;
      }
    }
    move(a, dt) {
      if (!a.goal) return this.idle(a);
      if (a.plan === "chase") {
        const p = this.player && this.player.scene === a.scene ? this.player : null;
        const t = a.target ? this.actors.find((q) => q.def.id === a.target) : null;
        if (t && t.scene === a.scene) a.goal = { x: t.x - Math.sign(t.x - a.x) * 70, y: t.y };
        else if (p) a.goal = { x: p.x - Math.sign(p.x - a.x) * 110, y: p.y || a.goal.y };
      }
      const dx = a.goal.x - a.x, dy = a.goal.y - a.y;
      const d3 = Math.hypot(dx, dy);
      const step2 = (a.def.speed ?? WALK) * a.run * dt;
      if (Math.abs(dx) > 1) a.face = dx > 0 ? 1 : -1;
      this.setClip(a, a.run > 1.2 && this.canClip(a, "run") ? "run" : "walk");
      if (d3 <= step2) {
        a.x = a.goal.x;
        a.y = a.goal.y;
        a.goal = null;
        return this.arrived(a);
      }
      a.x += dx / d3 * step2;
      a.y += dy / d3 * step2;
    }
    idle(a, min = 2, max = 7) {
      a.act = "idle";
      a.doing = "대기";

    resume(a) {
      if (a.post) return this.goPost(a);
      if (a.def.brain === "transit" && a.exitAt) return this.walkTo(a, a.exitAt, "지나가는 중", "exit");
      this.idle(a, 0.5, 2);
    }
    walkTo(a, p, doing, plan = "none") {
      a.act = "walk";
      a.goal = p;
      a.doing = doing;
      a.plan = plan;
    }
    arrived(a) {
      const hop = a.route[0];
      if (hop && hop.from === a.scene) {
        a.act = "door";
        a.doing = `${this.graph.scene(hop.to)?.name ?? hop.to}(으)로 이동 중`;
        a.wait = Math.max(1, hop.sec / 6);
        return;
      }
      switch (a.plan) {
        case "do":
          return this.startTask(a);
        case "post":
          a.act = "post";
          a.doing = `근무 중 (${a.post?.archetype ?? ""})`;
          a.wait = range2(this.rng, 4, 9);
          this.setClip(a, this.canClip(a, "work") ? "work" : "idle");

        case "patrol":
        case "roam-post":
          return this.goPost(a);
        case "exit":
          a.act = "gone";
          return;
        case "steal":
          return this.doSteal(a);
        case "harass":
          return this.doHarass(a);
        case "chase":
          a.act = "watch";
          a.doing = "막아섬";
          a.wait = 3;
          this.iconOn(a, "!", 2);
          return;
      }
      if (a.def.brain === "transit" && a.exitAt) {
        a.act = "gone";
        return;

    }
    nextLeg(a) {
      const hop = a.route[0];
      if (hop) {

        if (d3) return this.walkTo(a, d3, a.doing, a.plan);

      if (a.plan === "do" && a.task) return this.walkTo(a, this.randomSpot(a.scene), a.task.label + " 하러", "do");
      if (a.plan === "roam-post") return this.goPost(a);
      this.walkTo(a, this.randomSpot(a.scene), "자리 잡는 중");
    }

    namedNotice(a) {
      const p = this.player;
      if (!p || p.scene !== a.scene || a.noticeCd > 0) return;
      if (a.act !== "idle" && a.act !== "walk") return;
      if (a.plan === "do" || a.route.length) return;

      a.noticeCd = range2(this.rng, 20, 35);
      a.act = "notice";
      a.doing = "주인공을 봄";
      a.goal = null;
      a.face = p.x > a.x ? 1 : -1;
      a.wait = 2.4;
      this.setClip(a, this.canClip(a, "greet") ? "greet" : this.canClip(a, "bow") ? "bow" : "idle");

    }
    decide(a) {
      if (a.post) return this.goPost(a);
      if (a.def.brain === "transit") return this.idle(a, 1, 3);
      const roam = (a.def.roam ?? [a.def.home]).filter((s2) => this.graph.scene(s2));
      if (a.def.brain === "needs" && a.needs) {
        const c = choose({
          name: a.def.name,
          needs: a.needs,
          scene: a.scene,
          roam,
          graph: this.graph,
          has: (clip) => this.canClip(a, clip),
          duty: onDuty(this.hour, TEMPER[a.def.name]),
          partnerIn: (s2) => this.actors.some((q) => q !== a && q.def.kind === "named" && q.scene === s2 && (q.act === "idle" || q.act === "walk") && !q.route.length),
          rand: this.rng,
          last: a.last
        });
        if (c) {
          a.task = c.act;
          if (c.scene === a.scene) return this.walkTo(a, this.taskSpot(a, c.act), `${c.act.label} 하러`, "do");
          const path = this.graph.route(a.scene, c.scene);
          const d3 = path && this.graph.doorSpot(a.scene, path[0].exit);
          if (path && d3) {
            a.route = path;
            return this.walkTo(a, d3, `${c.act.label} 하러 ${this.graph.scene(c.scene)?.name ?? c.scene}(으)로`, "do");
          }
          a.task = void 0;
        }
      }
      const away = a.scene !== a.def.home;
      const r = this.rng();
      let dest = null;
      if (away && r < 0.45) dest = a.def.home;
      else if (!away && roam.length > 1 && r < 0.18) dest = pick6(this.rng, roam.filter((s2) => s2 !== a.scene));
      if (dest) {
        const path = this.graph.route(a.scene, dest);
        const d3 = path && this.graph.doorSpot(a.scene, path[0].exit);
        if (path && d3) {
          a.route = path;
          return this.walkTo(a, d3, `${this.graph.scene(dest)?.name ?? dest}(으)로 가는 중`);
        }
      }
      this.walkTo(a, this.randomSpot(a.scene), "서성이는 중");

    /** 수다는 상대 옆으로, 나머지는 장면 안 빈 곳 */

        if (!q) {

    // ── 몹 판단 (8-1장: 사건은 NPC 판단에서) ──
    mobThink(a, dt) {

      if (!t) return;

      const u = this.uproar.get(a.scene);

        if (t.courage < 0.25 + 0.15 * lvl && a.run < 1.5) return this.flee(a, u.x);
        if (t.curious > 0.6 && a.act !== "watch" && this.rng() < 0.4) {

          a.doing = "구경";
          a.goal = null;
          a.face = u.x > a.x ? 1 : -1;
          a.wait = range2(this.rng, 2, 5);
          this.setClip(a, "idle");
          this.iconOn(a, "!", 1.5);
          return;

      }
      if (a.act !== "walk" || a.plan !== "exit") return;
      const p = this.player;
      if (p && p.scene === a.scene && Math.abs(p.x - a.x) < 220 && t.curious > 0.7 && this.rng() < 0.25) {
        a.act = "watch";
        a.doing = "주인공을 힐끔";

        if (v) {
          a.target = v.def.id;
          return this.walkTo(a, { x: v.x - Math.sign(v.x - a.x) * 80, y: v.y }, "시비 걸러 감", "harass");

    incidents(scene) {

    nearMob(a, dist, ok) {

    gateFar(a, fromX) {
