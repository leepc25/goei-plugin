      if (r.talk && this.io.talking()) return false;
      return r.left <= 0;
    }

      if (this.done) return;
      const ms = dt * 1e3;
      this.elapsed += ms;
      for (let guard3 = 0; guard3 < 64; guard3++) {
        if (!this.cur) {
          const s2 = this.def.steps[this.i++];
          if (!s2) {

    if (!loop) return cut.frames - 1;

  function sanitize(raw, name) {
    let t = raw.replace(/<[^>]+>/g, "").trim();
    t = t.replace(new RegExp(`^\\s*(?:${name}|${name.slice(0, 2)})\\s*[:：]\\s*`), "");
    t = t.replace(/^[「"“'](.*)[」"”']$/s, "$1");
    t = t.replace(/\*[^*]*\*/g, "").replace(/[ \t]{2,}/g, " ").replace(/\n{2,}/g, "\n").trim();
    t = t.split(/\n\s*[^\s:]{1,8}\s*[:：]/)[0].trim();
    if (t.length > 160) {
      const cut = t.slice(0, 160);
      const end = Math.max(cut.lastIndexOf("."), cut.lastIndexOf("?"), cut.lastIndexOf("!"), cut.lastIndexOf("…"));
      t = end > 40 ? cut.slice(0, end + 1) : cut + "…";
    }
    return t;
  }
