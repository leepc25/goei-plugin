    /** 원래 E가 하던 일 (대화창의 「거래」「인사」 같은 버튼이 부른다) */
    interactWith(best) {
      const p = this.player;
      for (const h of this.interactHooks) h(best);
      if (best && this.stage?.onInteract) {
        p.play("pickup", this);
        this.stage.onInteract(this.stageApi, this.floor, best);
      }
    }
    useItem(slot) {
      const p = this.player;
      if (p.move || p.airborne) return;
      if (this.itemUse) {
        const r = this.itemUse(slot, this.stageApi);
        if (r) {
          p.play(r.anim, this);
          this.emit({ type: "item", x: p.x });
        }
        return;
      }
      const it = this.items[slot - 1];
      if (!it || it.n <= 0) return;
      if (it.id === "bandage") {
        if (p.hp >= p.hpMax) return this.say("다친 곳이 없다", 1);
        p.hp = Math.min(p.hpMax, p.hp + 60);
        p.play("bandage", this);
        this.emit({ type: "heal", x: p.x, id: it.id });
      } else if (it.id === "ammo") {
        p.reserve += 24;
        p.play("pickup", this);
        this.emit({ type: "item", x: p.x, id: it.id });
        this.say("탄약 +24", 1);
      }
      it.n--;
    }
    /** 적 모듈 몸(Fighter가 아님)이 요원 몸 안으로 파고들지 않게 최소 간격만큼 적을 밀어낸다 (검수 5-4).
     *  잡기형(선호 거리가 아주 가까운 종)·잡는 중·쓰러짐·공중은 빼고, 적은 자기가 있던 쪽으로 밀린다. */
    separateFromPlayer() {
      const p = this.player;
      if (!p.alive || p.z > 0) return;
      for (const b of this.bodies) {
        const e = b;
        if (!e.alive || b.team === "player" || b instanceof Fighter || !e.def || e.z > 0) continue;
        if (e.state && /^(dead|down|getup|grabbing|air|bound|hidden|dormant|spawn)$/.test(e.state)) continue;
        if (e.kind === "toucher" || e.def.keep && e.def.keep[1] < 30) continue;
        const dy = Math.abs(e.y - p.y);
        if (dy > 85) continue;
        const min = (84 + e.def.w * 0.35) * (dy > 45 ? 0.8 : 1);
        const dx = e.x - p.x;
        if (Math.abs(dx) >= min) continue;
        const s2 = dx === 0 ? -p.dir : Math.sign(dx);
        e.x += s2 * Math.min(16, min - Math.abs(dx));
      }
    }
    /** 적 모듈 몸끼리: 한 몸처럼 포개지지 않게 옆(x)으로 벌린다 (검수 16-3) */
    separateEnemies() {
      const skip = (e) => !e.alive || e.team === "player" || e.kind.startsWith("proj") || e instanceof Fighter || !e.def || e.z > 0 || !!e.state && /^(dead|down|getup|grabbing|air|bound|hidden|spawn)$/.test(e.state);
      const list = this.bodies.filter((e) => !skip(e));
      for (let i = 0; i < list.length; i++)
        for (let j = i + 1; j < list.length; j++) {
          const a = list[i], b = list[j];
          const dy = Math.abs(a.y - b.y);
          if (dy > 95) continue;
          const min = (70 + (a.def.w + b.def.w) * 0.5) * (dy > 50 ? 0.8 : 1);
          const dx = b.x - a.x;
          if (Math.abs(dx) >= min) continue;
          const s2 = dx === 0 ? a.id < b.id ? 1 : -1 : Math.sign(dx);
          const push = Math.min(10, (min - Math.abs(dx)) / 2);
          a.x -= s2 * push;
          b.x += s2 * push;
        }
    }
    /** 겹친 몸을 옆으로 밀어낸다 (벨트스크롤에서 서로 뚫고 지나가지 않게) */
    separate() {
      this.separateFromPlayer();
      this.separateEnemies();
      const list = [this.player, ...this.bodies].filter((b) => b.alive && b.kind !== "bullet" && b.z <= 0);
      for (let i = 0; i < list.length; i++)
        for (let j = i + 1; j < list.length; j++) {
          const a = list[i];
          const b = list[j];
          if (!(a instanceof Fighter) || !(b instanceof Fighter)) continue;
          if (a.phase === "held" || b.phase === "held" || a.phase === "down" || b.phase === "down" || a.phase === "dead" || b.phase === "dead") continue;
          if (a.invuln > 0 || b.invuln > 0) continue;
          if (Math.abs(a.y - b.y) > 30) continue;
          const min = (a.width + b.width) * 0.8;
          const dx = b.x - a.x;
          if (Math.abs(dx) < min) {
            const push = (min - Math.abs(dx)) / 2;
            const s2 = dx === 0 ? 1 : Math.sign(dx);
            if (a.team === "player" && a instanceof Player && a.move) b.x += s2 * push * 2;
            else {
              a.x -= s2 * push;
              b.x += s2 * push;
            }
          }
        }
    }
    resolveHits() {
      const all = [this.player, ...this.bodies].filter((b) => b.alive);
      const frames = new Map(all.map((b) => [b, b.frame()]));
      for (const a of all) {
        const fa = frames.get(a);
        if (!fa.hit.length) continue;
        for (const h of fa.hit)
          for (const t of all) {
            if (t === a || t.team === a.team || t.team === "neutral" || !t.alive) continue;
            const ft = frames.get(t);
            if (!ft.hurt.some((hb) => overlap(h, hb))) continue;
            const key2 = `${a.id}:${h.attackId}:${t.id}`;
            if (this.hitMemo.has(key2)) continue;
            this.hitMemo.set(key2, this.tick);
            this.applyHit(a, t, h);
            if (a.kind === "bullet") a.alive = false;
            if (!a.alive) break;
          }
      }
      if (this.hitMemo.size > 400) {
        for (const [k, v] of this.hitMemo) if (this.tick - v > 120) this.hitMemo.delete(k);
      }
    }
    applyHit(a, t, h) {
      if (a.kind === "bullet") h = progressionHit(h, this.player.progression, "ranged");
      const side = Math.sign(t.x - a.x);
      if (side && side !== h.dir) h = { ...h, dir: side };
      const r = t.applyHit(h, this);
      const at2 = { x: (Math.max(h.x0, t.x - 30) + Math.min(h.x1, t.x + 30)) / 2, y: t.y, z: (h.z0 + h.z1) / 2 };
      if (r.kind === "immune") return;
      const stop = h.stop ?? STOP_DEFAULT[h.react];
      if (r.kind === "guard") {
        a.stop = Math.max(a.stop, 3);
        t.stop = Math.max(t.stop, 3);
        this.fx("spark", at2.x, at2.y, at2.z);
        return;
      }
      a.stop = Math.max(a.stop, stop);
      t.stop = Math.max(t.stop, stop);
      const strength = h.strength ?? (h.react === "down" || h.react === "launch" ? 3 : h.react === "stagger" ? 2 : 1);
      this.fx(strength >= 3 ? "spark-big" : "spark", at2.x, at2.y, at2.z);
      if (this.hitFx) {
        if (strength >= 2) this.fx("ring", at2.x, at2.y, at2.z, String(strength));
        if (strength >= 3) this.shake(6, 3);
      }
      if (h.noise) this.emit({ type: "custom", name: "noise", data: h.noise });
      const p = this.player;
      if (a === p || a.kind === "bullet") {
        p.onConnect(this, !!r.killed);
        if (r.killed) this.kills++;
      }
      const mk = a.kind === "bullet" ? "bullet" : a !== p ? "enemy" : { airkick: "kick", lowkick: "low", dash: "dash", blast: "blast", groundblast: "blast", grabblast: "blast", super: "super", grabknee: "kick" }[p.moveKey] ?? "punch";
      this.emit({ type: "hit", strength, x: at2.x, y: at2.y, z: at2.z, dmg: h.dmg, react: h.react, target: t.kind, combo: a === p ? p.combo : 0, kind: mk });
      if (r.killed && t.team === "enemy" && !(t instanceof Fighter)) this.emit({ type: "enemyDie", x: t.x, kind: t.kind });
      if (a === p && p.moveKey === "grabknee") p.keepHeld();
    }
    stepEncounters() {
      const encs = this.floor.encounters;
      const p = this.player;
      if (this.encActive < 0) {
        const e2 = encs[this.encIndex];
        if (e2 && p.x >= e2.triggerX) {
          this.encActive = this.encIndex;
          this.goT = 0;
          this.encWave = 0;
          this.bounds = [e2.lockFrom, e2.lockTo];
          this.spawnWave();
          this.emit({ type: "combat:start", index: this.encActive });
          if (e2.waves.some((w) => w.some((s2) => s2.boss))) this.emit({ type: "combat:boss", index: this.encActive });
        }
        return;
      }
      const e = encs[this.encActive];
      this.encBodies = this.encBodies.filter((b) => b.alive && b.hp > 0 && this.threat(b) && !(b instanceof Fighter && (b.phase === "dead" || b.phase === "down" && b.hp <= 0)));
      if (this.encBodies.length) return;
      this.encWave++;
      if (this.encWave < e.waves.length) {
        this.spawnWave();
        return;
      }
      this.emit({ type: "combat:clear", index: this.encActive });
      this.encActive = -1;
      this.encIndex++;
      this.bounds = [0, this.floor.length];
      this.goT = 90;
    }
    spawnWave() {
      const e = this.floor.encounters[this.encActive];
      this.encBodies = e.waves[this.encWave].map((s2) => this.spawn(s2));
    }
    camera() {
      const p = this.player;
      const [b0, b1] = this.bounds;
      const target = p.x - SCREEN_W / 2 + p.dir * 90;
      let cx = this.camX + (target - this.camX) * 0.14;
      cx = Math.max(b0, Math.min(b1 - SCREEN_W, cx));
      cx = Math.max(0, Math.min(this.floor.length - SCREEN_W, cx));
      this.camX = cx;
      if (this.encActive >= 0) p.x = Math.max(cx + 90, Math.min(cx + SCREEN_W - 90, p.x));
    }
    /** 보스 체력 막대용 */
    boss() {
      const b = this.bodies.find((x) => x.alive && x.boss);
      return b ? { label: b.label ?? b.kind, hp: b.hp, hpMax: b.hpMax } : null;
    }
  };

  var Game = class {
    constructor(box2, art5, mods) {

      }

      this.input.attach();
      this.canvas.addEventListener("pointerdown", (e) => this.onPointer(e));
      this.scene = this.idleScene("title");
      this.toTitle();
    }
    canvas;
    ctx;
    input = new Input();
    world = null;
    scene;
    paused = false;
    ownCanvas;
    renderer;
    raf = 0;
    acc = 0;
    last = 0;
    menuIndex = 0;
    run = null;
    runStage = null;
    lastResult = null;

    wasPaused = false;
    menuItems = [];
    titleItems = [];
    /** 꼬리표 시작은 주 루프가 돈 뒤에 (start()에서) */
    booted = false;
    start() {
      if (!this.booted) {
        this.booted = true;
        queueMicrotask(() => this.mods.boot?.(this));
      }
      const frame2 = (now) => {
        if (this.dead) return;
        this.raf = requestAnimationFrame(frame2);
        if (!this.last) this.last = now;
        let dt = (now - this.last) / 1e3;
        this.last = now;
        if (dt > 0.25) dt = 0.25;
        this.mods.frame?.(dt);
        this.acc += dt;
        const T3 = 1 / TICK_HZ;
        while (this.acc >= T3) {
          this.acc -= T3;
          this.tickOnce();
        }
        const paused = this.paused || !!this.mods.paused?.();
        if (paused !== this.wasPaused) {
          this.wasPaused = paused;
          this.mods.onPause?.(paused);
        }
        if (!this.mods.covered?.()) {
          const w = this.world;
          const H7 = this.inHub ? view.worldH : SCREEN_H;
          view.zoom = this.canvas.height / H7 * Math.max(1, charSize.k);
          if (w) view.lane = w.lane;
          const vw = this.canvas.width / view.zoom;
          view.crop = 0;
          const frozen = paused || !!this.mods.blocking?.() || !!this.mods.hq?.isOpen?.();
          if (vw < SCREEN_W && w) {
            const a = frozen ? 1 : this.acc / T3, cam = Math.round(w.prevCamX + (w.camX - w.prevCamX) * a), px = (w.player.px ?? w.player.x) + (w.player.x - (w.player.px ?? w.player.x)) * a, raw = Math.max(0, Math.min(SCREEN_W - vw, px - cam - vw / 2));
            view.crop = Math.round((cam + raw) * view.zoom) / view.zoom - cam;
          }
          view.cropY = 0;
          const vh = this.canvas.height / view.zoom;
          if (vh < H7 - 0.5 && w) {
            const a = frozen ? 1 : this.acc / T3, p = w.player;
            const py = (p.py ?? p.y) + (p.y - (p.py ?? p.y)) * a;
            view.cropY = Math.round(Math.max(0, Math.min(H7 - vh, py - vh * 0.7)) * view.zoom) / view.zoom;
          }
          this.ctx.save();
          this.ctx.scale(view.zoom, view.zoom);
          this.ctx.translate(-view.crop, -view.cropY);
          this.scene.draw(paused ? 1 : this.acc / T3);
          this.ctx.restore();
          this.mods.overlay?.(this.ctx, this.world, dt);
        }
      };
      this.raf = requestAnimationFrame(frame2);
    }
    /** 시험용: 틱을 n번 바로 돌린다 */
    advance(n) {
      for (let i = 0; i < n; i++) this.tickOnce();
      this.scene.draw(1);
    }
    tickOnce() {
      const p = this.input.sample();
      if (typeof document !== "undefined" && document.hidden) return;
      if (this.mods.paused?.() && this.scene.name === "battle") return;
      this.scene.tick(p);
    }
    destroy() {
      this.checkpoint();
      this.dead = true;
      cancelAnimationFrame(this.raf);
      this.world = null;
      this.run = null;
      this.input.detach();
      if (this.ownCanvas) this.canvas.remove();
      if (this.hqRoot) {
        this.mods.hq?.unmount();
        this.hqRoot.remove();
      }

    }
    /* ───────── 제목·본부 ───────── */
    toTitle() {

      this.world = null;
      this.inHub = false;
      this.paused = false;
      if (this.hqRoot) this.hqRoot.hidden = true;
      if (this.mods.openTitle) {
        this.scene = this.idleScene("title");
        this.mods.openTitle();
      } else this.scene = this.titleScene();

    }
    /** 새로 시작: 본부가 있으면 본부로, 없으면 첫 층 모듈로 바로 출동 */
    start_new(stageId) {
      if (this.mods.hq && !stageId) return this.showHq();
      const stage = this.mods.stages.find((s2) => s2.id === stageId) ?? this.mods.stages[0];
      this.deploy({ seed: Math.floor(Math.random() * 1e9), agent: "eileen", stageId: stage.id });
    }
    dead = false;
    /** 본부를 걷는 중인가 (엔진 벨트 장면) */
    inHub = false;
    showHq() {
      const hq2 = this.mods.hq;
      this.world = null;
      this.scene = this.idleScene("hq");
      if (!hq2.hub) this.canvas.style.visibility = "hidden";
      if (!this.hqRoot) {
        this.hqRoot = document.createElement("div");
        this.hqRoot.className = "goei-engine-hq";
        (hq2.hub && this.mods.hqLayer?.() || this.box).appendChild(this.hqRoot);

      if (hq2.hub) {

        this.hqRoot.hidden = !hq2.isOpen?.();
        this.hqRoot.classList.add("goei-engine-hq-panel");
        this.canvas.style.visibility = "";
        this.inHub = true;
        const w = new World(run, this.art.clips, stage, this.mods.enemyFactory);
        for (const l of this.mods.listeners ?? []) w.on(l);
        this.mods.onWorld?.(w, run);
        w.enterFloor(stage.firstFloor(run));
        this.world = w;
        this.runStage = stage;
        this.inHub = true;
        this.paused = false;
        this.scene = this.battleScene(w);
        this.mods.onPlay?.();
      }
      this.mods.onScene?.("hq");
    }
    idleScene(name) {
      return {
        name,
        tick: () => {
        },
        draw: () => {
          this.ctx.fillStyle = "#05080a";
          this.ctx.fillRect(0, 0, SCREEN_W, SCREEN_H);
          this.mods.drawIdle?.(name, this.ctx, performance.now() / 1e3);
        }
      };
    }
    /* ───────── 출동 ───────── */
    deploy(run, saved) {
      const stage = this.mods.stages.find((s2) => s2.id === run.stageId) ?? this.mods.stages[0];
      run = { ...run, stageId: stage.id };
      this.inHub = false;
      this.run = run;
      this.runStage = stage;
      if (this.hqRoot) this.hqRoot.hidden = true;
      this.canvas.style.visibility = "";
      const w = new World(run, this.art.clips, stage, this.mods.enemyFactory);
      if (this.mods.itemUse) w.itemUse = this.mods.itemUse;
      for (const l of this.mods.listeners ?? []) w.on(l);
      this.mods.onWorld?.(w, run);
      w.enterFloor(saved && stage.restore ? stage.restore(saved.stage) : stage.firstFloor(run));
      if (saved) w.restore(saved.world);
      this.world = w;
      this.paused = false;
      this.scene = this.battleScene(w);
      this.mods.onPlay?.();

      this.checkpoint(true);
    }
    /** 전투 도중에는 직전 안전 지점을 유지한다. HP·탄·드랍도 함께 되돌린다. */
    checkpoint(force = false) {
      const w = this.world;
      if (!w || this.inHub || !this.run || !w.canCheckpoint() || !force && this.mods.blocking?.()) return;
      const stage = this.runStage?.snapshot?.();
      if (!stage) return;
      this.mods.saveCheckpoint?.({ version: 1, run: this.run, stage, world: w.snapshot(), savedAt: Date.now() });
    }
    async resume(saved) {
      const stageId = saved.run.stageId ?? this.mods.stages[0]?.id;
      if (saved.version !== 1 || !this.mods.stages.some((s2) => s2.id === stageId && s2.restore)) throw new Error("지원하지 않는 출동 저장");
      if (this.mods.hq) {
        this.showHq();
        await this.mods.hq.ready?.();
      }
      this.deploy({ ...saved.run, stageId }, saved);
      this.world?.say("출동 복원 완료. 전투 중 중단한 경우 마지막 안전 지점부터 이어집니다.", 4);
    }
    /** 출동 끝내기 (철수 메뉴 등에서도 부른다) */
    endRun(how) {
      const w = this.world;
      if (!w || this.inHub) return;
      const result = {
        how,
        floors: Number(w.notes["stage.visitedCount"] ?? w.floors),
        kills: w.kills,
        maxCombo: w.player.maxCombo,
        timeSec: Math.round(w.tick / TICK_HZ),
        hpLeft: Math.max(0, Math.round(w.player.hp)),
        notes: w.notes
      };
      w.emit({ type: "run:end", how });
      this.lastResult = result;
      this.world = null;
      this.paused = false;
      if (this.mods.hq && this.run) {
        const run = this.run;
        this.mods.hq.onReturn(this.mods.toHqResult?.(result, w) ?? result, run);
        this.mods.saveCheckpoint?.(null);
        const back = () => this.showHq();
        if (this.mods.afterRun) {
          this.scene = this.idleScene("cutscene");
          this.mods.afterRun(how, back);
        } else back();
        return;
      }
      this.mods.saveCheckpoint?.(null);
      this.scene = this.resultScene(result);
      this.mods.onScene?.("result");
    }
    battleScene(w) {
      const ownMenu = !this.mods.paused;
      return {
        name: "battle",
        tick: (p) => {
          if (ownMenu && p.pressed.has("pause")) {
            this.paused = !this.paused;
            this.menuIndex = 0;
          }
          if (this.paused) return this.pauseMenu(p, w);
          if (this.inHub && this.hqRoot) {
            const open = !!this.mods.hq?.isOpen?.();
            this.hqRoot.hidden = !open;
            if (open) return;
          }
          if (this.mods.blocking?.() || this.mods.frozen?.()) return;
          w.step(p);
          if (w.ended) {
            const how = w.ended;
            if (how === "next") {
              const next = this.runStage.nextFloor(this.run, w.floor, "next");
              if (next) {
                w.enterFloor(next);
                this.checkpoint(true);
              } else this.endRun("return");
            } else this.endRun(how);
          } else if (w.tick % 60 === 0) this.checkpoint();
        },

  var RETURN = /(돌아가|돌아가\s*있어|복귀해|원래\s*자리)/;
  function classify(text2) {
    const t = text2.replace(/\s+/g, " ");

    if (WAIT.test(t)) return "wait";
    if (RETURN.test(t)) return "return";
    if (FOLLOW.test(t) && !/\?|까요|할래|하겠/.test(t)) return "follow";
    return null;
  }

  var PC = { c: 0, d: 2, e: 4, f: 5, g: 7, a: 9, b: 11 };
  function noteToMidi(s2) {
    const m = /^([a-gA-G])([#b]?)(-?\d)$/.exec(s2);
    if (!m) throw new Error(`음 표기 오류: ${s2}`);
    const acc = m[2] === "#" ? 1 : m[2] === "b" ? -1 : 0;
    return 12 * (Number(m[3]) + 1) + PC[m[1].toLowerCase()] + acc;
  }
  function parseNotes(src) {
    const toks = [];
    for (const raw of src.split(/\s+/)) {
      if (!raw || raw === "|") continue;
      const m = /^(.*)\*(\d+)$/.exec(raw);
      if (m) {
        const n = Number(m[2]);
        if (m[1] === "." || m[1] === "-") for (let i = 0; i < n; i++) toks.push(m[1]);
        else {
          toks.push(m[1]);
          for (let i = 1; i < n; i++) toks.push("-");
        }
      } else toks.push(raw);
    }
    const evs = [];
    let cur = null;
    toks.forEach((t, i) => {
      if (t === "-") {
        if (cur) cur.len++;
        return;
      }
      cur = null;
      if (t === ".") return;
      let vel = 0.8;
      let body2 = t;
      if (body2.endsWith("!")) vel = 1, body2 = body2.slice(0, -1);
      else if (body2.endsWith("?")) vel = 0.5, body2 = body2.slice(0, -1);
      cur = { step: i, notes: body2.split("+").map(noteToMidi), len: 1, vel };
      evs.push(cur);
    });
    return { evs, steps: toks.length };
  }
  function parseDrums(src) {
    const s2 = src.replace(/[\s|]/g, "");
    const evs = [];
    for (let i = 0; i < s2.length; i++) if (s2[i] !== ".") evs.push({ step: i, kind: s2[i] });
    return { evs, steps: s2.length };
  }
  var QUAL = {
    "": [0, 4, 7],
    m: [0, 3, 7],
    "7": [0, 4, 7, 10],
    maj7: [0, 4, 7, 11],
    m7: [0, 3, 7, 10],
    dim: [0, 3, 6],
    m7b5: [0, 3, 6, 10],
    sus4: [0, 5, 7],
    aug: [0, 4, 8],
    "5": [0, 7],
    add9: [0, 4, 7, 14],
    madd9: [0, 3, 7, 14]
  };
  function chord(name) {
    const m = /^([A-G])([#b]?)(.*)$/.exec(name);
    if (!m || !(m[3] in QUAL)) throw new Error(`화음 표기 오류: ${name}`);
    const root = PC[m[1].toLowerCase()] + (m[2] === "#" ? 1 : m[2] === "b" ? -1 : 0);
    const tones = QUAL[m[3]].map((iv) => {
      let n = 48 + (root + iv) % 12;
      while (n < 52) n += 12;
      if (iv >= 12) n += 12;
      return n;
    });
    tones.sort((a, b) => a - b);
    return { root: (root + 12) % 12, tones };
  }
  var NAMES = ["c", "c#", "d", "d#", "e", "f", "f#", "g", "g#", "a", "a#", "b"];
  var tok = (notes) => notes.map((n) => `${NAMES[n % 12]}${Math.floor(n / 12) - 1}`).join("+");
  function comp(chords, rhythm, shift = 0) {
    return chords.split(/\s+/).filter(Boolean).map((c) => {
      const ns = chord(c).tones.map((n) => n + shift);
      return [...rhythm].map((ch) => ch === "x" ? tok(ns) : ch === "X" ? tok(ns) + "!" : ch).join(" ");
    }).join(" ");
  }
  function bass(chords, rhythm, oct = 2) {
    return chords.split(/\s+/).filter(Boolean).map((c) => {
      const { root } = chord(c);
      const r = 12 * (oct + 1) + root;
      const pick6 = (ch) => ch === "r" ? r : ch === "o" ? r + 12 : ch === "f" ? r + 7 : ch === "b" ? r + 10 : -1;
      return [...rhythm].map((ch) => pick6(ch) >= 0 ? tok([pick6(ch)]) : ch).join(" ");
    }).join(" ");
  }
  function arp(chords, shape2, shift = 12) {
    return chords.split(/\s+/).filter(Boolean).map((c) => {
      const ts = chord(c).tones;
      return [...shape2].map((ch) => {
        if (!/\d/.test(ch)) return ch;
        const i = Number(ch);
        return tok([ts[i % ts.length] + 12 * Math.floor(i / ts.length) + shift]);
      }).join(" ");
    }).join(" ");
  }
  function fm(o, dest, t, f, dur, vel, ratio, idx, idxDecay, a, d3, s2, r) {

  function compileSong(def) {
    const chans = {};
    for (const [name, c] of Object.entries(def.chans)) chans[name] = { def: c, at: /* @__PURE__ */ new Map() };
    let base2 = 0;
    for (const entry of def.order) {
      const m = /^([A-Za-z0-9_]+)([+-]\d+)?$/.exec(entry);
      if (!m || !def.pats[m[1]]) throw new Error(`곡 순서 오류: ${entry}`);
      const pat = def.pats[m[1]];
      const tr = Number(m[2] ?? 0);
      let len = 0;
      const parsed = {};
      for (const [name, src] of Object.entries(pat)) {
        const c = def.chans[name];
        if (!c) throw new Error(`없는 악기 줄: ${name}`);
        parsed[name] = c.inst === "drums" ? parseDrums(src) : parseNotes(src);
        len = Math.max(len, parsed[name].steps);
      }
      for (const [name, p] of Object.entries(parsed)) {
        for (const e of p.evs) {
          const ev = "notes" in e ? { ...e, notes: e.notes.map((n) => n + tr) } : e;
          const at2 = chans[name].at;
          const k = base2 + e.step;
          (at2.get(k) ?? at2.set(k, []).get(k)).push(ev);
        }
      }
      base2 += len;
    }
    return { def, steps: base2, chans };
  }

  function startAmbience(o, scene, fadeIn = 1.5) {
    const scope = new AudioScope(busOut(o, "amb"));
    o = scope.out;
    const ctx = o.ctx;
    const spec = AMBIENCE[scene];
    const out = gainNode(o, 0, o.amb);
    const verb = gainNode(o, 1, o.verb), room2 = gainNode(o, 1, o.room);
    o = { ...o, verb, room: room2 };

    out.gain.setValueAtTime(1e-4, t0);
    out.gain.linearRampToValueAtTime(1, t0 + fadeIn);
    const long = 86400;
    for (const l of spec.layers) {
      const g2 = gainNode(o, l.vol, out);
      if (l.kind === "hum") {
        for (const [k, v] of [[1, 1], [2, 0.5], [3, 0.3], [4, 0.15]]) osc(o, "sine", l.f * k, t0, long, gainNode(o, v, g2));
        const wob = gainNode(o, l.vol * 0.3);
        wob.connect(g2.gain);
        osc(o, "sine", 0.13, t0, long, wob);
      } else if (l.kind === "rumble") {
        noise2(o, t0, long, filt(o, "lowpass", l.f, 1.5, g2), true);
        const wob = gainNode(o, l.vol * 0.4);
        wob.connect(g2.gain);
        osc(o, "sine", 0.07, t0, long, wob);
      } else if (l.kind === "air") {
        noise2(o, t0, long, filt(o, "bandpass", l.f, 0.6, g2));
        const wob = gainNode(o, l.vol * 0.6);
        wob.connect(g2.gain);
        osc(o, "sine", 0.05, t0, long, wob);
      } else {
        osc(o, "sine", l.f, t0, long, g2);
      }
    }
    const timers = /* @__PURE__ */ new Set();
    const later = (f, ms) => {
      const id = window.setTimeout(() => {
        timers.delete(id);
        f();
      }, ms);
      timers.add(id);
    };
    let live = true;
    const fire = (e) => {
      if (!live) return;
      if (ctx.state !== "running") {
        later(() => fire(e), e.gap[0] * 1e3);
        return;
      }
      const vol = rnd2(e.vol[0], e.vol[1]);
      const pan = (Math.random() * 2 - 1) * (e.pan ?? 0);
      const pitch = e.pitch ? rnd2(e.pitch[0], e.pitch[1]) : 1;

      if (e.id === "typing") {
