  function fitText(c, text, width) {
    if (c.measureText(text).width <= width) return text;
    const chars2 = Array.from(text);
    while (chars2.length && c.measureText(chars2.join("") + "…").width > width) chars2.pop();
    return chars2.join("") + "…";
  }
  function wrapText(c, text, width, maxRows) {
    const rows = [];
    let line2 = "";
    const chars2 = Array.from(text.replace(/\s+/g, " "));
    for (let i = 0; i < chars2.length; i++) {
      if (line2 && c.measureText(line2 + chars2[i]).width > width) {
        rows.push(line2);
        if (rows.length === maxRows - 1) return [...rows, fitText(c, chars2.slice(i).join(""), width)];
        line2 = "";
      }
      line2 += chars2[i];
    }
    if (line2) rows.push(line2);
    return rows;
  }

  var SLOT_COUNT = 4;
  function createItemRun(agent, loadout) {
    const sig = SIGNATURES[agent] ?? SIGNATURES.jyaldi;
    const slots = [];
    for (let i = 0; i < SLOT_COUNT; i++) {
      const s = loadout?.[i];
      slots.push(s && ITEMS[s.id] && Number.isFinite(s.n) && s.n >= 1 ? { id: s.id, n: Math.min(ITEMS[s.id].max, Math.floor(s.n)) } : null);
    }
    return {
      agent: sig.agent,
      loadout: slots,
      mag: sig.mag,
      reserve: sig.reserve,
      reloadT: 0,
      shotCd: 0,
      held: null,
      carry: null,
      ward: 0,
      itemCd: 0
    };
  }
  function addItem(run, id, n = 1) {
    const def = ITEMS[id];
    if (!def || !Number.isSafeInteger(n) || n <= 0) return 0;
    let left = n;
    for (const s of run.loadout) {
      if (s && s.id === id && s.n < def.max) {
        const k = Math.min(left, def.max - s.n);
        s.n += k;
        left -= k;
        if (!left) return n;
      }
    }
    for (let i = 0; i < run.loadout.length && left; i++) {
      if (!run.loadout[i]) {
        const k = Math.min(left, def.max);
        run.loadout[i] = { id, n: k };
        left -= k;
      }
    }
    return n - left;
  }
  function useless(effect, c) {
    if (!c) return false;
    switch (effect.type) {
      case "heal":
        return c.hp >= c.max || !!c.noHeal;
      case "cure":
        if (effect.status === "bleed") return !c.bleed && c.hp >= c.max;
        if (effect.status === "taint") return !c.taint;
        if (effect.status === "headache") return !c.headache;
        if (effect.status === "mind") return !c.mind;
        return false;
      case "energy":
        return c.energy !== void 0 && c.energyMax !== void 0 && c.energy >= c.energyMax;
      case "gauge":
        return c.gauge !== void 0 && c.gauge >= 1;
      default:
        return false;
    }
  }
  function useItem(run, slot, ctx) {
    const s = run.loadout[slot];
    if (!s || s.n <= 0 || run.itemCd > 0) return null;
    const def = ITEMS[s.id];
    if (!def || !def.always && useless(def.effect, ctx)) return null;
    s.n--;
    if (s.n <= 0) run.loadout[slot] = null;
    run.itemCd = 0.35;
    if (def.effect.type === "ward") run.ward += def.effect.hits;
    return { id: def.id, effect: def.effect };
  }
  var pickupHealAmount = (max, amount = 1) => max * 0.08 * amount;
  var usesAmmo = (run) => (SIGNATURES[run.agent]?.ammo ?? "none") !== "none";
  function addAmmo(run, amount) {
    const sig = SIGNATURES[run.agent];
    if (!sig || sig.ammo === "none") return 0;
    const want = amount ?? AMMO_PICKUP[sig.ammo].amount;
    const k = Math.max(0, Math.min(want, sig.reserveMax - run.reserve));
    run.reserve += k;
    return k;
  }
  function fireSignature(run) {
    if (!usesAmmo(run)) return "free";
    if (run.reloadT > 0) return "reloading";
    if (run.shotCd > 0) return "cooldown";
    if (run.mag <= 0) {
      if (startReload(run)) return "reloading";
      return "dry";
    }
    run.mag--;
    run.shotCd = 0.16;
    if (run.mag === 0) startReload(run);
    return "fire";
  }
  function startReload(run) {
    const sig = SIGNATURES[run.agent];
    if (!sig || sig.ammo === "none" || run.reloadT > 0 || run.reserve <= 0 || run.mag >= sig.mag) return false;
    run.reloadT = sig.reload;
    return true;
  }
  function tickRun(run, dt) {
    run.itemCd = Math.max(0, run.itemCd - dt);
    run.shotCd = Math.max(0, run.shotCd - dt);
    if (run.reloadT > 0) {
      run.reloadT -= dt;
      if (run.reloadT <= 0) {
        run.reloadT = 0;
        const sig = SIGNATURES[run.agent];
        const k = Math.min(sig.mag - run.mag, run.reserve);
        run.mag += k;
        run.reserve -= k;
        return true;
      }
    }
    return false;
  }
  function equip(run, id, uses) {
    const def = WEAPONS[id];
    if (!def) return null;
    const prev = run.held;
    run.held = { id, uses: uses ?? def.uses };
    return prev;
  }
  function useWeapon(run, cls) {
    const h = run.held;
    if (!h) return null;
    const def = WEAPONS[h.id];
    if (def.cls !== cls) return null;
    h.uses--;
    const broke = h.uses <= 0;
    if (broke) run.held = null;
    return { def, broke, left: Math.max(0, h.uses) };
  }

  function rand(f) {
    let t = f.seed = f.seed + 1831565813 | 0;
    t = Math.imul(t ^ t >>> 15, t | 1);
    t ^= t + Math.imul(t ^ t >>> 7, t | 61);
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  }
  var pick = (f, a) => a[Math.floor(rand(f) * a.length)];
  function createField(seed = 1, depth = 1, bounds = { x0: 40, x1: 1240, y0: 480, y1: 680 }) {

  }

  function spawnPickup(field, kind, x, y, ref, amount = 1, uses) {
    const a = rand(field) * Math.PI * 2, s = (40 + rand(field) * 70) * (kind === "weapon" ? 0.6 : 1);
    const p = {
      id: field.nextId++,
      type: "pickup",
      kind,
      ref,
      amount,
      uses,
      x,
      y,
      z: 30,
      vx: Math.cos(a) * s,
      vy: Math.sin(a) * s * 0.4,
      vz: 260 + rand(field) * 120,
      age: 0
    };
    field.ents.push(p);
    return p;
  }
  function rollDrops(field, run, table, x, y) {
    const rows = DROP_TABLES[table];
    const out = [];
    if (!rows) return out;
    for (const r of rows) {
      let p = r.p;
      if (r.kind === "ammo" && usesAmmo(run) && run.mag + run.reserve < 8) p = Math.min(1, p + 0.25);
      if (rand(field) >= p) continue;
      const kind = r.kind;
      if (kind === "ammo" && !usesAmmo(run)) continue;
      const ref = r.pool ? pick(field, r.pool) : void 0;
      spawnPickup(field, kind, x, y, ref, 1);
      out.push({ t: "drop", kind, ref, x, y });
    }
    return out;
  }

  function burst(field, kind, x, y, r, life, slow) {

  }
  function slowAt(field, x, y) {
    let m = 1;
    for (const z of field.zones) if (z.kind === "smoke" && Math.hypot(x - z.x, (y - z.y) * 2) < z.r) m = Math.min(m, 1 - (z.slow ?? 0));
    return m;
  }
  var near = (a, b) => Math.hypot(a.x - b.x, (a.y - b.y) * 1.6);
  function hitsTarget(p, t, pad = 0) {
    return Math.abs(p.x - t.x) < t.w / 2 + pad && Math.abs(p.y - t.y) < 34 + pad && p.z >= (t.z ?? 0) - 10 && p.z <= (t.z ?? 0) + t.h;
  }
  function propBox(e, p, pad = 0) {
    const d3 = PROPS[e.prop];
    return Math.abs(p.x - e.x) < d3.w / 2 + pad && Math.abs(p.y - e.y) < 30 + pad && p.z <= d3.h + pad;
  }
  function damageProp(field, run, e, dmg, targets = []) {
    if (!field.ents.includes(e)) return [];
    const def = PROPS[e.prop];
    if (def.use === "throw") return [];
    e.hp -= dmg;
    e.flash = 0.12;
    if (e.hp > 0) return [{ t: "propHit", prop: e.prop, x: e.x, y: e.y }];
    field.ents.splice(field.ents.indexOf(e), 1);
    const ev = [{ t: "break", prop: e.prop, x: e.x, y: e.y }];
    if (def.loot) ev.push(...rollDrops(field, run, def.loot, e.x, e.y));
    if (def.burst) ev.push(...explode(field, run, def.burst, e.x, e.y, def.id, targets, def.knockdown));
    return ev;
  }
  function explode(field, run, b, x, y, source, targets, knockdown) {
    const ev = [burst(field, b.type, x, y, b.radius, b.time ?? 0.5, b.slow)];
    if (b.damage) {
      for (const t of targets)
        if (Math.hypot(t.x - x, (t.y - y) * 2) < b.radius)
          ev.push({ t: "hit", target: t.id, damage: b.damage, knockback: 180, knockdown, status: b.type === "shock" ? "stun" : void 0, stun: b.type === "shock" ? 1 : void 0, x: t.x, y: t.y, source });
      for (const e of field.ents.slice())
        if (e.type === "prop" && Math.hypot(e.x - x, (e.y - y) * 2) < b.radius && field.ents.includes(e))
          ev.push(...damageProp(field, run, e, b.damage, targets));
    }
    return ev;
  }
  function attack(field, run, actor, targets, melee = { damage: 12, reach: 110 }) {
    const ev = [];
    if (run.carry) {
      const def = PROPS[run.carry.prop];
      field.projs.push({
        id: field.nextId++,
        kind: "prop",
        ref: def.id,
        x: actor.x + actor.dir * 30,
        y: actor.y,
        z: actor.z + 120,
        vx: actor.dir * 760,
        vz: 120,
        life: 1.6,
        damage: def.damage,
        knockback: 220,
        knockdown: def.knockdown,
        hit: []
      });
      ev.push({ t: "throw", prop: def.id });
      run.carry = null;
      return { events: ev, handled: true };
    }
    const w = useWeapon(run, "melee");
    const reach = w ? w.def.reach : melee.reach, dmg = w ? w.def.damage : melee.damage;
    const box2 = { x: actor.x + actor.dir * reach * 0.5, y: actor.y, z: actor.z + 80 };
    if (w) {
      ev.push({ t: "swing", weapon: w.def.id, left: w.left });
      for (const t of targets)
        if (Math.abs(t.x - box2.x) < reach * 0.5 + t.w / 2 && Math.abs(t.y - actor.y) < 40)
          ev.push({
            t: "hit",
            target: t.id,
            damage: dmg,
            knockback: w.def.knockback,
            knockdown: false,
            status: w.def.status,
            armorBreak: w.def.armorBreak,
            x: t.x,
            y: t.y,
            source: w.def.id
          });
      if (w.broke) ev.push({ t: "weaponBreak", weapon: w.def.id });
    }
    for (const e of field.ents.slice())
      if (e.type === "prop" && PROPS[e.prop].use !== "throw" && Math.abs(e.x - box2.x) < reach * 0.5 + PROPS[e.prop].w / 2 && Math.abs(e.y - actor.y) < 44)
        ev.push(...damageProp(field, run, e, dmg, targets));
    return { events: ev, handled: !!w };
  }
  function shoot(field, run, actor, held = false) {
    const ev = [];
    const g = run.held && WEAPONS[run.held.id].cls === "gun" ? WEAPONS[run.held.id] : null;
    if (g) {
      if (held && !g.rate) return ev;
      if (run.shotCd > 0) return ev;
      const w = useWeapon(run, "gun");
      run.shotCd = g.rate ?? 0.28;
      const n = g.pellets ?? 1;
      for (let i = 0; i < n; i++) spawnBullet(field, actor, g.damage, g.knockback, g.reach, g.id, n > 1 ? (i / (n - 1) - 0.5) * (g.spread ?? 0) : 0, g.vsHuman);
      ev.push({ t: "shot", source: g.id, x: actor.x, y: actor.y, dir: actor.dir });
      if (w.broke) ev.push({ t: "weaponBreak", weapon: g.id });
      return ev;
    }
    if (held) return ev;
    const r = fireSignature(run);
    if (r === "fire") {
      spawnBullet(field, actor, 10, 30, 640, "signature", 0);
      ev.push({ t: "shot", source: "signature", x: actor.x, y: actor.y, dir: actor.dir });
      if (run.reloadT > 0) ev.push({ t: "reload" });
    } else if (r === "dry") ev.push({ t: "dry" });
    else if (r === "reloading" && run.reloadT > 0 && run.shotCd === 0) {
      run.shotCd = 0.2;
      ev.push({ t: "reload" });
    }
    return ev;
  }
  function spawnBullet(field, a, damage, knockback, reach, ref, dy, vsHuman) {
    const speed = 1500;
    field.projs.push({
      id: field.nextId++,
      kind: "bullet",
      ref,
      x: a.x + a.dir * 50,
      y: a.y + dy,
      z: a.z + 130,
      vx: a.dir * speed,
      vz: 0,
      life: reach / speed,
      damage,
      knockback,
      knockdown: false,
      vsHuman,
      hit: []
    });
  }
  function interact(field, run, actor) {
    if (run.carry) {
      const e = placeProp(field, run.carry.prop, actor.x + actor.dir * 50, actor.y);
      e.hp = run.carry.hp;
      run.carry = null;
      return [];
    }
    let best = null, bd = PICKUP.interact;
    for (const e of field.ents) {
      const d3 = near(e, actor);
      if (d3 >= bd) continue;
      if (e.type === "pickup" && (e.kind === "weapon" || e.kind === "item") && e.z < 8) {
        best = e;
        bd = d3;
      } else if (e.type === "prop" && PROPS[e.prop].use !== "break" && !run.held) {
        best = e;
        bd = d3;
      }
    }
    if (!best) return [];
    if (best.type === "prop") {
      field.ents.splice(field.ents.indexOf(best), 1);
      run.carry = { prop: best.prop, hp: best.hp };
      return [{ t: "lift", prop: best.prop }];
    }
    return take(field, run, best, true);
  }
  function take(field, run, p, manual, ctx) {
    if (!field.ents.includes(p)) return [];
    const at2 = { x: p.x, y: p.y };
    let amount = p.amount;
    switch (p.kind) {
      case "heal":
        if (ctx) {
          if (ctx.hp >= ctx.max) return [];
          amount = Math.min(p.amount, Math.ceil((ctx.max - ctx.hp) / pickupHealAmount(ctx.max)));
          ctx.hp = Math.min(ctx.max, ctx.hp + pickupHealAmount(ctx.max, amount));
        }
        break;
      case "ammo": {
        const ammo = SIGNATURES[run.agent]?.ammo;
        if (!ammo || ammo === "none") return [];
        p.ammoLeft ??= AMMO_PICKUP[ammo].amount * p.amount;
        amount = addAmmo(run, p.ammoLeft);
        if (!amount) return manual ? [{ t: "full", kind: "ammo" }] : [];
        p.ammoLeft -= amount;
        if (!p.ammoLeft) field.ents.splice(field.ents.indexOf(p), 1);
        return [{ t: "pickup", kind: p.kind, amount, ...at2 }];
      }
      case "item":
        amount = addItem(run, p.ref, p.amount);
        if (!amount) return manual ? [{ t: "full", kind: "item", ref: p.ref }] : [];
        break;
      case "weapon": {
        const prev = equip(run, p.ref, p.uses);
        field.ents.splice(field.ents.indexOf(p), 1);
        if (prev) spawnPickup(field, "weapon", at2.x, at2.y, prev.id, 1, prev.uses);
        return [{ t: "equip", weapon: p.ref, dropped: prev?.id }];
      }
    }
    p.amount -= amount;
    if (p.amount <= 0) field.ents.splice(field.ents.indexOf(p), 1);
    return [{ t: "pickup", kind: p.kind, ref: p.ref, amount, ...at2 }];
  }
  function stepField(field, run, actor, input, targets, dt, ctx, melee) {
    const ev = [];
    ctx = ctx ? { ...ctx } : void 0;
    if (tickRun(run, dt)) ev.push({ t: "reloaded" });
    const b = field.bounds;
    for (const e of field.ents.slice()) {
      if (e.type === "prop") {
        e.flash = Math.max(0, e.flash - dt);
        continue;
      }
      e.age += dt;
      if (e.z > 0 || e.vz > 0) {
        e.vz -= 1400 * dt;
        e.z += e.vz * dt;
        e.x = Math.max(b.x0, Math.min(b.x1, e.x + e.vx * dt));
        e.y = Math.max(b.y0, Math.min(b.y1, e.y + e.vy * dt));
        if (e.z <= 0) {
          e.z = 0;
          e.vz = Math.abs(e.vz) > 200 ? -e.vz * 0.3 : 0;
          e.vx *= 0.5;
          e.vy *= 0.5;
        }
        continue;
      }
      if (e.age < 0.35 || actor.z > 40) continue;
      const auto = e.kind === "heal" || e.kind === "ammo";
      const d3 = near(e, actor);
      if (auto && d3 < PICKUP.magnet) {
        e.x += (actor.x - e.x) * Math.min(1, dt * 14);
        e.y += (actor.y - e.y) * Math.min(1, dt * 14);
      }
      if (auto && d3 < PICKUP.touch || e.kind === "item" && d3 < PICKUP.touch) ev.push(...take(field, run, e, false, ctx));
    }
    if (input.interact) ev.push(...interact(field, run, actor));
    if (input.attack) ev.push(...attack(field, run, actor, targets, melee).events);
    if (input.shoot || input.shootHeld) ev.push(...shoot(field, run, actor, !input.shoot));
    if (input.use !== void 0) {
      const u = useItem(run, input.use, ctx);
      if (u) {
        ev.push({ t: "use", slot: input.use, id: u.id, effect: u.effect });
        const e = u.effect;
        if (e.type === "salt") ev.push(burst(field, "salt", actor.x, actor.y, e.radius, e.time));
        if (e.type === "flash" || e.type === "smoke")
          field.projs.push({
            id: field.nextId++,
            kind: "grenade",
            ref: u.id,
            x: actor.x + actor.dir * 30,
            y: actor.y,
            z: actor.z + 120,
            vx: actor.dir * 380,
            vz: 360,
            life: 3,
            damage: 0,
            knockback: 0,
            knockdown: false,
            hit: []
          });
      }
    }
    for (const p of field.projs.slice()) {
      p.life -= dt;
      p.x += p.vx * dt;
      if (p.kind !== "bullet") {
        p.vz -= 1300 * dt;
        p.z += p.vz * dt;
      }
      let done2 = p.life <= 0 || p.x < b.x0 - 60 || p.x > b.x1 + 60;
      if (p.kind === "grenade") {
        if (p.x < b.x0 || p.x > b.x1) {
          p.x = Math.max(b.x0, Math.min(b.x1, p.x));
          p.vx = 0;
        }
        done2 = p.life <= 0;
        if (p.z <= 0 || done2) {
          const def = ITEMS[p.ref].effect;
          if (def.type === "flash") {
            ev.push(burst(field, "flash", p.x, p.y, def.radius, 0.4));
            for (const t of targets)
              if (Math.hypot(t.x - p.x, (t.y - p.y) * 2) < def.radius)
                ev.push({ t: "hit", target: t.id, damage: 0, knockback: 0, knockdown: false, status: "stun", stun: def.stun, x: t.x, y: t.y, source: p.ref });
          } else if (def.type === "smoke") ev.push(burst(field, "smoke", p.x, p.y, def.radius, def.time, def.slow));
          done2 = true;
        }
      } else {
        for (const t of targets) {
          if (p.hit.includes(t.id) || !hitsTarget(p, t, p.kind === "prop" ? 16 : 0)) continue;
          p.hit.push(t.id);
          const dmg = Math.round(p.damage * (t.human && p.vsHuman ? p.vsHuman : 1));
          ev.push({ t: "hit", target: t.id, damage: dmg, knockback: p.knockback, knockdown: p.knockdown, status: p.status, armorBreak: p.armorBreak, x: t.x, y: t.y, source: p.ref });
          if (p.kind === "bullet" || p.kind === "prop") done2 = true;
          break;
        }
        if (!done2) {
          for (const e of field.ents)
            if (e.type === "prop" && propBox(e, p) && PROPS[e.prop].use !== "throw") {
              ev.push(...damageProp(field, run, e, Math.max(p.damage, 8), targets));
              done2 = true;
              break;
            }
        }
        if (p.kind === "prop" && (done2 || p.z <= 0)) {
          done2 = true;
          const def = PROPS[p.ref];
          ev.push({ t: "break", prop: p.ref, x: p.x, y: p.y });
          if (def.burst) ev.push(...explode(field, run, def.burst, p.x, p.y, def.id, targets, def.knockdown));
          else ev.push(burst(field, "blast", p.x, p.y, 40, 0.25));
        }
      }
      if (done2) field.projs.splice(field.projs.indexOf(p), 1);
    }
    for (const z of field.zones.slice()) {
      z.t += dt;
      if (z.t >= z.life) field.zones.splice(field.zones.indexOf(z), 1);
    }
    return ev;
  }

  var Proj = class {
    constructor(def, name, x, y, dir, owner, attackBase, mult = 1) {
      this.def = def;
      this.owner = owner;
      this.mult = mult;
      this.kind = "proj:" + name;
      this.label = def.label;
      this.x = x;
      this.y = y;
      this.z = def.z;
      this.dir = dir;
      this.vx = def.speed * dir;
      this.vz = def.vz ?? 0;
      this.half = def.w / 2;
      this.attackId = attackBase;
    }
    id = newBodyId();
    team = "enemy";
    kind;
    x;
    y;
    z;
    dir;
    hp = 1;
    hpMax = 1;
    alive = true;
    stop = 0;
    label;
    vx;
    vz;
    t = 0;
    half;
    done = false;
    attackId;
    box() {
      const d3 = this.def;
      return { x0: this.x - this.half, x1: this.x + this.half, z0: this.z, z1: this.z + d3.h, y: this.y, dy: d3.low ? 16 : 20 };
    }
    frame() {
      const d3 = this.def;
      if (!this.alive || d3.harmless || d3.arm && this.t < d3.arm) return { hit: [], hurt: [] };
      const hit2 = {
        ...this.box(),
        attackId: this.attackId,
        dmg: Math.round(d3.dmg * this.mult),
        react: d3.react,
        dir: this.dir,
        push: d3.push ?? 2,
        stop: 2,
        low: d3.low,
        armorBreak: d3.unblockable,
        strength: d3.react === "down" ? 3 : d3.react === "stagger" ? 2 : 1
      };
      if (d3.debuff) {
        hit2.debuff = d3.debuff;
        armDebuff(hit2, d3.debuff, this.id);
      }
      return { hit: [hit2], hurt: [] };
    }
    step(world) {
      const d3 = this.def;
      checkDebuffs(world);
      if (this.done) return void (this.alive = false);
      this.t++;
      if (d3.every && this.t % d3.every === 0) this.attackId = newAttackId();
      this.x += this.vx;
      if (d3.gravity) {
        this.z += this.vz;
        this.vz -= d3.gravity;
        if (this.z <= 0) {
          this.z = 0;
          world.fx("dust", this.x, this.y, 0);
          world.shake(4, 2);
          this.alive = false;
        }
      }
      if (d3.grow) this.half += d3.grow;
      if (d3.homeY) {
        const dy = world.player.y - this.y;
        this.y += Math.max(-d3.homeY, Math.min(d3.homeY, dy));
      }
      if (this.t >= d3.life) this.alive = false;
      if (!d3.pierce && !d3.harmless && !(d3.arm && this.t < d3.arm)) {
        const p = world.player;
        if (p.alive && p.frame().hurt.some((h) => overlap2(this.box(), h))) this.done = true;
      }
    }
    applyHit() {
      return { kind: "immune" };
    }
    draw(ctx) {
      const d3 = this.def;
      const b = this.box();
      ctx.save();
      if (d3.arm && this.t < d3.arm) {
        ctx.globalAlpha = 0.25 + 0.5 * (this.t >> 2 & 1);
        ctx.fillStyle = "#ff4040";
        ctx.fillRect(b.x0, this.y - b.z1 + d3.h / 2 - 2, b.x1 - b.x0, 4);
        ctx.restore();
        return;
      }
      ctx.globalAlpha = d3.harmless ? 0.55 * (1 - this.t / d3.life) + 0.2 : d3.pierce ? 0.55 : 0.95;
      ctx.fillStyle = d3.color;
      if (d3.grow) {
        ctx.fillRect(b.x0, this.y - b.z1, 10, b.z1 - b.z0);
        ctx.fillRect(b.x1 - 10, this.y - b.z1, 10, b.z1 - b.z0);
      } else ctx.fillRect(b.x0, this.y - b.z1, b.x1 - b.x0, b.z1 - b.z0);
      ctx.globalAlpha = 0.3;
      ctx.fillStyle = "#000";
      ctx.beginPath();
      ctx.ellipse(this.x, this.y, Math.min(60, this.half), 6, 0, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  };

  function poseOf(e) {
    const f = e.frameNow();
    const p = { tilt: 0, lean: 0, step: 0, armF: 0.2, armB: -0.15, reach: 1, crouch: 0, lie: 0, open: 0 };
    switch (e.state) {
      case "walk":
      case "flee": {
        const s = Math.sin(e.age * (e.state === "flee" ? 0.5 : 0.32));
        p.step = s;
        p.armF = 0.15 - s * 0.5;
        p.armB = -0.15 + s * 0.5;
        p.lean = e.state === "flee" ? 0.25 : 0.04;
        p.crouch = Math.abs(s) * -3;
        break;
      }
      case "attack": {
        if (f?.tele) {
          p.lean = -0.16;
          p.armF = 2.5;
          p.armB = -0.6;
          p.crouch = 8;
          p.open = 0.5;
        } else if (f?.hit.length) {
          const b = f.hit[0];
          const shoulderZ = e.def.h * 0.78;
          const midZ = (b.z0 + b.z1) / 2;
          const midX = Math.max(10, (b.fx0 + b.fx1) / 2);
          p.armF = Math.max(0.3, Math.min(2.6, Math.atan2(midX, shoulderZ - midZ)));
          p.reach = Math.min(1.7, Math.max(1, Math.hypot(midX, shoulderZ - midZ) / (e.def.h * 0.42)));
          p.lean = 0.22;
          p.armB = -0.9;
          p.step = 0.7;
          p.open = 1;
        } else {
          p.lean = 0.1;
          p.armF = 1;
          p.step = 0.4;
        }
        break;
      }
      case "grabbing":
        p.armF = 1.5;
        p.armB = 1.3;
        p.lean = 0.15;
        p.open = 1;
        break;
      case "hurt":
      case "bound": {
        const k = Math.max(0, 1 - e.st / 10);
        p.tilt = -0.28 * (0.4 + 0.6 * k);
        p.lean = -0.18 * k;
        p.armF = 1.1;
        p.armB = 0.9;
        break;
      }
      case "stunned":
        p.tilt = Math.sin(e.age * 0.25) * 0.12;
        p.lean = 0.2;
        p.armF = 0.1;
        p.armB = 0.1;
        p.crouch = 10;
        break;
      case "air":
        p.tilt = -0.5 - Math.min(0.9, e.st * 0.05);
        p.armF = 2.2;
        p.armB = 2;
        p.step = 0.6;
        break;
      case "down":
      case "dead":
        p.lie = 1;

        p.armB = 3.05;
        break;
      case "getup":
        p.lie = Math.max(0, 1 - e.st / 12);
        p.crouch = 18 * p.lie;
        break;
      default:
        p.crouch = Math.sin(e.age * 0.09) * 2;
    }
    return p;
  }

    const m = c.startsWith("#") ? [parseInt(c.slice(1, 3), 16), parseInt(c.slice(3, 5), 16), parseInt(c.slice(5, 7), 16)] : (c.match(/\d+/g) ?? ["0", "0", "0"]).slice(0, 3).map(Number);

    return `rgb(${m.map((v) => Math.round(v * k)).join(",")})`;
  }
