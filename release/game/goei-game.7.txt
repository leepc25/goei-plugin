            if (this.hitMemo.has(key)) continue;
            this.hitMemo.set(key, this.tick);
            this.applyHit(a, t, h);
            if (a.kind === "bullet") a.alive = false;
            if (!a.alive) break;
          }
      }
      if (this.hitMemo.size > 400) {
        for (const [k, v] of this.hitMemo) if (this.tick - v > 120) this.hitMemo.delete(k);
      }
    }
    applyHit(a, t, h) {
      if (a.kind === "bullet") h = progressionHit(h, this.player.progression, "ranged");
      const side = Math.sign(t.x - a.x);
      if (side && side !== h.dir) h = { ...h, dir: side };
      const r = t.applyHit(h, this);
      const at2 = { x: (Math.max(h.x0, t.x - 30) + Math.min(h.x1, t.x + 30)) / 2, y: t.y, z: (h.z0 + h.z1) / 2 };
      if (r.kind === "immune") return;
      const stop = h.stop ?? STOP_DEFAULT[h.react];
      if (r.kind === "guard") {
        a.stop = Math.max(a.stop, 3);
        t.stop = Math.max(t.stop, 3);
        this.fx("spark", at2.x, at2.y, at2.z);
        return;
      }
      a.stop = Math.max(a.stop, stop);
      t.stop = Math.max(t.stop, stop);
      const strength = h.strength ?? (h.react === "down" || h.react === "launch" ? 3 : h.react === "stagger" ? 2 : 1);
      this.fx(strength >= 3 ? "spark-big" : "spark", at2.x, at2.y, at2.z);
      if (this.hitFx) {
        if (strength >= 2) this.fx("ring", at2.x, at2.y, at2.z, String(strength));
        if (strength >= 3) this.shake(6, 3);
      }
      if (h.noise) this.emit({ type: "custom", name: "noise", data: h.noise });
      const p = this.player;
      if (a === p || a.kind === "bullet") {
        p.onConnect(this, !!r.killed);
        if (r.killed) this.kills++;
      }
      const mk = a.kind === "bullet" ? "bullet" : a !== p ? "enemy" : { airkick: "kick", lowkick: "low", dash: "dash", blast: "blast", groundblast: "blast", grabblast: "blast", super: "super", grabknee: "kick" }[p.moveKey] ?? "punch";
      this.emit({ type: "hit", strength, x: at2.x, y: at2.y, z: at2.z, dmg: h.dmg, react: h.react, target: t.kind, combo: a === p ? p.combo : 0, kind: mk });
      if (r.killed && t.team === "enemy" && !(t instanceof Fighter)) this.emit({ type: "enemyDie", x: t.x, kind: t.kind });
      if (a === p && p.moveKey === "grabknee") p.keepHeld();
    }
    stepEncounters() {
      const encs = this.floor.encounters;
      const p = this.player;
      if (this.encActive < 0) {
        const e2 = encs[this.encIndex];
        if (e2 && p.x >= e2.triggerX) {
          this.encActive = this.encIndex;
          this.goT = 0;
          this.encWave = 0;
          this.bounds = [e2.lockFrom, e2.lockTo];
          this.spawnWave();
          this.emit({ type: "combat:start", index: this.encActive });
          if (e2.waves.some((w) => w.some((s) => s.boss))) this.emit({ type: "combat:boss", index: this.encActive });
        }
        return;
      }
      const e = encs[this.encActive];
      this.encBodies = this.encBodies.filter((b) => b.alive && b.hp > 0 && this.threat(b) && !(b instanceof Fighter && (b.phase === "dead" || b.phase === "down" && b.hp <= 0)));
      if (this.encBodies.length) return;
      this.encWave++;
      if (this.encWave < e.waves.length) {
        this.spawnWave();
        return;
      }
      this.emit({ type: "combat:clear", index: this.encActive });
      this.encActive = -1;
      this.encIndex++;
      this.bounds = [0, this.floor.length];
      this.goT = 90;
    }
    spawnWave() {
      const e = this.floor.encounters[this.encActive];
      this.encBodies = e.waves[this.encWave].map((s) => this.spawn(s));
    }
    camera() {
      const p = this.player;
      const [b0, b1] = this.bounds;
      const W5 = view.w;
      if (this.floor.length <= W5) {
        this.camX = -(W5 - this.floor.length) / 2;
        return;
      }
      const target = p.x - W5 / 2 + p.dir * 90;
      let cx = this.camX + (target - this.camX) * 0.14;
      cx = Math.max(b0, Math.min(b1 - W5, cx));
      cx = Math.max(0, Math.min(this.floor.length - W5, cx));
      this.camX = cx;
      if (this.encActive >= 0) p.x = Math.max(cx + 90, Math.min(cx + W5 - 90, p.x));
    }
    /** 보스 체력 막대용 */
    boss() {
      const b = this.bodies.find((x) => x.alive && x.boss);
      return b ? { label: b.label ?? b.kind, hp: b.hp, hpMax: b.hpMax } : null;
    }
  };

  var Game = class {
    constructor(box2, art4, mods) {
      this.box = box2;

      this.input.attach();
      this.canvas.addEventListener("pointerdown", (e) => this.onPointer(e));

    }
    canvas;
    ctx;
    input = new Input();
    world = null;
    scene;
    paused = false;
    ownCanvas;
    renderer;
    raf = 0;
    acc = 0;
    last = 0;
    menuIndex = 0;
    run = null;

    /** 꼬리표 시작은 주 루프가 돈 뒤에 (start()에서) */
    booted = false;
    start() {
      if (!this.booted) {
        this.booted = true;
        queueMicrotask(() => this.mods.boot?.(this));
      }
      const frame = (now) => {
        if (this.dead) return;
        this.raf = requestAnimationFrame(frame);
        if (!this.last) this.last = now;
        let dt = (now - this.last) / 1e3;
        this.last = now;
        if (dt > 0.25) dt = 0.25;
        this.mods.frame?.(dt);
        this.acc += dt;
        const T2 = 1 / TICK_HZ;
        while (this.acc >= T2) {
          this.acc -= T2;
          this.tickOnce();
        }
        const paused = this.paused || !!this.mods.paused?.();
        if (paused !== this.wasPaused) {
          this.wasPaused = paused;
          this.mods.onPause?.(paused);
        }
        if (!this.mods.covered?.()) {
          const w = this.world;
          const H5 = this.inHub ? view.worldH : SCREEN_H;
          view.zoom = this.canvas.height / H5 * Math.max(1, charSize.k) * (this.inHub ? view.sceneZoom : 1);
          if (w) view.lane = w.lane;
          const vw = this.canvas.width / view.zoom;
          view.w = this.inHub ? Math.max(SCREEN_W, vw) : SCREEN_W;
          view.hub = this.inHub;
          view.crop = 0;
          const frozen = paused || !!this.mods.blocking?.() || !!this.mods.hq?.isOpen?.();
          if (vw < SCREEN_W && w) {
            const a = frozen ? 1 : this.acc / T2, cam = Math.round(w.prevCamX + (w.camX - w.prevCamX) * a), px = (w.player.px ?? w.player.x) + (w.player.x - (w.player.px ?? w.player.x)) * a, raw = Math.max(0, Math.min(SCREEN_W - vw, px - cam - vw / 2));
            view.crop = Math.round((cam + raw) * view.zoom) / view.zoom - cam;
          }
          view.cropY = 0;
          const vh = this.canvas.height / view.zoom;
          if (this.inHub) {
            this.ctx.fillStyle = "#000";
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
          }
          if (vh > H5 + 0.5) {
            view.cropY = -Math.round((vh - H5) / 2 * view.zoom) / view.zoom;
          } else if (vh < H5 - 0.5 && w) {
            const a = frozen ? 1 : this.acc / T2, p = w.player;
            const py = (p.py ?? p.y) + (p.y - (p.py ?? p.y)) * a;
            view.cropY = Math.round(Math.max(0, Math.min(H5 - vh, py - vh * 0.7)) * view.zoom) / view.zoom;
          }
          this.ctx.save();
          this.ctx.scale(view.zoom, view.zoom);
          this.ctx.translate(-view.crop, -view.cropY);
          this.scene.draw(paused ? 1 : this.acc / T2);
          this.ctx.restore();
          this.mods.overlay?.(this.ctx, this.world, dt);
        }
      };
      this.raf = requestAnimationFrame(frame);
    }
    /** 시험용: 틱을 n번 바로 돌린다 */
    advance(n) {
      for (let i = 0; i < n; i++) this.tickOnce();
      this.scene.draw(1);
    }
    tickOnce() {
      const p = this.input.sample();
      if (typeof document !== "undefined" && document.hidden) return;
      if (this.mods.paused?.() && this.scene.name === "battle") return;
      this.scene.tick(p);
    }
    destroy() {
      this.checkpoint();
      this.dead = true;
      cancelAnimationFrame(this.raf);
      this.world = null;
      this.run = null;
      this.input.detach();
      if (this.ownCanvas) this.canvas.remove();
      if (this.hqRoot) {
        this.mods.hq?.unmount();
        this.hqRoot.remove();
      }
      this.mods.dispose?.();
    }
    /* ───────── 제목·본부 ───────── */
    toTitle() {
      this.checkpoint();
      this.world = null;
      this.inHub = false;
      this.paused = false;
      if (this.hqRoot) this.hqRoot.hidden = true;
      if (this.mods.openTitle) {
        this.scene = this.idleScene("title");
        this.mods.openTitle();
      } else this.scene = this.titleScene();
      this.mods.onScene?.("title");
    }
    /** 새로 시작: 본부가 있으면 본부로, 없으면 첫 층 모듈로 바로 출동 */
    start_new(stageId) {
      if (this.mods.hq && !stageId) return this.showHq();
      const stage = this.mods.stages.find((s) => s.id === stageId) ?? this.mods.stages[0];
      this.deploy({ seed: Math.floor(Math.random() * 1e9), agent: "eileen", stageId: stage.id });
    }
    dead = false;
    /** 본부를 걷는 중인가 (엔진 벨트 장면) */
    inHub = false;
    showHq() {
      const hq2 = this.mods.hq;
      this.world = null;
      this.scene = this.idleScene("hq");
      if (!hq2.hub) this.canvas.style.visibility = "hidden";
      if (!this.hqRoot) {
        this.hqRoot = document.createElement("div");
        this.hqRoot.className = "goei-engine-hq";
        (hq2.hub && this.mods.hqLayer?.() || this.box).appendChild(this.hqRoot);
        hq2.mount(this.hqRoot, {
          deploy: (run) => this.deploy(run),
          stages: () => this.mods.stages.map((s) => ({ id: s.id, label: s.label }))
        });
      }
      this.hqRoot.hidden = false;
      if (hq2.hub) {
        const stage = hq2.hub();
        const run = { seed: Math.floor(Math.random() * 1e9), agent: "eileen", stageId: stage.id };
        this.hqRoot.hidden = !hq2.isOpen?.();
        this.hqRoot.classList.add("goei-engine-hq-panel");
        this.canvas.style.visibility = "";
        this.inHub = true;
        const w = new World(run, this.art.clips, stage, this.mods.enemyFactory);
        for (const l of this.mods.listeners ?? []) w.on(l);
        this.mods.onWorld?.(w, run);
        w.enterFloor(stage.firstFloor(run));
        this.world = w;
        this.runStage = stage;
        this.inHub = true;
        this.paused = false;
        this.scene = this.battleScene(w);
        this.mods.onPlay?.();
      }
      this.mods.onScene?.("hq");
    }
    idleScene(name) {
      return {
        name,
        tick: () => {
        },
        draw: () => {
          this.ctx.fillStyle = "#05080a";
          this.ctx.fillRect(0, 0, SCREEN_W, SCREEN_H);
          this.mods.drawIdle?.(name, this.ctx, performance.now() / 1e3);
        }
      };

    /* ───────── 출동 ───────── */
    deploy(run, saved) {
      const stage = this.mods.stages.find((s) => s.id === run.stageId) ?? this.mods.stages[0];
      run = { ...run, stageId: stage.id };
      this.inHub = false;
      this.run = run;
      this.runStage = stage;
      if (this.hqRoot) this.hqRoot.hidden = true;
      this.canvas.style.visibility = "";
      const w = new World(run, this.art.clips, stage, this.mods.enemyFactory);
      if (this.mods.itemUse) w.itemUse = this.mods.itemUse;
      for (const l of this.mods.listeners ?? []) w.on(l);
      this.mods.onWorld?.(w, run);
      w.enterFloor(saved && stage.restore ? stage.restore(saved.stage) : stage.firstFloor(run));
      if (saved) w.restore(saved.world);
      this.world = w;
      this.paused = false;
      this.scene = this.battleScene(w);
      this.mods.onPlay?.();
      this.mods.onScene?.("battle");
      this.checkpoint(true);

    /** 전투 도중에는 직전 안전 지점을 유지한다. HP·탄·드랍도 함께 되돌린다. */
    checkpoint(force = false) {
      const w = this.world;
      if (!w || this.inHub || !this.run || !w.canCheckpoint() || !force && this.mods.blocking?.()) return;
      const stage = this.runStage?.snapshot?.();
      if (!stage) return;
      this.mods.saveCheckpoint?.({ version: 1, run: this.run, stage, world: w.snapshot(), savedAt: Date.now() });
    }
    async resume(saved) {
      const stageId = saved.run.stageId ?? this.mods.stages[0]?.id;
      if (saved.version !== 1 || !this.mods.stages.some((s) => s.id === stageId && s.restore)) throw new Error("지원하지 않는 출동 저장");
      if (this.mods.hq) {
        this.showHq();
        await this.mods.hq.ready?.();
      }
      this.deploy({ ...saved.run, stageId }, saved);
      this.world?.say("출동 복원 완료. 전투 중 중단한 경우 마지막 안전 지점부터 이어집니다.", 4);
    }
    /** 출동 끝내기 (철수 메뉴 등에서도 부른다) */
    endRun(how) {
      const w = this.world;
      if (!w || this.inHub) return;
      const result = {
        how,
        floors: Number(w.notes["stage.visitedCount"] ?? w.floors),
        kills: w.kills,
        maxCombo: w.player.maxCombo,
        timeSec: Math.round(w.tick / TICK_HZ),
        hpLeft: Math.max(0, Math.round(w.player.hp)),
        notes: w.notes
      };
      w.emit({ type: "run:end", how });
      this.lastResult = result;
      this.world = null;
      this.paused = false;
      if (this.mods.hq && this.run) {
        const run = this.run;
        this.mods.hq.onReturn(this.mods.toHqResult?.(result, w) ?? result, run);
        this.mods.saveCheckpoint?.(null);
        const back = () => this.showHq();
        if (this.mods.afterRun) {
          this.scene = this.idleScene("cutscene");
          this.mods.afterRun(how, back);
        } else back();
        return;
      }
      this.mods.saveCheckpoint?.(null);
      this.scene = this.resultScene(result);
      this.mods.onScene?.("result");
    }
    battleScene(w) {
      const ownMenu = !this.mods.paused;
      return {
        name: "battle",
        tick: (p) => {
          if (ownMenu && p.pressed.has("pause")) {
            this.paused = !this.paused;
            this.menuIndex = 0;
          }
          if (this.paused) return this.pauseMenu(p, w);
          if (this.inHub && this.hqRoot) {
            const open = !!this.mods.hq?.isOpen?.();
            this.hqRoot.hidden = !open;
            if (open) return;
          }
          if (this.mods.blocking?.() || this.mods.frozen?.()) return;
          w.step(p);
          if (w.ended) {
            const how = w.ended;
            if (how === "next") {
              const next = this.runStage.nextFloor(this.run, w.floor, "next");
              if (next) {
                w.enterFloor(next);
                this.checkpoint(true);
              } else this.endRun("return");
            } else this.endRun(how);
          } else if (w.tick % 60 === 0) this.checkpoint();
        },
        draw: (a) => {
          this.renderer.draw(w, this.paused || this.mods.blocking?.() || this.mods.hq?.isOpen?.() ? 1 : a);
          if (this.paused) this.drawMenu("일시정지", ["계속", w.showBoxes ? "판정 보기 끄기" : "판정 보기 켜기", "철수 (출동 끝내기)", "처음으로"]);
        }
      };
    }
    pauseMenu(p, w) {
      const n = 4;
      if (p.pressed.has("up")) this.menuIndex = (this.menuIndex + n - 1) % n;
      if (p.pressed.has("down")) this.menuIndex = (this.menuIndex + 1) % n;
      if (p.pressed.has("attack") || p.pressed.has("jump") || p.pressed.has("interact")) this.pick(this.menuIndex, w);
    }
    pick(i, w) {
      this.paused = false;
      if (i === 1) w.showBoxes = !w.showBoxes;
      else if (i === 2) this.endRun("withdraw");
      else if (i === 3) this.toTitle();
    }
    drawMenu(title2, items, bg = true) {

      if (bg) {

        c.textAlign = "center";

        c.fillText(title2, SCREEN_W / 2, 210);

      c.textAlign = "center";

        c.fillRect(SCREEN_W / 2 - 230, y - 40, 460, 54);
        c.fillStyle = i === this.menuIndex ? "#1a1f22" : "#e8eef0";

      });

    onPointer(e) {

      else if (this.scene.name === "title") this.titlePick(i);
    }
    /* ───────── 엔진 기본 제목·정산 ───────── */

  function rankStars(c, grade, rank, x, y, z) {

    /** 마지막 그리기의 창 → 캔버스 변환 (배율, 왼쪽 위) */
    fit = { k: 1, ox: 0, oy: 0 };
    toggle(on = !this.open) {
      this.open = on;
    }
    /** 창이 열려 있을 때 눌린 키 (닫기 키는 껍데기가 먼저 받는다) */
    key(code) {
      const n = CHAR_TABS.length;
      if (code === "KeyQ") this.tab = (this.tab + n - 1) % n;
      else if (code === "KeyE") this.tab = (this.tab + 1) % n;
      const tab = CHAR_TABS[this.tab];
      const dx = code === "ArrowLeft" ? -1 : code === "ArrowRight" ? 1 : 0, dy = code === "ArrowUp" ? -1 : code === "ArrowDown" ? 1 : 0;
      if (!dx && !dy) return;
      if (tab === "기술" && this.skills) {
        const i = this.skill + dx + dy * 3;
        if (i >= 0 && i < this.skills) this.skill = i;
      } else if (tab === "능력" && this.perks) {
        const i = this.perk + dx + dy * 5;
        if (i >= 0 && i < this.perks) this.perk = i;
      }
    }
    /** 캔버스 픽셀 좌표를 눌렀을 때: 탭 이름 위면 그 탭으로 바꾸고 true */
    tap(px, py) {
      const { k, ox, oy } = this.fit;
      const x = (px - ox) / k, y = (py - oy) / k;
      const i = CHAR_TAB_HITS.findIndex((r) => x >= r.x && x < r.x + r.w && y >= r.y && y < r.y + r.h);
      if (i < 0) return false;
      this.tab = i;
      return true;
    }
    draw(c, d3) {
      this.skills = d3.skills.length;
      this.perks = d3.perks.length;
      this.skill = Math.min(this.skill, Math.max(0, this.skills - 1));
      this.perk = Math.min(this.perk, Math.max(0, this.perks - 1));
      c.save();
      const cw = c.canvas.width, ch = c.canvas.height, k = Math.min(cw / 1280, ch / 720);
      c.setTransform(1, 0, 0, 1, 0, 0);
      c.fillStyle = "rgba(0,0,0,0.55)";
      c.fillRect(0, 0, cw, ch);
      this.fit = { k, ox: Math.round((cw - 1280 * k) / 2), oy: Math.round((ch - 720 * k) / 2) };
      c.setTransform(k, 0, 0, k, this.fit.ox, this.fit.oy);
      c.imageSmoothingEnabled = false;
      drawCharWindow(c, { ...d3, pick: d3.pick(this.skill), perkPick: this.perk }, CHAR_TABS[this.tab]);
      c.restore();
    }
  };
