      items.forEach((it, i) => {
        const b = this.btns.find((x) => x.id === `item${i + 1}`);
        if (!b?.el) return;
        const key2 = `${it.id ?? ""}|${it.label}`;
        const fresh = b.el.dataset.icon !== key2;
        if (fresh) {
          b.el.dataset.icon = key2;
          b.el.innerHTML = `<canvas width="64" height="64"></canvas><i>${i + 1}</i><u></u>`;
        }
        if (fresh || this.iconRedraw-- > 0) {
          const cv = b.el.querySelector("canvas");
          const g2 = cv.getContext("2d");
          if (g2) {
            g2.clearRect(0, 0, 64, 64);
            g2.imageSmoothingEnabled = false;
            if (it.label || it.id) drawHudItemIcon(g2, it, 0, 0, 64);
          }
        }
        const u = b.el.querySelector("u");
        const n = it.label || it.id ? String(it.count) : "";
        if (u.textContent !== n) u.textContent = n;
        b.el.classList.toggle("empty", it.count <= 0);
      });
    }
    /** 필살 게이지 0~1 */
    setGauge(v) {
      const b = this.btns.find((x) => x.id === "ult").el;
      b.style.setProperty("--g", String(Math.max(0, Math.min(1, v))));
      b.classList.toggle("full", v >= 1);
      if (TOUCH_IMG.ult_ready) b.style.backgroundImage = `url("${TOUCH_IMG[v >= 1 ? "ult_ready" : "ult"]}")`;
    }
    /** 버튼 강조 (조사할 것이 있을 때 E, 탄이 적을 때 R 등) */
    hint(id, on) {
      this.btns.find((x) => x.id === id)?.el?.classList.toggle("hint", on);
    }
    hit(x, y) {
      let best = null, bd = Infinity;
      for (const b of this.btns) {
        if (!this.usable(b)) continue;
        if (this.root.closest(".exploration-map") && b.id !== "map" && b.id !== "menu") continue;
        if (b.hw) {
          const d4 = Math.max(Math.abs(x - b.cx) / (b.hw + 3), Math.abs(y - b.cy) / (b.hh + 3));
          if (d4 < 1 && d4 < bd) best = b, bd = d4;
          continue;
        }
        const d3 = Math.hypot(x - b.cx, y - b.cy);
        const lim = b.r * (b.group === "cluster" ? 1.18 : 1.05);
        if (d3 < lim && d3 / b.r < bd) best = b, bd = d3 / b.r;
      }
      return best;
    }
    local(e) {
      if (this.toLocal) return this.toLocal(e.clientX, e.clientY, this.layer);
      const r = this.layer.getBoundingClientRect();
      return { x: e.clientX - r.left, y: e.clientY - r.top };
    }
    hold(b, on) {
      const n = (this.count.get(b.code) ?? 0) + (on ? 1 : -1);
      this.count.set(b.code, Math.max(0, n));
      b.el.classList.toggle("on", n > 0);
      if (b.code === "Escape" || b.code === "KeyZ") {
        if (on && n === 1) {
          this.releaseAll();
          this.onMenu(b.code);
        }
        return;
      }
      if (on && n === 1) {
        this.hub.press(b.code, "touch");
        if (this.st().vibrate) navigator.vibrate?.(8);
      } else if (!on && n <= 0) this.hub.release(b.code, "touch");
    }
    down(e) {
      if (!this.shown || e.pointerType === "mouse" && e.button !== 0) return;
      e.preventDefault();
      const p = this.local(e);
      const b = this.hit(p.x, p.y);
      try {
        this.root.setPointerCapture(e.pointerId);
      } catch {
      }
      if (b) {
        this.ptr.set(e.pointerId, { kind: "btn", b });
        this.hold(b, true);
        return;
      }
      if (this.root.closest(".exploration-map")) return;
      const zr = this.zr;
      const zx = p.x - zr.x, zy = p.y - zr.y;
      if (this.stickPid === null && zx >= 0 && zy >= 0 && zx <= zr.w && zy <= zr.h) {
        this.stickPid = e.pointerId;
        this.ptr.set(e.pointerId, { kind: "stick" });
        this.placeBase(zx, zy);
        this.base.classList.add("on");
        this.stickTo(zx, zy);
      }
    }
    move(e) {
      const st = this.ptr.get(e.pointerId);
      if (!st) return;
      e.preventDefault();
      const p = this.local(e);
      if (p.x < 0 || p.y < 0 || p.x >= this.layer.clientWidth || p.y >= this.layer.clientHeight) {
        this.up(e);
        return;
      }
      if (st.kind === "stick") {
        this.stickTo(p.x - this.zr.x, p.y - this.zr.y);
        return;
      }
      const b = this.hit(p.x, p.y);
      if (b !== st.b) {
        if (st.b) this.hold(st.b, false);
        st.b = b;
        if (b) this.hold(b, true);
      }
    }
    up(e) {
      const st = this.ptr.get(e.pointerId);
      if (!st) return;
      this.ptr.delete(e.pointerId);
      if (this.root.hasPointerCapture(e.pointerId)) this.root.releasePointerCapture(e.pointerId);
      if (st.kind === "btn") {
        if (st.b) this.hold(st.b, false);
      } else {
        this.stickPid = null;
        this.setStick(0, 0);
        this.restBase();
      }
    }
    stickTo(x, y) {
      const R = this.sr * 0.95;
      let dx = x - this.baseX, dy = y - this.baseY;
      const m = Math.hypot(dx, dy);
      if (m > R * 1.5) {
        const k = (m - R * 1.5) / m;
        this.placeBase(this.baseX + dx * k, this.baseY + dy * k);
        dx = x - this.baseX;
        dy = y - this.baseY;
      }
      const mm = Math.hypot(dx, dy);
      const cl = mm > R ? R / mm : 1;
      this.knob.style.transform = `translate(${dx * cl}px, ${dy * cl}px)`;
      let nx = dx * cl / R, ny = dy * cl / R;
      if (Math.hypot(nx, ny) < 0.18) nx = ny = 0;
      this.setStick(nx, ny);
    }
    setStick(x, y) {
      this.sx = x;
      this.sy = y;
      this.hub.setAnalog("touch", x, y);
      const dir = (code, v) => {
        if (this.dirOn[code] === v) return;
        this.dirOn[code] = v;
        if (v) this.hub.press(code, "touch");
        else this.hub.release(code, "touch");
      };
      const th = (on) => on ? 0.3 : 0.42;
      dir("ArrowLeft", x < -th(this.dirOn.ArrowLeft));
      dir("ArrowRight", x > th(this.dirOn.ArrowRight));
      dir("ArrowUp", y < -th(this.dirOn.ArrowUp) * 1.15);
      dir("ArrowDown", y > th(this.dirOn.ArrowDown) * 1.15);
      const wantRun = this.st().stickRun && Math.abs(x) > 0.9;
      const shiftBtn = this.btns.find((b) => b.code === "ShiftLeft");
      if (wantRun && !this.runOn) {
        this.runOn = true;
        this.runSince = performance.now();
        const pending = this.runTimer !== 0;
        clearTimeout(this.runTimer);
        this.runTimer = 0;
        if (!pending) this.hold(shiftBtn, true);
      } else if (!wantRun && this.runOn) {
        this.runOn = false;
        const left = 250 - (performance.now() - this.runSince);
        clearTimeout(this.runTimer);
        this.runTimer = 0;
        if (left > 0) this.runTimer = window.setTimeout(() => {
          this.runTimer = 0;
          this.hold(shiftBtn, false);
        }, left);
        else this.hold(shiftBtn, false);
      }
    }
    releaseAll() {
      const ids = [...this.ptr.keys()];
      this.ptr.clear();
      for (const id of ids) if (this.root.hasPointerCapture(id)) this.root.releasePointerCapture(id);
      clearTimeout(this.runTimer);
      this.runTimer = 0;
      this.runOn = false;
      this.stickPid = null;
      this.setStick(0, 0);
      this.count.clear();
      for (const b of this.btns) b.el.classList.remove("on");
      this.hub.releaseSource("touch");
      this.restBase();
    }
    dispose() {
      window.removeEventListener("pointerup", this.onPointerEnd);
      window.removeEventListener("pointercancel", this.onPointerEnd);
      window.removeEventListener("blur", this.onBlur);
      document.removeEventListener("focusin", this.onIme);
      document.removeEventListener("focusout", this.onIme);
      if (typeof visualViewport !== "undefined") visualViewport?.removeEventListener("resize", this.onIme);
      document.removeEventListener("visibilitychange", this.onVisibility);
      window.removeEventListener("touchstart", this.onAnyTouch, { capture: true });
      this.releaseAll();
      this.root.remove();
    }
  };

    };

      });

      }

  function rng(seed) {
    let a = seed >>> 0;
    return () => {
      a = a + 1831565813 >>> 0;
      let t = a;
      t = Math.imul(t ^ t >>> 15, t | 1);
      t ^= t + Math.imul(t ^ t >>> 7, t | 61);
      return ((t ^ t >>> 14) >>> 0) / 4294967296;
    };
  }
  function mix2(...n) {
    let h = 2166136261;
    for (const v of n) {
      h ^= v | 0;
      h = Math.imul(h, 16777619);
      h ^= h >>> 13;
    }
    return h >>> 0;
  }
  var pick3 = (r, list) => list[Math.floor(r() * list.length)];

  function shuffle2(r, list) {
    for (let i = list.length - 1; i > 0; i--) {
      const j = Math.floor(r() * (i + 1));
      [list[i], list[j]] = [list[j], list[i]];
    }
    return list;
  }

  function pickType(depth, last, seed, epoch, salt = 0) {
    if (depth <= 3) return "lobby";
    if (depth === B12) return "deep";
    const r = rng(mix2(seed, depth, epoch, salt, 77));
    const recent = last.slice(-3);
    if (!recent.includes("trial") && last.length >= 2 && r() < 0.6) return "trial";
    const pool = DEEP_TYPES.filter((t) => !recent.includes(t));
    const order = shuffle2(rng(mix2(epoch, 991)), [...DEEP_TYPES]);
    const weights = pool.map((t) => 1 + (order.indexOf(t) < 3 ? 0.8 : 0));
    let roll = r() * weights.reduce((a, b) => a + b, 0);
    for (let i = 0; i < pool.length; i++) if ((roll -= weights[i]) <= 0) return pool[i];
    return pool[0] ?? "chase";
  }

  }
  function pickSequence(ctx, r, count) {
    const pool = chunkPool(ctx.type, ctx.depth).filter((c) => !c.endOnly && !(ctx.room && c.device && !ctx.room.need.includes(c.id)));
    const seq = [];
    for (let i = 0; i < count; i++) {
      let t = pick3(r, pool);
      if (seq.length && seq[seq.length - 1].id === t.id) t = pick3(r, pool);
      seq.push(t);
    }
    const need = [];
    if (ctx.room) need.push(...ctx.room.need);
    else if (ctx.type === "puzzle") need.push("p_fork", "p_switch");
    if (!ctx.room) {
      if (ctx.type === "replica" && (ctx.sub ?? "pretending") === "pretending") need.push(r() < 0.5 ? "r_living" : "r_kitchen");
      if (ctx.type === "replica" && ctx.sub === "vacant") need.push("r_storage");
