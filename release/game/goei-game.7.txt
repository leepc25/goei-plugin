      return b;
    }
    /* ───────── 층 ───────── */
    enterFloor(f) {
      this.floor = f;
      this.lane = f.lane ?? [440, 690];
      this.bounds = [0, f.length];
      this.props = (f.props ?? []).map((p2) => ({ ...p2 }));
      this.bodies = [];
      this.encIndex = 0;
      this.encActive = -1;
      this.encBodies = [];
      this.ended = null;
      this.noise = 0;
      this.light = 1;
      const p = this.player;
      p.x = p.px = f.startX;
      p.y = p.py = (this.lane[0] + this.lane[1]) / 2;
      p.z = p.pz = 0;
      p.vx = p.vy = p.vz = 0;
      p.seat = null;
      p.move = null;
      p.moveKey = "";
      p.grabbed = null;
      p.enter("free");
      p.stop = 0;
      p.resetLocomotion();
      p.dir = 1;
      this.camX = f.length <= view.w ? -(view.w - f.length) / 2 : Math.max(0, Math.min(f.length - view.w, p.x - view.w / 2));
      this.prevCamX = this.camX;
      this.banner = { text: f.banner ?? f.title, t: 70 };
      this.subtitle = null;
      this.floors++;
      this.emit({ type: "enter:floor", floor: f });
    }
    /* ───────── 한 틱 ───────── */
    step(pad) {
      this.prevCamX = this.camX;
      this.tick++;
      if (pad.pressed.has("hitbox")) this.showBoxes = !this.showBoxes;
      for (const f of this.fxs) f.t++;
      this.fxs = this.fxs.filter((f) => f.t < f.life);
      if (this.subtitle && --this.subtitle.t <= 0) this.subtitle = null;
      if (this.banner && --this.banner.t <= 0) this.banner = null;
      if (this.shakeT > 0) this.shakeT--;
      if (this.goT > 0) this.goT--;
      for (const b of [this.player, ...this.bodies]) {
        const f = b;
        f.px = b.x;
        f.py = b.y;
        f.pz = b.z;
      }
      if (this.cutin > 0) {
        this.cutin--;
        return;
      }
      const p = this.player;
      if (p.stop > 0) {
        p.stop--;
        p.listen(pad);
      } else {
        p.tick(this, pad);
        if (p.wantSuper) {
          p.wantSuper = false;
          this.cutin = 22;
          this.emit({ type: "super", x: p.x, move: "super" });
        }
        if (p.wantInteract) this.interact();
        if (p.wantItem) this.useItem(p.wantItem);
      }
      for (const b of this.bodies) {
        if (!b.alive) continue;
        if (b.stop > 0) {
          b.stop--;
          continue;
        }
        b.step(this);
      }
      this.separate();
      this.resolveHits();
      this.bodies = this.bodies.filter((b) => {
        if (b instanceof Fighter && b.phase === "dead" && b.t > 30) b.alive = false;
        return b.alive;
      });
      for (const h of this.hooks) h();
      this.stepEncounters();
      if (this.stage?.tick) this.stage.tick(this.stageApi, this.floor);
      const exitX = this.floor.exitX ?? this.floor.length - 80;
      if (this.encActive < 0 && p.x >= exitX && !this.ended) this.ended = "next";
      if (p.phase === "dead" || p.hp <= 0 && p.phase === "down") {
        if (++this.deadT > 75 && !this.ended) this.ended = p.killed ? "death" : "evac";
      }
      this.camera();
      p.finishLocomotion();
    }
    interact() {
      const p = this.player;
      let best = null;
      for (const pr of this.props) {
        const d3 = Math.abs(pr.x - p.x);
        if (d3 < 60 && (!best || d3 < Math.abs(best.x - p.x))) best = pr;
      }
      if (!p.carry && this.talkHook?.(best)) return;
      if ((p.carry || !best) && this.pickHook?.()) return;
      this.interactWith(best);
    }
    /** 원래 E가 하던 일 (대화창의 「거래」「인사」 같은 버튼이 부른다) */
    interactWith(best) {
      const p = this.player;
      for (const h of this.interactHooks) h(best);
      if (best && this.stage?.onInteract) {
        p.play("pickup", this);
        this.stage.onInteract(this.stageApi, this.floor, best);
      }
    }
    useItem(slot) {
      const p = this.player;
      if (p.move || p.airborne) return;
      if (this.itemUse) {
        const r = this.itemUse(slot, this.stageApi);
        if (r) {
          p.play(r.anim, this);
          this.emit({ type: "item", x: p.x });
        }
        return;
      }
      const it = this.items[slot - 1];
      if (!it || it.n <= 0) return;
      if (it.id === "bandage") {
        if (p.hp >= p.hpMax) return this.say("다친 곳이 없다", 1);
        p.hp = Math.min(p.hpMax, p.hp + 60);
        p.play("bandage", this);
        this.emit({ type: "heal", x: p.x, id: it.id });
      } else if (it.id === "ammo") {
        p.reserve += 24;
        p.play("pickup", this);
        this.emit({ type: "item", x: p.x, id: it.id });
        this.say("탄약 +24", 1);
      }
      it.n--;
    }
    /** 적 모듈 몸(Fighter가 아님)이 요원 몸 안으로 파고들지 않게 최소 간격만큼 적을 밀어낸다 (검수 5-4).
     *  잡기형(선호 거리가 아주 가까운 종)·잡는 중·쓰러짐·공중은 빼고, 적은 자기가 있던 쪽으로 밀린다. */
    separateFromPlayer() {
      const p = this.player;
      if (!p.alive || p.z > 0) return;
      for (const b of this.bodies) {
        const e = b;
        if (!e.alive || b.team === "player" || b instanceof Fighter || !e.def || e.z > 0) continue;
        if (e.state && /^(dead|down|getup|grabbing|air|bound|hidden|dormant|spawn)$/.test(e.state)) continue;
        if (e.kind === "toucher" || e.def.keep && e.def.keep[1] < 30) continue;
        const dy = Math.abs(e.y - p.y);
        if (dy > 85) continue;
        const min = (84 + e.def.w * 0.35) * (dy > 45 ? 0.8 : 1);
        const dx = e.x - p.x;
        if (Math.abs(dx) >= min) continue;
        const s = dx === 0 ? -p.dir : Math.sign(dx);
        e.x += s * Math.min(16, min - Math.abs(dx));
      }
    }
    /** 적 모듈 몸끼리: 한 몸처럼 포개지지 않게 옆(x)으로 벌린다 (검수 16-3) */
    separateEnemies() {
      const skip = (e) => !e.alive || e.team === "player" || e.kind.startsWith("proj") || e instanceof Fighter || !e.def || e.z > 0 || !!e.state && /^(dead|down|getup|grabbing|air|bound|hidden|spawn)$/.test(e.state);
      const list = this.bodies.filter((e) => !skip(e));
      for (let i = 0; i < list.length; i++)
        for (let j = i + 1; j < list.length; j++) {
          const a = list[i], b = list[j];
          const dy = Math.abs(a.y - b.y);
          if (dy > 95) continue;
          const min = (70 + (a.def.w + b.def.w) * 0.5) * (dy > 50 ? 0.8 : 1);
          const dx = b.x - a.x;
          if (Math.abs(dx) >= min) continue;
          const s = dx === 0 ? a.id < b.id ? 1 : -1 : Math.sign(dx);
          const push = Math.min(10, (min - Math.abs(dx)) / 2);
          a.x -= s * push;
          b.x += s * push;
        }
    }
    /** 겹친 몸을 옆으로 밀어낸다 (벨트스크롤에서 서로 뚫고 지나가지 않게) */
    separate() {
      this.separateFromPlayer();
      this.separateEnemies();
      const list = [this.player, ...this.bodies].filter((b) => b.alive && b.kind !== "bullet" && b.z <= 0);
      for (let i = 0; i < list.length; i++)
        for (let j = i + 1; j < list.length; j++) {
          const a = list[i];
          const b = list[j];
          if (!(a instanceof Fighter) || !(b instanceof Fighter)) continue;
          if (a.phase === "held" || b.phase === "held" || a.phase === "down" || b.phase === "down" || a.phase === "dead" || b.phase === "dead") continue;
          if (a.invuln > 0 || b.invuln > 0) continue;
          if (Math.abs(a.y - b.y) > 30) continue;
          const min = (a.width + b.width) * 0.8;
          const dx = b.x - a.x;
          if (Math.abs(dx) < min) {
            const push = (min - Math.abs(dx)) / 2;
            const s = dx === 0 ? 1 : Math.sign(dx);
            if (a.team === "player" && a instanceof Player && a.move) b.x += s * push * 2;
            else {
              a.x -= s * push;
              b.x += s * push;
            }
          }
        }
    }
    resolveHits() {
      const all = [this.player, ...this.bodies].filter((b) => b.alive);
      const frames = new Map(all.map((b) => [b, b.frame()]));
      for (const a of all) {
        const fa = frames.get(a);
        if (!fa.hit.length) continue;
        for (const h of fa.hit)
          for (const t of all) {
            if (t === a || t.team === a.team || t.team === "neutral" || !t.alive) continue;
            const ft = frames.get(t);
            if (!ft.hurt.some((hb) => overlap(h, hb))) continue;
            const key = `${a.id}:${h.attackId}:${t.id}`;
            if (this.hitMemo.has(key)) continue;
            this.hitMemo.set(key, this.tick);
            this.applyHit(a, t, h);
            if (a.kind === "bullet") a.alive = false;
            if (!a.alive) break;
          }
      }
      if (this.hitMemo.size > 400) {
        for (const [k, v] of this.hitMemo) if (this.tick - v > 120) this.hitMemo.delete(k);
      }
    }
    applyHit(a, t, h) {
      if (a.kind === "bullet") h = progressionHit(h, this.player.progression, "ranged");
      const side = Math.sign(t.x - a.x);
      if (side && side !== h.dir) h = { ...h, dir: side };
      const r = t.applyHit(h, this);
      const at2 = { x: (Math.max(h.x0, t.x - 30) + Math.min(h.x1, t.x + 30)) / 2, y: t.y, z: (h.z0 + h.z1) / 2 };
      if (r.kind === "immune") return;
      const stop = h.stop ?? STOP_DEFAULT[h.react];
      if (r.kind === "guard") {
        a.stop = Math.max(a.stop, 3);
        t.stop = Math.max(t.stop, 3);
        this.fx("spark", at2.x, at2.y, at2.z);
        return;
      }
      a.stop = Math.max(a.stop, stop);
      t.stop = Math.max(t.stop, stop);
      const strength = h.strength ?? (h.react === "down" || h.react === "launch" ? 3 : h.react === "stagger" ? 2 : 1);
      this.fx(strength >= 3 ? "spark-big" : "spark", at2.x, at2.y, at2.z);
      if (this.hitFx) {
        if (strength >= 2) this.fx("ring", at2.x, at2.y, at2.z, String(strength));
        if (strength >= 3) this.shake(6, 3);
      }
      if (h.noise) this.emit({ type: "custom", name: "noise", data: h.noise });
      const p = this.player;
      if (a === p || a.kind === "bullet") {
        p.onConnect(this, !!r.killed);
        if (r.killed) this.kills++;
      }
      const mk = a.kind === "bullet" ? "bullet" : a !== p ? "enemy" : { airkick: "kick", lowkick: "low", dash: "dash", blast: "blast", groundblast: "blast", grabblast: "blast", super: "super", grabknee: "kick" }[p.moveKey] ?? "punch";
      this.emit({ type: "hit", strength, x: at2.x, y: at2.y, z: at2.z, dmg: h.dmg, react: h.react, target: t.kind, combo: a === p ? p.combo : 0, kind: mk });
      if (r.killed && t.team === "enemy" && !(t instanceof Fighter)) this.emit({ type: "enemyDie", x: t.x, kind: t.kind });
      if (a === p && p.moveKey === "grabknee") p.keepHeld();
    }
    stepEncounters() {
      const encs = this.floor.encounters;
      const p = this.player;
      if (this.encActive < 0) {
        const e2 = encs[this.encIndex];
        if (e2 && p.x >= e2.triggerX) {
          this.encActive = this.encIndex;
          this.goT = 0;
          this.encWave = 0;
          this.bounds = [e2.lockFrom, e2.lockTo];
          this.spawnWave();
          this.emit({ type: "combat:start", index: this.encActive });
          if (e2.waves.some((w) => w.some((s) => s.boss))) this.emit({ type: "combat:boss", index: this.encActive });
        }
        return;
      }
      const e = encs[this.encActive];
      this.encBodies = this.encBodies.filter((b) => b.alive && b.hp > 0 && this.threat(b) && !(b instanceof Fighter && (b.phase === "dead" || b.phase === "down" && b.hp <= 0)));
      if (this.encBodies.length) return;
      this.encWave++;
      if (this.encWave < e.waves.length) {
        this.spawnWave();
        return;
      }
      this.emit({ type: "combat:clear", index: this.encActive });
      this.encActive = -1;
      this.encIndex++;
      this.bounds = [0, this.floor.length];
      this.goT = 90;
    }
    spawnWave() {
      const e = this.floor.encounters[this.encActive];
      this.encBodies = e.waves[this.encWave].map((s) => this.spawn(s));
    }
    camera() {
      const p = this.player;
      const [b0, b1] = this.bounds;
      const W5 = view.w;
      if (this.floor.length <= W5) {
        this.camX = -(W5 - this.floor.length) / 2;
        return;
      }
      const target = p.x - W5 / 2 + p.dir * 90;
      let cx = this.camX + (target - this.camX) * 0.14;
      cx = Math.max(b0, Math.min(b1 - W5, cx));
      cx = Math.max(0, Math.min(this.floor.length - W5, cx));
      this.camX = cx;
      if (this.encActive >= 0) p.x = Math.max(cx + 90, Math.min(cx + W5 - 90, p.x));
    }
    /** 보스 체력 막대용 */
    boss() {
      const b = this.bodies.find((x) => x.alive && x.boss);
      return b ? { label: b.label ?? b.kind, hp: b.hp, hpMax: b.hpMax } : null;
    }
  };

    canvas;
    ctx;

    world = null;
    scene;
    paused = false;
    ownCanvas;
    renderer;

    last = 0;

    /** 꼬리표 시작은 주 루프가 돈 뒤에 (start()에서) */
    booted = false;
    start() {
      if (!this.booted) {
        this.booted = true;
        queueMicrotask(() => this.mods.boot?.(this));
      }
      const frame = (now) => {
        if (this.dead) return;
        this.raf = requestAnimationFrame(frame);
        if (!this.last) this.last = now;
        let dt = (now - this.last) / 1e3;
        this.last = now;
        if (dt > 0.25) dt = 0.25;
        this.mods.frame?.(dt);
        this.acc += dt;
        const T2 = 1 / TICK_HZ;
        while (this.acc >= T2) {
          this.acc -= T2;
          this.tickOnce();
        }
        const paused = this.paused || !!this.mods.paused?.();
        if (paused !== this.wasPaused) {
          this.wasPaused = paused;
          this.mods.onPause?.(paused);
        }
        if (!this.mods.covered?.()) {
          const w = this.world;
          const H5 = this.inHub ? view.worldH : SCREEN_H;
          const cssH = this.canvas.getBoundingClientRect().height;
          const fixed = this.inHub && cssH >= 600;
          view.zoom = (fixed ? BASE_ZOOM * (this.canvas.height / cssH) : this.canvas.height / H5) * Math.max(1, charSize.k);
          if (w) view.lane = w.lane;
          const vw = this.canvas.width / view.zoom;
          view.w = this.inHub ? Math.max(SCREEN_W, vw) : SCREEN_W;
          view.hub = this.inHub;
          view.crop = 0;
          const frozen = paused || !!this.mods.blocking?.() || !!this.mods.hq?.isOpen?.();
          if (vw < SCREEN_W && w) {
            const a = frozen ? 1 : this.acc / T2, cam = Math.round(w.prevCamX + (w.camX - w.prevCamX) * a), px = (w.player.px ?? w.player.x) + (w.player.x - (w.player.px ?? w.player.x)) * a, raw = Math.max(0, Math.min(SCREEN_W - vw, px - cam - vw / 2));
            view.crop = Math.round((cam + raw) * view.zoom) / view.zoom - cam;
          }
          view.cropY = 0;
          const vh = this.canvas.height / view.zoom;
          if (this.inHub) {
            this.ctx.fillStyle = "#000";
            this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
          }
          if (vh > H5 + 0.5) {
            view.cropY = -Math.round((vh - H5) / 2 * view.zoom) / view.zoom;
          } else if (vh < H5 - 0.5 && w) {
            const a = frozen ? 1 : this.acc / T2, p = w.player;
            const py = (p.py ?? p.y) + (p.y - (p.py ?? p.y)) * a;
            view.cropY = Math.round(Math.max(0, Math.min(H5 - vh, py - vh * 0.7)) * view.zoom) / view.zoom;
          }
          this.ctx.save();
          this.ctx.scale(view.zoom, view.zoom);
          this.ctx.translate(-view.crop, -view.cropY);
          this.scene.draw(paused ? 1 : this.acc / T2);
          this.ctx.restore();
          this.mods.overlay?.(this.ctx, this.world, dt);
        }
      };
      this.raf = requestAnimationFrame(frame);
    }
    /** 시험용: 틱을 n번 바로 돌린다 */
    advance(n) {
      for (let i = 0; i < n; i++) this.tickOnce();
      this.scene.draw(1);
    }
    tickOnce() {
      const p = this.input.sample();
      if (typeof document !== "undefined" && document.hidden) return;
      if (this.mods.paused?.() && this.scene.name === "battle") return;

    }
    destroy() {

      cancelAnimationFrame(this.raf);

      this.run = null;

      if (this.hqRoot) {

        this.hqRoot.remove();

    }

    }
    idleScene(name) {
      return {
        name,
        tick: () => {
        },
        draw: () => {
          this.ctx.fillStyle = "#05080a";
          this.ctx.fillRect(0, 0, SCREEN_W, SCREEN_H);
          this.mods.drawIdle?.(name, this.ctx, performance.now() / 1e3);
        }
      };
    }
    /* ───────── 출동 ───────── */
    deploy(run, saved) {
      const stage = this.mods.stages.find((s) => s.id === run.stageId) ?? this.mods.stages[0];
      run = { ...run, stageId: stage.id };
      this.inHub = false;
      this.run = run;
      this.runStage = stage;
      if (this.hqRoot) this.hqRoot.hidden = true;
      this.canvas.style.visibility = "";
      const w = new World(run, this.art.clips, stage, this.mods.enemyFactory);
      if (this.mods.itemUse) w.itemUse = this.mods.itemUse;
      for (const l of this.mods.listeners ?? []) w.on(l);
      this.mods.onWorld?.(w, run);
      w.enterFloor(saved && stage.restore ? stage.restore(saved.stage) : stage.firstFloor(run));
      if (saved) w.restore(saved.world);
      this.world = w;
      this.paused = false;
      this.scene = this.battleScene(w);
      this.mods.onPlay?.();
      this.mods.onScene?.("battle");
      this.checkpoint(true);
    }
    /** 전투 도중에는 직전 안전 지점을 유지한다. HP·탄·드랍도 함께 되돌린다. */
    checkpoint(force = false) {
      const w = this.world;
      if (!w || this.inHub || !this.run || !w.canCheckpoint() || !force && this.mods.blocking?.()) return;
      const stage = this.runStage?.snapshot?.();
      if (!stage) return;
      this.mods.saveCheckpoint?.({ version: 1, run: this.run, stage, world: w.snapshot(), savedAt: Date.now() });
    }
    async resume(saved) {
      const stageId = saved.run.stageId ?? this.mods.stages[0]?.id;
      if (saved.version !== 1 || !this.mods.stages.some((s) => s.id === stageId && s.restore)) throw new Error("지원하지 않는 출동 저장");
      if (this.mods.hq) {
        this.showHq();
        await this.mods.hq.ready?.();
      }
      this.deploy({ ...saved.run, stageId }, saved);
      this.world?.say("출동 복원 완료. 전투 중 중단한 경우 마지막 안전 지점부터 이어집니다.", 4);
    }
    /** 출동 끝내기 (철수 메뉴 등에서도 부른다) */
    endRun(how) {
      const w = this.world;
      if (!w || this.inHub) return;
      const result = {
        how,
        floors: Number(w.notes["stage.visitedCount"] ?? w.floors),
        kills: w.kills,
        maxCombo: w.player.maxCombo,
        timeSec: Math.round(w.tick / TICK_HZ),
        hpLeft: Math.max(0, Math.round(w.player.hp)),
        notes: w.notes
      };
      w.emit({ type: "run:end", how });
      this.lastResult = result;
      this.world = null;
      this.paused = false;
      if (this.mods.hq && this.run) {
        const run = this.run;
        this.mods.hq.onReturn(this.mods.toHqResult?.(result, w) ?? result, run);
        this.mods.saveCheckpoint?.(null);
        const back = () => this.showHq();
        if (this.mods.afterRun) {
          this.scene = this.idleScene("cutscene");
          this.mods.afterRun(how, back);
        } else back();
        return;
      }
      this.mods.saveCheckpoint?.(null);
      this.scene = this.resultScene(result);
      this.mods.onScene?.("result");
    }
    battleScene(w) {
      const ownMenu = !this.mods.paused;
      return {
        name: "battle",
        tick: (p) => {
          if (ownMenu && p.pressed.has("pause")) {
            this.paused = !this.paused;
            this.menuIndex = 0;
          }
          if (this.paused) return this.pauseMenu(p, w);
          if (this.inHub && this.hqRoot) {
            const open = !!this.mods.hq?.isOpen?.();
            this.hqRoot.hidden = !open;
            if (open) return;
          }
          if (this.mods.blocking?.() || this.mods.frozen?.()) return;
          w.step(p);
          if (w.ended) {
            const how = w.ended;
            if (how === "next") {
              const next = this.runStage.nextFloor(this.run, w.floor, "next");
              if (next) {
                w.enterFloor(next);
                this.checkpoint(true);
              } else this.endRun("return");
            } else this.endRun(how);
          } else if (w.tick % 60 === 0) this.checkpoint();
        },
        draw: (a) => {
          this.renderer.draw(w, this.paused || this.mods.blocking?.() || this.mods.hq?.isOpen?.() ? 1 : a);
          if (this.paused) this.drawMenu("일시정지", ["계속", w.showBoxes ? "판정 보기 끄기" : "판정 보기 켜기", "철수 (출동 끝내기)", "처음으로"]);
        }
      };
    }
    pauseMenu(p, w) {
      const n = 4;
      if (p.pressed.has("up")) this.menuIndex = (this.menuIndex + n - 1) % n;
      if (p.pressed.has("down")) this.menuIndex = (this.menuIndex + 1) % n;
      if (p.pressed.has("attack") || p.pressed.has("jump") || p.pressed.has("interact")) this.pick(this.menuIndex, w);
    }
    pick(i, w) {
      this.paused = false;
      if (i === 1) w.showBoxes = !w.showBoxes;
      else if (i === 2) this.endRun("withdraw");
      else if (i === 3) this.toTitle();
    }
    drawMenu(title2, items, bg = true) {
      const c = this.ctx;
      if (bg) {
        c.fillStyle = "rgba(0,0,0,0.6)";
        c.fillRect(0, 0, SCREEN_W, SCREEN_H);
        c.fillStyle = "#e8eef0";
        c.textAlign = "center";
        c.font = pfont(44).f;
        c.fillText(title2, SCREEN_W / 2, 210);
      }
      c.textAlign = "center";
      c.font = pfont(30).f;
      items.forEach((t, i) => {
        const y = 300 + i * 64;
        c.fillStyle = i === this.menuIndex ? "#ffd36a" : "rgba(255,255,255,0.1)";
        c.fillRect(SCREEN_W / 2 - 230, y - 40, 460, 54);
        c.fillStyle = i === this.menuIndex ? "#1a1f22" : "#e8eef0";
        c.fillText(t, SCREEN_W / 2, y - 3);
      });
      c.textAlign = "left";
      this.menuItems = items.map((_, i) => ({ y0: 300 + i * 64 - 40, y1: 300 + i * 64 + 14 }));
    }
    onPointer(e) {
      const r = this.canvas.getBoundingClientRect();
      const x = (e.clientX - r.left) / r.width * SCREEN_W;
      const y = (e.clientY - r.top) / r.height * SCREEN_H;
      if (this.scene.name === "result") return this.leaveResult();
      if (Math.abs(x - SCREEN_W / 2) > 230) return;
      const i = this.menuItems.findIndex((m) => y >= m.y0 && y <= m.y1);
      if (i < 0) return;
      if (this.world && this.paused) this.pick(i, this.world);
      else if (this.scene.name === "title") this.titlePick(i);
    }
    /* ───────── 엔진 기본 제목·정산 ───────── */
    titlePick(i) {
      if (i === 0) this.start_new();
      else this.start_new(i === 1 && this.mods.hq ? this.mods.stages[0].id : "dummy");
    }
    titleScene() {
      this.menuIndex = 0;
      this.menuItems = [];
      const first = this.mods.stages[0];
      this.titleItems = this.mods.hq ? ["본부로 출근", `바로 출동 · ${first?.label ?? ""}`, "허수아비 연습"] : [`출동 · ${first?.label ?? ""}`, "허수아비 연습"];
      return {
        name: "title",
        tick: (p) => {
          const n = this.titleItems.length;
          if (p.pressed.has("up")) this.menuIndex = (this.menuIndex + n - 1) % n;
          if (p.pressed.has("down")) this.menuIndex = (this.menuIndex + 1) % n;
          if (p.pressed.has("attack") || p.pressed.has("jump") || p.pressed.has("interact")) this.titlePick(this.menuIndex);
        },
        draw: () => {
          const c = this.ctx;
          c.fillStyle = "#0d1114";
          c.fillRect(0, 0, SCREEN_W, SCREEN_H);
          c.textAlign = "center";
          c.font = "bold 30px monospace";
          c.fillStyle = "#8e9aa1";
          c.fillText("ANOMALY SIMULATION", SCREEN_W / 2, 120);
          c.font = pfont(58).f;
          c.fillStyle = "#e8eef0";
          c.fillText("괴이 ː ANOMALY", SCREEN_W / 2, 190);
          this.drawMenu("", this.titleItems, false);
          c.fillStyle = "#8e9aa1";
          c.font = pfont(20).f;
          c.textAlign = "center";
          const help = [
            "방향키 이동 · A 3연타 · 달리며 A 돌진 · ↓↓ 앉기, 앉아 A 하단 · Space 점프, 공중 A 점프 킥",
            "S 권총(탄창 8, R 재장전) · D 폭파 · Q 가드 · W 필살기 · Shift 짧게 회피 / 길게 달리기",
            "경직된 적에게 걸어 붙으면 잡기 → A 무릎, 방향+A 잡기 폭파, Space 놓기 · 1~4 아이템 · H 판정 보기 · Esc 메뉴"
          ];
          help.forEach((h, i) => c.fillText(h, SCREEN_W / 2, 560 + i * 34));
          c.textAlign = "left";
        }
      };
    }
    leaveResult() {
      if (this.mods.hq) this.showHq();
      else this.toTitle();
    }
    resultScene(r) {
      this.menuIndex = 0;
      this.menuItems = [];
      const how = { return: "귀환", withdraw: "철수", evac: "후송", death: "사망" }[r.how];
      let t = 0;
      return {
        name: "result",
        tick: (p) => {
          if (++t > 20 && (p.pressed.has("attack") || p.pressed.has("jump") || p.pressed.has("interact") || p.pressed.has("pause"))) this.leaveResult();
        },
        draw: () => {
          const c = this.ctx;
          c.fillStyle = "#0d1114";
          c.fillRect(0, 0, SCREEN_W, SCREEN_H);
          c.textAlign = "center";
          c.fillStyle = r.how === "death" ? "#c23030" : r.how === "evac" ? "#ff6a5a" : "#7fe0a0";
          c.font = pfont(64).f;
          c.fillText(how, SCREEN_W / 2, 200);
          c.fillStyle = "#e8eef0";
          c.font = pfont(30).f;
          const lines = [`처치 ${r.kills}`, `최대 콤보 ${r.maxCombo}`, `층 ${r.floors}`, `시간 ${r.timeSec}초`, `남은 체력 ${r.hpLeft}`];
          lines.forEach((l, i) => c.fillText(l, SCREEN_W / 2, 290 + i * 48));
          c.fillStyle = "#8e9aa1";
          c.font = pfont(22).f;
          c.fillText("A 또는 화면을 누르면 처음으로", SCREEN_W / 2, 600);
          c.textAlign = "left";
        }
      };
    }
  };

  // src/core/fieldclock.ts
  function clockText(sec) {
    const m = Math.floor(Math.max(0, sec) / 60) % 1440;
    return `${String(Math.floor(m / 60)).padStart(2, "0")}:${String(m % 60).padStart(2, "0")}`;
  }

  function rankStars(c, grade, rank, x, y, z) {

    /** 마지막 그리기의 창 → 캔버스 변환 (배율, 왼쪽 위) */
    fit = { k: 1, ox: 0, oy: 0 };
    toggle(on = !this.open) {
      this.open = on;
    }
    /** 창이 열려 있을 때 눌린 키 (닫기 키는 껍데기가 먼저 받는다) */
    key(code) {
      const n = CHAR_TABS.length;
      if (code === "KeyQ") this.tab = (this.tab + n - 1) % n;
      else if (code === "KeyE") this.tab = (this.tab + 1) % n;
      const tab = CHAR_TABS[this.tab];
      const dx = code === "ArrowLeft" ? -1 : code === "ArrowRight" ? 1 : 0, dy = code === "ArrowUp" ? -1 : code === "ArrowDown" ? 1 : 0;
      if (!dx && !dy) return;
      if (tab === "기술" && this.skills) {
        const i = this.skill + dx + dy * 3;
        if (i >= 0 && i < this.skills) this.skill = i;
      } else if (tab === "능력" && this.perks) {
        const i = this.perk + dx + dy * 5;
        if (i >= 0 && i < this.perks) this.perk = i;
      }
    }
    /** 캔버스 픽셀 좌표를 눌렀을 때: 탭 이름 위면 그 탭으로 바꾸고 true */
    tap(px, py) {
      const { k, ox, oy } = this.fit;
      const x = (px - ox) / k, y = (py - oy) / k;
      const i = CHAR_TAB_HITS.findIndex((r) => x >= r.x && x < r.x + r.w && y >= r.y && y < r.y + r.h);
      if (i < 0) return false;
      this.tab = i;
      return true;
    }
    draw(c, d3) {
      this.skills = d3.skills.length;
      this.perks = d3.perks.length;
      this.skill = Math.min(this.skill, Math.max(0, this.skills - 1));
      this.perk = Math.min(this.perk, Math.max(0, this.perks - 1));
      c.save();
      const cw = c.canvas.width, ch = c.canvas.height, k = Math.min(cw / 1280, ch / 720);
      c.setTransform(1, 0, 0, 1, 0, 0);
      c.fillStyle = "rgba(0,0,0,0.55)";
      c.fillRect(0, 0, cw, ch);
      this.fit = { k, ox: Math.round((cw - 1280 * k) / 2), oy: Math.round((ch - 720 * k) / 2) };
      c.setTransform(k, 0, 0, k, this.fit.ox, this.fit.oy);
      c.imageSmoothingEnabled = false;
      drawCharWindow(c, { ...d3, pick: d3.pick(this.skill), perkPick: this.perk }, CHAR_TABS[this.tab]);
      c.restore();
    }
  };

  function loadBadges(urlOf) {
    sheets = [];
    cache5.clear();
    for (let k = 1; k <= 6; k++) {

  function badgeOf(name, grade = "희귀") {
    const n = CW_BADGE[name];
    if (n === void 0) return null;
    const key = `${n}|${grade}`;
    const hit2 = cache5.get(key);
    if (hit2) return hit2;
    const sh = sheets[Math.floor(n / 32)];
    if (!sh) return null;
    const cv = document.createElement("canvas");
    cv.width = cv.height = 64;
    const g = cv.getContext("2d");
    g.imageSmoothingEnabled = false;
    g.drawImage(sh, n % 8 * 64, Math.floor(n % 32 / 8) * 64, 64, 64, 0, 0, 64, 64);
    const ramp = RAMP[grade];
    if (ramp) {
      const d3 = g.getImageData(0, 0, 64, 64), px = d3.data;
      for (let i = 0; i < px.length; i += 4) {
        if (!px[i + 3]) continue;
        const t = Math.round(0.299 * px[i] + 0.587 * px[i + 1] + 0.114 * px[i + 2]) / 255;
        const [a, b, u] = t < 0.5 ? [ramp[0], ramp[1], t * 2] : [ramp[1], ramp[2], (t - 0.5) * 2];
        for (let c = 0; c < 3; c++) px[i + c] = Math.round(a[c] + (b[c] - a[c]) * u);
      }
      g.putImageData(d3, 0, 0);
    }
    cache5.set(key, cv);
    return cv;
  }

  function wrap(c, text, max) {
    const lines = [];
    let cur = "";
    for (const ch of [...text]) {
      if (c.measureText(cur + ch).width > max && cur) {
        const sp = cur.lastIndexOf(" ");
        if (sp > cur.length / 2) {
          lines.push(cur.slice(0, sp));
          cur = cur.slice(sp + 1) + ch;
        } else {
          lines.push(cur);
          cur = ch;
        }
      } else cur += ch;
    }
    if (cur) lines.push(cur);
    return lines;
  }
