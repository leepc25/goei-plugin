          a.wait -= dt;
          if (a.wait > 0) return;
          if (!a.goal) return this.resume(a);
          a.act = "walk";
          a.doing = a.def.brain === "transit" ? "지나가는 중" : a.doing;
          return;
        }
      }
    }
    move(a, dt) {
      if (!a.goal) return this.idle(a);
      if (a.plan === "chase") {
        const p = this.player && this.player.scene === a.scene ? this.player : null;
        const t = a.target ? this.actors.find((q) => q.def.id === a.target) : null;
        if (t && t.scene === a.scene) a.goal = { x: t.x - Math.sign(t.x - a.x) * 70, y: t.y };
        else if (p) a.goal = { x: p.x - Math.sign(p.x - a.x) * 110, y: p.y || a.goal.y };
      }
      const fleeing = (a.doing === "도망" || a.doing === "기어서 도망") && a.fleeClip && this.canClip(a, a.fleeClip);
      const casual = a.run <= 1.2 && !fleeing && a.plan !== "chase" && a.doing !== "도망";
      const lead = this.leaderOf(a);
      if (this.nav && !a.detour?.length) {
        if (!this.nav.ok(a.scene, a.goal.x, a.goal.y)) a.goal = this.navSnap(a.scene, a.goal.x, a.goal.y);
        if (!this.nav.ok(a.scene, a.x, a.y)) {
          const c = this.navSnap(a.scene, a.x, a.y);
          a.x = c.x;
          a.y = c.y;
        }
        if (!this.clearLine(a.scene, a.x, a.y, a.goal.x, a.goal.y)) {
          const path = this.navPath(a.scene, a, a.goal);
          if (!path) {
            a.goal = null;
            return this.idle(a);
          }
          if (path.length) {
            a.detour = path.slice(0, -1);
            a.goal = path[path.length - 1];
          }
        }
      }
      if (!a.detour?.length && this.planDetour(a)) return;
      const wp = a.detour?.[0];
      const goal = wp ?? a.goal;
      const dx = goal.x - a.x;
      let speed = (lead?.def.speed ?? a.def.speed ?? WALK) * a.run;
      if (casual) speed *= 1 + 0.07 * Math.sin(this.time * 1.9 + (a.phase ?? 0));
      if (lead) speed *= Math.max(0.75, Math.min(1.25, 1 + (lead.x - a.x) * Math.sign(dx || 1) / 120));
      const step2 = speed * dt;
      if (Math.abs(dx) > 1) a.face = dx > 0 ? 1 : -1;
      this.setClip(a, fleeing ? a.fleeClip : a.run > 1.2 && this.canClip(a, "run") ? "run" : "walk");
      if (casual && !wp) this.sidestep(a, Math.sign(dx), dt);
      const far = !wp && Math.abs(dx) > 60;
      const ty = lead && far ? lead.y + (a.buddyDy ?? 0) : goal.y + (far ? a.side ?? 0 : 0);
      const dy = ty - a.y;
      if (Math.abs(dx) <= step2 && Math.abs(goal.y - a.y) <= step2) {
        a.x = goal.x;
        a.y = goal.y;
        if (wp) return void a.detour.shift();
        a.goal = null;
        a.side = 0;
        return this.arrived(a);
      }
      let vx, vy;
      if (far) {
        vy = Math.sign(dy) * Math.min(Math.abs(dy), step2 * 0.35);
        vx = Math.sign(dx) * Math.min(Math.abs(dx), Math.sqrt(step2 * step2 - vy * vy));
      } else {
        const d3 = Math.hypot(dx, goal.y - a.y);
        vx = dx / d3 * step2;
        vy = (goal.y - a.y) / d3 * step2;
      }
      const ox = a.x, oy = a.y;
      a.x += vx;
      const [top, bot] = this.graph.lane(a.scene);
      a.y = Math.max(Math.min(top, goal.y), Math.min(Math.max(bot, goal.y), a.y + vy));
      if (this.nav && !this.nav.ok(a.scene, a.x, a.y)) {
        const ny = a.y;
        if (Math.abs(ny - oy) > 0.01 && this.nav.ok(a.scene, ox, ny)) {
          a.x = ox;
          a.y = ny;
          return;
        }
        if (Math.abs(a.x - ox) > 0.01 && this.nav.ok(a.scene, a.x, oy)) {
          a.y = oy;
          return;
        }
        for (const d3 of [step2, -step2]) if (this.nav.ok(a.scene, ox, oy + d3) && this.nav.ok(a.scene, ox + vx, oy + d3)) {
          a.x = ox;
          a.y = oy + d3;
          a.detour = [];
          return;
        }
        a.x = ox;
        a.y = oy;
        if (!this.nav.ok(a.scene, ox, oy)) {
          const c = this.navSnap(a.scene, ox, oy);
          a.x = c.x;
          a.y = c.y;
        }
        a.side = 0;
        a.detour = this.clearLine(a.scene, a.x, a.y, a.goal.x, a.goal.y) ? [{ x: a.goal.x, y: a.goal.y }] : [];
        return;
      }
      if (casual && !wp && !lead) this.maybePause(a, dt);
    }
    /** 같이 걷는 일행의 앞사람 (같은 장면에서 걷거나 멈춰 있을 때만). 떨어지면 혼자 걷는다 */
    leaderOf(a) {
      if (!a.buddy) return null;
      const l = this.actors.find((q) => q.def.id === a.buddy);
      if (!l || l.scene !== a.scene || l.fighting || !["walk", "pause"].includes(l.act) || l.doing === "도망") {
        a.buddy = void 0;
        return null;
      }
      return l;
    }
    /** 서로 비켜 가기: 앞쪽 같은 줄에 사람이 있으면 위·아래로 살짝 비켜선다(그 사람 반대쪽, 띠 안). 없으면 천천히 제 줄로 */
    sidestep(a, dir, dt) {
      const ahead = this.actors.find((q) => q !== a && q.scene === a.scene && !q.fighting && q.act !== "door" && q.act !== "gone" && q.def.id !== a.buddy && q.buddy !== a.def.id && (q.x - a.x) * dir > 0 && Math.abs(q.x - a.x) < 110 && Math.abs(q.y - a.y - (a.side ?? 0)) < 22);
      if (!ahead) {
        if (a.side) a.side = Math.abs(a.side) < 1 ? 0 : a.side * (1 - Math.min(1, dt * 0.7));
        return;
      }
      const [top, bot] = this.graph.lane(a.scene);
      let s = Math.sign(a.y - ahead.y) || (this.rng() < 0.5 ? 1 : -1);
      if (a.y + s * 30 < top || a.y + s * 30 > bot) s = -s;
      a.side = ahead.y - a.y + s * 30;
    }
    /** 막힌 곳 우회 경유점. 앞길(가는 방향 200px 안)에 막힌 곳이 있고 지금 줄이 그 안이면 위·아래 바깥으로 돌아간다. 길을 다 막았으면 돌아섬(true) */
    planDetour(a) {
      const list = this.blocks.get(a.scene);
      if (!list || !a.goal) return false;
      const g = a.goal, dir = Math.sign(g.x - a.x);
      if (!dir) return false;
      const [top, bot] = this.graph.lane(a.scene);
      for (const b of list) {
        const near2 = dir > 0 ? b.x0 : b.x1, far = dir > 0 ? b.x1 : b.x0;
        if ((near2 - a.x) * dir < -2 || (near2 - a.x) * dir > 200) continue;
        if ((g.x - near2) * dir < 0) continue;
        if (b.wall || b.y0 <= top + 20 && b.y1 >= bot - 20) {
          const len = this.graph.scene(a.scene)?.length ?? 1280;
          a.detour = [];
          a.route = [];
          const back = { x: Math.max(MARGIN, Math.min(len - MARGIN, near2 - dir * range(this.rng, 120, 400))), y: a.y };
          if (a.exitAt) a.exitAt = { x: dir > 0 ? 30 : len - 30, y: a.y };
          this.walkTo(a, a.exitAt ?? back, "길이 막혀 돌아감", a.exitAt ? "exit" : "none");
          return true;
        }
        const y = a.y + (a.side ?? 0);
        if (y < b.y0 - 14 || y > b.y1 + 14) continue;
        const upY = b.y0 - 18, downY = b.y1 + 18;
        const okUp = upY >= top, okDown = downY <= bot;
        const yy = okUp && (!okDown || a.y - upY < downY - a.y) ? upY : okDown ? downY : upY;
        a.detour = [{ x: near2 - dir * 12, y: yy }, { x: far + dir * 12, y: yy }];
        return false;
      }
      return false;
    }
    /** 가다 잠깐 멈추기: 성급하면 덜, 호기심 많으면 더. 그 인물이 가진 줄만, 때·곳·성향에 맞게 고름 */
    maybePause(a, dt) {
      if (!["none", "exit", "do"].includes(a.plan) || a.route.length || a.drunk) return;
      if ((a.pauseCd = (a.pauseCd ?? 10) - dt) > 0) return;
      const t = a.temper ?? { courage: 0.5, social: 0.5, haste: 0.5, curious: 0.5, sly: 0 };
      a.pauseCd = range(this.rng, 10, 26) * (0.7 + t.haste);
      if (this.rng() > 0.25 + 0.35 * t.curious - 0.2 * t.haste) return;
      const h = this.hour, morning = h != null && h >= 6 && h < 11, night = this.isNight();
      const kind = this.graph.kind(a.scene), outside = kind === "street" || kind === "shop";
      const role = ARCHETYPES[a.def.archetype ?? ""]?.role;
      const buddy = this.actors.find((q) => q.buddy === a.def.id && q.scene === a.scene);
      if (a.patrol) {
        const po = PATROL_PAUSES.filter((o3) => this.canClip(a, o3.clip));
        if (!po.length) return;
        const o2 = pick2(this.rng, po);
        return this.pauseAs(a, o2.clip, o2.doing, range(this.rng, o2.sec[0], o2.sec[1]), buddy);
      }
      const w = {
        phone: 3,
        call: 1.2,
        coffee: morning ? 3 : 0.8,
        limber: morning ? 2 : 0.7,
        smoke: (night ? 2.5 : 1) * (role === "thug" ? 2 : 1),
        browse: 0.5 + 2.5 * t.curious,
        snack: 0.8,
        front: 0.8,
        back: outside ? 1.5 : 0.4,
        spot: t.curious,
        rest: 1,
        idle: 2
      };
      if (buddy && this.canClip(a, "talk") && this.canClip(buddy, "talk")) return this.pauseAs(a, "talk", "일행과 수다", range(this.rng, 4, 9), buddy);
      const opts = PAUSES.filter((o2) => this.canClip(a, o2.clip));
      if (!opts.length) return;
      let r = this.rng() * opts.reduce((n, o2) => n + w[o2.clip], 0);
      const o = opts.find((x) => (r -= w[x.clip]) <= 0) ?? opts[opts.length - 1];
      this.pauseAs(a, o.clip, o.doing, range(this.rng, o.sec[0], o.sec[1]), buddy);
    }
    pauseAs(a, clip, doing, sec, buddy) {
      a.act = "pause";
      a.doing = doing;
      a.wait = sec;
      a.side = 0;
      this.setClip(a, clip);
      a.clipUntil = this.time + sec;
      if (!buddy) return;
      buddy.act = "pause";
      buddy.doing = doing === "일행과 수다" ? doing : "일행 기다림";
      buddy.wait = sec;
      if (doing === "일행과 수다") {
        a.face = buddy.x >= a.x ? 1 : -1;
        buddy.face = -a.face;
      }
      this.setClip(buddy, doing === "일행과 수다" ? "talk" : "idle");
    }
    idle(a, min = 2, max = 7) {
      a.seatPick = void 0;
      a.act = "idle";
      a.doing = "대기";
      a.run = 1;
      a.plan = "none";
      a.wait = range(this.rng, min, max);
      this.setClip(a, "idle");
    }
    /** 잠깐 멈췄다 하던 일로 (몹은 가던 길, 근무형은 자리) */
    resume(a) {
      if (a.post) return this.goPost(a);
      if (a.def.brain === "transit" && a.exitAt) return this.walkTo(a, a.exitAt, "지나가는 중", "exit");
      this.idle(a, 0.5, 2);
    }
    walkTo(a, p, doing, plan = "none") {
      a.act = "walk";
      a.goal = p;
      a.detour = [];
      a.doing = doing;
      a.plan = plan;
    }
    arrived(a) {
      const hop = a.route[0];
      if (hop && hop.from === a.scene) {
        a.act = "door";
        a.doing = `${this.graph.scene(hop.to)?.name ?? hop.to}(으)로 이동 중`;
        a.wait = Math.max(1, hop.sec / 6);
        return;
      }
      switch (a.plan) {
        case "do":
          return this.startTask(a);
        case "post":
          a.act = "post";
          if (a.post?.how === "train") {
            a.doing = `훈련 중 (${{ pushup: "팔굽혀펴기", work: "장비 점검", limber: "기지개" }[a.trainClip ?? ""] ?? "대기"})`;
            a.wait = range(this.rng, 10, 25);
            this.setClip(a, a.trainClip ?? "idle");
            return;
          }
          a.doing = `근무 중 (${a.post?.archetype ?? ""})`;
          a.wait = range(this.rng, 4, 9);
          this.setClip(a, this.canClip(a, "work") ? "work" : "idle");
          return;
        case "patrol":
        case "roam-post":
          return this.goPost(a);
        case "exit":
          a.act = "gone";
          return;
        case "cordon":
          return this.arriveCordon(a);
        case "treat": {
          const pt = this.actors.find((q) => q.def.id === a.patient);
          a.patient = void 0;
          a.act = "pause";
          a.doing = "진료 중";
          a.wait = range(this.rng, 6, 10);
          this.setClip(a, "exam");
          if (pt && pt.scene === a.scene) {
            a.face = pt.x > a.x ? 1 : -1;
            if (pt.act !== "do" && this.canClip(pt, "treat")) {
              pt.act = "pause";
              pt.goal = null;
              pt.doing = "치료받는 중";
              pt.wait = a.wait;
              pt.face = -a.face;
              this.setClip(pt, "treat");
            }
          }
          return;
        }
        case "steal":
          return this.doSteal(a);
        case "harass":
          return this.doHarass(a);
        case "react":
          a.act = "watch";
          a.doing = a.doing === "동료에게 달려감" ? "동료 곁을 지킴" : "살핌";
          a.wait = 3.5;
          a.run = 1;
          this.setClip(a, a.reactClip ?? "idle");
          a.reactClip = void 0;
          return;
        case "chase":
          a.act = "watch";
          a.doing = "막아섬";
          a.wait = 3;
          this.iconOn(a, "!", 2);
          return;
      }
      if (a.def.brain === "transit" && a.exitAt) {
        a.act = "gone";
        return;
      }
      this.idle(a);
    }
    nextLeg(a) {
      const hop = a.route[0];
      if (hop) {
        const d3 = this.graph.doorSpot(a.scene, hop.exit);
        if (d3) return this.walkTo(a, d3, a.doing, a.plan);
      }
      a.route = [];
      if (a.plan === "do" && a.task) return this.walkTo(a, this.randomSpot(a.scene), a.task.label + " 하러", "do");
      if (a.plan === "roam-post") return this.goPost(a);
      this.walkTo(a, this.randomSpot(a.scene), "자리 잡는 중");
    }
    // ── 이름 인물 ──
    /** 호감이 적대 구간인 이름 인물 (10-31 사용자 C4 다안): 주인공을 보면 막아서고(몹 적대 세력과 같은 길), 막아서면 전투 AI가 싸운다.
     *  싸움이 끝난 뒤(이기든 지든) 1분은 다시 막지 않는다 */
    namedBlocks(a) {
      const p = this.player;
      if (!p || p.scene !== a.scene || !this.namedHostile(a.def.name)) return false;
      if (a.plan === "chase" || a.fighting || ["down", "dead", "gone", "door", "argue"].includes(a.act) || a.seat) return false;
      if (this.time < (a.blockAt ?? -1e9) + 60 || Math.abs(p.x - a.x) > 520) return false;
      a.perks ??= ["스탯 뻥튀기", "체력 뻥튀기"];
      a.task = void 0;
      a.run = 1.4;
      this.walkTo(a, { x: p.x + (a.x > p.x ? 70 : -70), y: p.y || a.y }, "주인공을 막아섬", "chase");
      this.iconOn(a, "!", 2);
      return true;
    }
    namedNotice(a) {
      const p = this.player;
      if (!p || p.scene !== a.scene || a.noticeCd > 0 || a.toilet) return;
      if (a.act !== "idle" && a.act !== "walk" && !(a.post && a.act === "post")) return;
      if (a.plan === "do" || a.route.length) return;
      if (Math.abs(p.x - a.x) > NOTICE_PX || Math.abs(p.y - a.y) > 160) return;
      a.noticeCd = range(this.rng, 20, 35);
      a.act = "notice";
      a.doing = "주인공을 봄";
      a.goal = null;
      a.face = p.x > a.x ? 1 : -1;
      a.wait = 2.4;
      this.setClip(a, this.canClip(a, "greet") ? "greet" : this.canClip(a, "bow") ? "bow" : "idle");
      this.iconOn(a, "!", 1.2);
      this.events.push({ kind: "notice", scene: a.scene, who: a.def.name });
    }
    decide(a) {
      if (a.hold) {
        a.act = "idle";
        a.wait = 5;
        a.doing = "기다리는 중";
        return;
      }
      if (a.commute?.state === "out") {
        a.commute = void 0;
        return this.idle(a, 0.2, 0.5);
      }
      if (a.post) return this.goPost(a);
      if (a.def.brain === "transit") return this.idle(a, 1, 3);
      const roam = (a.def.roam ?? [a.def.home]).filter((s) => this.graph.scene(s) && this.sexOk(a, s));
      if (a.def.brain === "needs" && a.needs) {
        const c = choose({
          name: a.def.name,
          needs: a.needs,
          scene: a.scene,
          roam,
          graph: this.graph,
          has: (clip) => this.canClip(a, clip),
          duty: onDuty(this.hour, TEMPER[a.def.name]),
          partnerIn: (s) => this.actors.some((q) => q !== a && q.def.kind === "named" && q.scene === s && (q.act === "idle" || q.act === "walk") && !q.route.length),
          rand: this.rng,
          last: a.last,
          fighter: true
          // 이름 요원은 모두 싸우는 몸
        });
        if (c) {
          a.task = c.act;
          if (c.scene === a.scene) return this.walkTo(a, this.taskSpot(a, c.act), `${c.act.label} 하러`, "do");
          const path = this.graph.route(a.scene, c.scene);
          const d3 = path && this.graph.doorSpot(a.scene, path[0].exit);
          if (path && d3) {
            a.route = path;
            return this.walkTo(a, d3, `${c.act.label} 하러 ${this.graph.scene(c.scene)?.name ?? c.scene}(으)로`, "do");
          }
          a.task = void 0;
        }
      }
      const away = a.scene !== a.def.home;
      const r = this.rng();
      let dest = null;
      if (away && r < 0.45) dest = a.def.home;
      else if (!away && roam.length > 1 && r < 0.18) dest = pick2(this.rng, roam.filter((s) => s !== a.scene));
      if (dest) {
        const path = this.graph.route(a.scene, dest);
        const d3 = path && this.graph.doorSpot(a.scene, path[0].exit);
        if (path && d3) {
          a.route = path;
          return this.walkTo(a, d3, `${this.graph.scene(dest)?.name ?? dest}(으)로 가는 중`);
        }
      }
      this.walkTo(a, this.randomSpot(a.scene), "서성이는 중");
    }
    /** 수다는 상대 옆으로, 앉아서 하는 일은 빈 의자로(서서 하는 줄이 없거나 의자가 있으면), 나머지는 장면 안 빈 곳 */
    taskSpot(a, t) {
      a.seatPick = void 0;
      a.station = void 0;
      a.opt = void 0;
      if (t.id === "train") {
        const st = this.trainSpot(a);
        if (st) return st;
      }
      if (t.id === "shower") {
        const st = this.graph.scene(a.scene)?.lockers?.length ? this.lockerSpot(a) : this.showerSpot(a);
        if (st) return st;
      }
      if (t.id === "pee" || t.id === "poo") {
        const st = this.toiletSpot(a, t.id === "pee" && this.sexOf(a) === "남" ? ["urinal", "stall"] : ["stall"]);
        if (st) return st;
      }
      const look = a.def.look ?? a.def.name;
      if (!t.partner) {
        const st = this.stationSpot(a, t);
        if (st) return st;
      }
      const opts = (ACT_OPTS[t.id] ?? []).filter((o) => this.has(look, `line:${o.line}`));
      if (opts.length && !t.partner) {
        if (t.id === "work") {
          const own = this.ownSeat(a);
          const st = own === null ? null : this.graph.scene(a.scene).seats[own];
          const o = st ? opts.find((q) => q.seat === seatDir(st.face)) : void 0;
          if (st && o && own !== null) {
            a.opt = o;
            a.seatPick = own;
            return { x: st.x, y: st.y };
          }
        }

    // ── 샤워 (11-02, NPC 샤워. 사용자 「탈의실에서 옷을 벗고 샤워실로 이동해야지」「캐비닛·샤워기에 상호작용」) ──

    atSpot(a, tag2) {
      const m = a.station?.match(/^(.+)#([sl]):(.+)$/);
      if (!m || m[1] !== a.scene || m[2] !== tag2) return null;
      const sc = this.graph.scene(a.scene);
      const s = (tag2 === "s" ? sc?.showers : sc?.lockers)?.find((q) => q.id === m[3]);
      if (!s || (tag2 === "s" ? this.showerBusy(a.scene, s.id, a) : this.lockerBusy(a.scene, s.id, a))) return null;
      return Math.hypot(s.x - a.x, s.y - a.y) < 40 ? s : null;

    /** 가장 가까운 같은 성별 탈의실(lockers)·샤워실(showers) */
    nearestWith(a, what) {
      let best = null, bh = 1e9;
      for (const id of this.graph.ids()) {
        const sc = this.graph.scene(id);
        if (!sc?.[what]?.length || !this.sexOk(a, id)) continue;
        const h = id === a.scene ? 0 : this.graph.route(a.scene, id)?.length ?? 1e9;
        if (h < bh) {
          bh = h;
          best = id;
        }
      }
      return best;
    }

      if (sh.step === "start" || sh.step === "back") {

          if (!to) return this.showerAbort(a);
          return this.travel(a, to, "do");

          a.x = l.x;
          a.y = l.y;
          a.face = l.face;

            sh.step = "undress";

            this.setClip(a, "undress");

          } else {
            sh.step = "dress";
            a.doing = "옷 입는 중";
            this.setClip(a, "dress");
            a.wait = this.nOfLine(a, "SH_A") * DRESS_D + 0.4;
          }
          a.clipUntil = this.time + a.wait;
          return;
        }
        const st = this.lockerSpot(a);
        if (st) return this.walkTo(a, st, sh.step === "start" ? "옷 벗으러" : "옷 입으러", "do");
        return this.showerWait(a, "캐비닛 자리 기다림");
      }
      if (sh.step === "go") {
        if (!sc?.showers?.length) {
          const to = this.nearestWith(a, "showers");
          if (!to) return this.showerAbort(a);
          return this.travel(a, to, "do");
        }
        const s = this.atSpot(a, "s");
        if (s) {
          a.x = s.x;
          a.y = s.y;
          a.face = s.face;
          a.act = "do";
          a.plan = "none";
          sh.step = "wash";
          sh.id = s.id;
          a.doing = "샤워 중";
          a.wait = range(this.rng, a.task?.sec[0] ?? 600, a.task?.sec[1] ?? 780);
          this.setClip(a, "shower");
          a.clipUntil = this.time + a.wait;
          return;

        return this.showerWait(a, "샤워 자리 기다림");
      }

    stepStall(a, dt) {
      const tl = a.toilet;
      const t = this.graph.scene(a.scene)?.toilets?.find((q) => q.id === tl.id);
      if (!t?.seat) {
        this.leaveToilet(a);
        return this.idle(a, 1, 3);
      }
      if (tl.phase === "open") {
        if ((tl.t -= dt) > 0) return;
        const f = seatFeet(t.seat, a.def.look ?? a.def.name);
        a.x = f.x;
        a.y = f.y;
        a.face = 1;
        tl.phase = "in";
        this.setClip(a, "poo");
        a.clipUntil = this.time + a.wait;
        return;
      }
      if (tl.phase === "in") {
        if (a.task && a.needs) a.needs[a.task.need] = Math.min(100, a.needs[a.task.need] + a.task.fill * dt);
        if ((a.wait -= dt) > 0) return;
        tl.phase = "leave";
        tl.t = STALL_DOOR_SEC;
        a.x = t.x;
        a.y = t.y;
        this.setClip(a, "idle");
        return;
      }
      if ((tl.t -= dt) > 0) return;
      a.last = a.task?.id;
      a.task = void 0;
      this.leaveToilet(a);
      if (this.goWash(a)) return;
      this.idle(a, 1, 3);
    }

"S" : "N" : (p.face ?? 1) > 0 ? "E" : "W", x: p.x, y: p.y, ...pt?.scene === scene ? { spot: pt.id, phase: pt.phase } : {} });

    cordonBoxes(scene) {
      const [top, bot] = this.graph.lane(scene);
      const boxes = this.corpses(scene).map((c) => ({ x0: c.x - 75, x1: c.x + 75, y0: Math.max(top, c.y - 30), y1: Math.min(bot, c.y + 22) }));
      let merged = true;
      while (merged) {
        merged = false;
        for (let i = 0; i < boxes.length && !merged; i++) for (let j = i + 1; j < boxes.length && !merged; j++) {
          const p = boxes[i], q = boxes[j];
          if (p.x0 - 20 < q.x1 && q.x0 - 20 < p.x1 && p.y0 - 20 < q.y1 && q.y0 - 20 < p.y1) {
            boxes[i] = { x0: Math.min(p.x0, q.x0), x1: Math.max(p.x1, q.x1), y0: Math.min(p.y0, q.y0), y1: Math.max(p.y1, q.y1) };
            boxes.splice(j, 1);
            merged = true;
          }
        }
      }
      return boxes.map((b, i) => ({ id: `cordon${i}`, ...b }));
    }
    setCordon(scene) {
      for (const b of this.obstacles(scene).filter((o) => o.id.startsWith("cordon"))) this.setObstacle(scene, null, b.id);
      if (!this.cordoned.has(scene)) return;
      for (const b of this.cordonBoxes(scene)) this.setObstacle(scene, b);
    }

  var FieldClock = class {
    constructor(start, rules = DEFAULT_CLOCK) {
      this.rules = rules;

    /** 자정부터 분. 본편 시각을 모르면 null (시계를 세지 않는다) */
    minute;

    /** 사건 하나 → 흐른 분 (시계를 모르면 0) */
    advance(ev) {
      let add = 0;
      switch (ev.kind) {
        case "scene":
          add = this.rules.scene;
          this.foughtHere = false;
          break;
        case "region":
          add = this.rules.region;
          this.foughtHere = false;
          break;
        case "talk":
          add = this.rules.talk;
          break;
        case "fight":
          if (this.foughtHere) return 0;
          this.foughtHere = true;
          add = this.rules.fight;
          break;
        case "wait":
          add = Math.max(0, ev.minutes);
          break;
      }
      if (this.minute === null) return 0;
      this.minute += add;
      return add;
    }
    /** 시 (0~24, 날짜는 버림) */
    hour() {
      return this.minute === null ? null : this.minute / 60 % 24;

      if (this.minute === null) return "--:--";

    }
